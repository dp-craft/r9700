// FILE: types.ts
export type Amount =
  | { readonly kind: "fixed"; readonly value: number }
  | { readonly kind: "percent"; readonly pct: number }
  | { readonly kind: "deferred"; readonly load: () => Promise<number> };

export interface LineItem {
  readonly id: string;
  readonly base: number;
  readonly adjustment: Amount;
}

// FILE: pricing.ts
import type { Amount, LineItem } from "./types";

export const PERCENT_BASE = 100;

function computeSyncAdjustment(base: number, amount: Amount): number {
  if (amount.kind === "fixed") return amount.value;
  if (amount.kind === "percent") return (base * amount.pct) / PERCENT_BASE;
  return 0;
}

export async function lineTotal(item: LineItem): Promise<number> {
  let adjustment: number;
  if (item.adjustment.kind === "deferred") {
    adjustment = await item.adjustment.load();
  } else {
    adjustment = computeSyncAdjustment(item.base, item.adjustment);
  }
  return item.base + adjustment;
}

export async function computeTotal(items: readonly LineItem[]): Promise<number> {
  const totals = await Promise.all(items.map(lineTotal));
  return totals.reduce((sum: number, t: number): number => sum + t, 0);
}

// FILE: report.ts
import type { Amount, LineItem } from "./types";
import { lineTotal } from "./pricing";

const KIND_LABEL: Readonly<Record<Amount["kind"], string>> = {
  fixed: "flat",
  percent: "rate",
  deferred: "deferred",
};
const UNKNOWN_LABEL = "other";

export function amountLabel(amount: Amount): string {
  return KIND_LABEL[amount.kind] ?? UNKNOWN_LABEL;
}

export async function reportLine(item: LineItem): Promise<string> {
  const total = await lineTotal(item);
  return `${item.id} [${amountLabel(item.adjustment)}]: ${String(total)}`;
}

export async function buildReport(items: readonly LineItem[]): Promise<readonly string[]> {
  const lines = await Promise.all(items.map(reportLine));
  return lines;
}

// FILE: cache.ts
import type { Clock } from "../../lib/clock";

export interface PriceCache {
  get(key: string, compute: () => Promise<number>): Promise<number>;
}

interface CacheEntry {
  readonly value: number;
  readonly expiresAt: number;
}

const DEFAULT_TTL_MS = 60_000;

export function createPriceCache(clock: Clock, ttlMs: number = DEFAULT_TTL_MS): PriceCache {
  const entries = new Map<string, CacheEntry>();
  const inFlight = new Map<string, Promise<number>>();

  return {
    async get(key: string, compute: () => Promise<number>): Promise<number> {
      const now = clock.now();
      const cached = entries.get(key);
      if (cached !== undefined && cached.expiresAt > now) return cached.value;
      const flight = inFlight.get(key);
      if (flight !== undefined) {
        try {
          const v = await flight;
          if (!entries.has(key)) entries.set(key, { value: v, expiresAt: clock.now() + ttlMs });
          return v;
        } catch { inFlight.delete(key); throw; }
      }
      const p = compute();
      inFlight.set(key, p);
      try {
        const v = await p;
        entries.set(key, { value: v, expiresAt: clock.now() + ttlMs });
        return v;
      } catch { inFlight.delete(key); throw; }
    },
  };
}

// FILE: pricing.test.ts
import { describe, it, expect, vi } from "vitest";
import { computeTotal, lineTotal } from "./pricing";
import { amountLabel, buildReport } from "./report";
import { createPriceCache } from "./cache";

describe("pricing", () => {
  it("should add the fixed value when the adjustment is fixed", async () => {
    const result = await lineTotal({ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } });
    expect(result).toBe(120);
  });

  it("should apply the rate when the adjustment is a percent", async () => {
    const result = await lineTotal({ id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } });
    expect(result).toBe(220);
  });

  it("should await and add the deferred value when the adjustment is deferred", async () => {
    const load = vi.fn().mockResolvedValue(30);
    const item = { id: "c", base: 100, adjustment: { kind: "deferred", load } };
    const result = await lineTotal(item);
    expect(result).toBe(130);
    expect(load).toHaveBeenCalledTimes(1);
  });

  it("should sum the line totals when computing an order total", async () => {
    const items = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } },
    ] as const;
    const result = await computeTotal(items);
    expect(result).toBe(340);
  });
});

describe("report", () => {
  it("should label each adjustment kind correctly", () => {
    expect(amountLabel({ kind: "fixed", value: 5 })).toBe("flat");
    expect(amountLabel({ kind: "percent", pct: 5 })).toBe("rate");
    expect(amountLabel({ kind: "deferred", load: async () => 0 })).toBe("deferred");
  });

  it("should render one line per item with resolved totals", async () => {
    const lines = await buildReport([{ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } }]);
    expect(lines).toEqual(["a [flat]: 120"]);
  });

  it("should render deferred adjustments with resolved totals and deferred label", async () => {
    const lines = await buildReport([{ id: "d", base: 100, adjustment: { kind: "deferred", load: async () => 50 } }]);
    expect(lines).toEqual(["d [deferred]: 150"]);
  });
});

describe("createPriceCache", () => {
  it("should reuse the cached total when the entry is still fresh", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => { calls += 1; return 42; };
    expect(await cache.get("k", compute)).toBe(42);
    expect(await cache.get("k", compute)).toBe(42);
    expect(calls).toBe(1);
  });

  it("should recompute the total when the entry has expired", async () => {
    let t = 0;
    let calls = 0;
    const cache = createPriceCache({ now: (): number => t }, 1000);
    const compute = async (): Promise<number> => { calls += 1; return calls; };
    expect(await cache.get("k", compute)).toBe(1);
    t = 1500;
    expect(await cache.get("k", compute)).toBe(2);
    expect(calls).toBe(2);
  });

  it("should share one computation for concurrent calls to the same fresh key", async () => {
    let calls = 0;
    let resolver: ((value: number) => void) | undefined = undefined;
    const compute = async (): Promise<number> => {
      calls += 1;
      return new Promise<number>(resolve => { resolver = resolve; });
    };
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const p1 = cache.get("shared", compute);
    const p2 = cache.get("shared", compute);
    if (resolver) resolver(99);
    expect(await p1).toBe(99);
    expect(await p2).toBe(99);
    expect(calls).toBe(1);
  });

  it("should not cache failed computations and allow retry", async () => {
    let attempts = 0;
    const compute = async (): Promise<number> => {
      attempts += 1;
      if (attempts === 1) throw new Error("Boom");
      return 88;
    };
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    await expect(cache.get("fail", compute)).rejects.toThrow("Boom");
    expect(attempts).toBe(1);
    expect(await cache.get("fail", compute)).toBe(88);
    expect(attempts).toBe(2);
  });

  it("should keep different keys independent of each other", async () => {
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const gen = (val: number) => async () => val;
    expect(await cache.get("x", gen(10))).toBe(10);
    expect(await cache.get("y", gen(20))).toBe(20);
  });
});
