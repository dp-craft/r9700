// FILE: impl.ts
import type { Clock } from '../../lib/clock';

interface Entry<V> {
  readonly value: V;
  readonly insertedAt: number;
  lastAccessedAt: number;
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

export function createLruCache<K, V>(capacity: number, ttlMs: number, clock: Clock): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();

  function isExpired(entry: Entry<V>): boolean {
    return clock.now() >= entry.insertedAt + ttlMs;
  }

  function refreshRecency(key: K, entry: Entry<V>): void {
    entry.lastAccessedAt = clock.now();
    store.delete(key);
    store.set(key, entry);
  }

  function evictExpired(): void {
    for (const [key, entry] of store) {
      if (isExpired(entry)) {
        store.delete(key);
      }
    }
  }

  function evictOldest(): void {
    evictExpired();
    const oldestKey = store.keys().next().value;
    if (oldestKey !== undefined) {
      store.delete(oldestKey);
    }
  }

  function countLive(): number {
    let count = 0;
    for (const entry of store.values()) {
      if (!isExpired(entry)) {
        count += 1;
      }
    }
    return count;
  }

  const cache: LruCache<K, V> = {
    get(key: K): V | undefined {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) {
        if (entry) {
          store.delete(key);
        }
        return undefined;
      }
      refreshRecency(key, entry);
      return entry.value;
    },

    set(key: K, value: V): void {
      const now = clock.now();
      if (store.has(key)) {
        store.delete(key);
      }
      if (countLive() >= capacity) {
        evictOldest();
      }
      store.set(key, { value, insertedAt: now, lastAccessedAt: now });
    },

    has(key: K): boolean {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) {
        if (entry) {
          store.delete(key);
        }
        return false;
      }
      return true;
    },

    get size(): number {
      return countLive();
    },
  };

  return cache;
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import type { Clock } from '../../lib/clock';
import { createLruCache } from './impl';

function buildFakeClock(initialMs: number): Clock & { tick: (ms: number) => void } {
  let now = initialMs;
  return {
    now: () => now,
    tick: (ms: number) => { now += ms; },
  };
}

describe('createLruCache', () => {
  it('should return undefined when key is absent', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 100, clock);
    expect(cache.get('missing')).toBeUndefined();
  });

  it('should store and retrieve a value when within TTL', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 100, clock);
    cache.set('a', 1);
    expect(cache.get('a')).toBe(1);
  });

  it('should return undefined when entry expires', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 50, clock);
    cache.set('a', 1);
    clock.tick(50);
    expect(cache.get('a')).toBeUndefined();
  });

  it('should report size as 0 when all entries expire', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 50, clock);
    cache.set('a', 1);
    clock.tick(50);
    expect(cache.size).toBe(0);
  });

  it('should return false from has when entry expires', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 50, clock);
    cache.set('a', 1);
    clock.tick(50);
    expect(cache.has('a')).toBe(false);
  });

  it('should refresh recency on get when within TTL', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 50, clock);
    cache.set('a', 1);
    clock.tick(20);
    cache.get('a');
    clock.tick(20);
    expect(cache.get('a')).toBe(1);
  });

  it('should evict least recently used entry when capacity is exceeded', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.get('a');
    cache.set('c', 3);
    expect(cache.get('a')).toBe(1);
    expect(cache.get('b')).toBeUndefined();
  });

  it('should evict expired entries before live entries when capacity is exceeded', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 50, clock);
    cache.set('a', 1);
    clock.tick(60);
    cache.set('b', 2);
    cache.set('c', 3);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBe(2);
    expect(cache.get('c')).toBe(3);
  });

  it('should handle capacity edge of 1 when inserting second key', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(1, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    expect(cache.size).toBe(1);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBe(2);
  });

  it('should update existing key without changing capacity when re-setting', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('a', 10);
    expect(cache.size).toBe(2);
    expect(cache.get('a')).toBe(10);
    expect(cache.get('b')).toBe(2);
  });
});
