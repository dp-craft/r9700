// FILE: impl.ts
import type { Clock } from '../../lib/clock';

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

interface CacheEntry<V> {
  value: V;
  expiresAt: number;
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> {
  const store = new Map<K, CacheEntry<V>>();

  const isExpired = (entry: CacheEntry<V>): boolean =>
    clock.now() >= entry.expiresAt;

  const removeExpired = (): void => {
    for (const [key, entry] of store) {
      if (isExpired(entry)) {
        store.delete(key);
      }
    }
  };

  const evictLeastRecent = (): void => {
    if (store.size <= capacity) return;
    const oldestKey = store.keys().next().value;
    if (oldestKey !== undefined) {
      store.delete(oldestKey);
    }
  };

  return {
    get(key: K): V | undefined {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) {
        store.delete(key);
        return undefined;
      }
      store.delete(key);
      store.set(key, { ...entry, expiresAt: clock.now() + ttlMs });
      return entry.value;
    },

    set(key: K, value: V): void {
      const now = clock.now();
      const entry: CacheEntry<V> = { value, expiresAt: now + ttlMs };

      if (store.has(key)) {
        store.delete(key);
      } else {
        removeExpired();
        evictLeastRecent();
      }
      store.set(key, entry);
    },

    has(key: K): boolean {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) {
        store.delete(key);
        return false;
      }
      return true;
    },

    get size(): number {
      removeExpired();
      return store.size;
    },
  };
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import type { Clock } from '../../lib/clock';

import { createLruCache } from './impl';

const createFakeClock = (initialTime = 0) => {
  let time = initialTime;
  return {
    now: () => time,
    advance: (ms: number) => {
      time += ms;
    },
  };
};

describe('createLruCache', () => {
  it('should return undefined when key is absent', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    expect(cache.get('missing')).toBeUndefined();
  });

  it('should return value when key exists and is not expired', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    cache.set('a', 'val-a');
    expect(cache.get('a')).toBe('val-a');
  });

  it('should return undefined and delete entry when key is expired', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    cache.set('a', 'val-a');
    clock.advance(1001);

    expect(cache.get('a')).toBeUndefined();
    expect(cache.has('a')).toBe(false);
  });

  it('should refresh expiration on live get', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    cache.set('a', 'val-a');
    clock.advance(500);
    cache.get('a');
    clock.advance(600);

    expect(cache.get('a')).toBe('val-a');
  });

  it('should evict least-recently-used entry when capacity is exceeded', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 10000, clock);

    cache.set('a', 'val-a');
    cache.set('b', 'val-b');
    cache.get('a');
    cache.set('c', 'val-c');

    expect(cache.get('a')).toBe('val-a');
    expect(cache.get('b')).toBeUndefined();
    expect(cache.get('c')).toBe('val-c');
  });

  it('should not count expired entries toward size', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    cache.set('a', 'val-a');
    cache.set('b', 'val-b');
    clock.advance(1001);

    expect(cache.size).toBe(0);
  });

  it('should return false for expired entries in has', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    cache.set('a', 'val-a');
    clock.advance(1001);

    expect(cache.has('a')).toBe(false);
  });

  it('should handle capacity edge case: exactly at capacity', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 10000, clock);

    cache.set('a', 'val-a');
    cache.set('b', 'val-b');

    expect(cache.size).toBe(2);
    expect(cache.get('a')).toBe('val-a');
    expect(cache.get('b')).toBe('val-b');
  });

  it('should not evict if capacity is not exceeded', () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, string>(2, 10000, clock);

    cache.set('a', 'val-a');
    cache.set('b', 'val-b');
    cache.set('a', 'val-a-updated');

    expect(cache.size).toBe(2);
    expect(cache.get('a')).toBe('val-a-updated');
    expect(cache.get('b')).toBe('val-b');
  });
});
