// FILE: impl.ts
import type { Clock } from '@/lib/clock';

const DEFAULT_TOKENS = 1;

interface MutableBucket {
  tokens: number;
  lastRefillAt: number;
}

export interface RateLimiter {
  tryAcquire(key: string, tokens?: number): boolean;
}

export function createRateLimiter(
  capacity: number,
  refillPerSec: number,
  clock: Clock
): RateLimiter {
  const store = new Map<string, MutableBucket>();

  const refill = (bucket: MutableBucket): void => {
    const now = clock.now();
    const elapsed = (now - bucket.lastRefillAt) / 1000;
    const added = Math.min(capacity - bucket.tokens, elapsed * refillPerSec);
    bucket.tokens += added;
    bucket.lastRefillAt = now;
  };

  const tryAcquire = (key: string, tokens: number = DEFAULT_TOKENS): boolean => {
    let bucket = store.get(key);
    if (!bucket) {
      bucket = { tokens: capacity, lastRefillAt: clock.now() };
      store.set(key, bucket);
    }
    refill(bucket);
    if (bucket.tokens >= tokens) {
      bucket.tokens -= tokens;
      return true;
    }
    return false;
  };

  return { tryAcquire };
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import type { Clock } from '@/lib/clock';
import { createRateLimiter } from './impl';

const buildFakeClock = (initialTime: number): Clock & { advance(ms: number): void } => {
  let time = initialTime;
  return {
    now: () => time,
    advance: (ms: number) => { time += ms; },
  };
};

describe('createRateLimiter', () => {
  it('should allow first-burst up to capacity when bucket is initially full', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(5, 1, clock);

    expect(limiter.tryAcquire('key-1')).toBe(true);
    expect(limiter.tryAcquire('key-1')).toBe(true);
    expect(limiter.tryAcquire('key-1')).toBe(true);
    expect(limiter.tryAcquire('key-1')).toBe(true);
    expect(limiter.tryAcquire('key-1')).toBe(true);
  });

  it('should reject acquisition when bucket is empty', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(2, 1, clock);

    limiter.tryAcquire('key-1');
    limiter.tryAcquire('key-1');

    expect(limiter.tryAcquire('key-1')).toBe(false);
  });

  it('should lazily refill tokens when time elapses', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(2, 2, clock);

    limiter.tryAcquire('key-1', 2);

    clock.advance(500);
    expect(limiter.tryAcquire('key-1', 1)).toBe(true);
    expect(limiter.tryAcquire('key-1', 1)).toBe(false);

    clock.advance(500);
    expect(limiter.tryAcquire('key-1', 1)).toBe(true);
  });

  it('should cap refill at capacity when excessive time elapses', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(3, 10, clock);

    limiter.tryAcquire('key-1', 3);
    clock.advance(5000);
    expect(limiter.tryAcquire('key-1', 3)).toBe(true);
    expect(limiter.tryAcquire('key-1', 1)).toBe(false);
  });

  it('should maintain isolation when multiple keys are used', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(2, 1, clock);

    expect(limiter.tryAcquire('key-A')).toBe(true);
    expect(limiter.tryAcquire('key-A')).toBe(true);
    expect(limiter.tryAcquire('key-A')).toBe(false);

    expect(limiter.tryAcquire('key-B')).toBe(true);
    expect(limiter.tryAcquire('key-B')).toBe(true);
  });

  it('should deduct custom amount when tokens parameter is provided', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(5, 1, clock);

    expect(limiter.tryAcquire('key-1', 3)).toBe(true);
    expect(limiter.tryAcquire('key-1', 3)).toBe(false);
    expect(limiter.tryAcquire('key-1', 2)).toBe(true);
  });
});
