// FILE: impl.ts
export function createAsyncMemo<K extends string | number, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const pending = new Map<K, Promise<V>>();
  const resolvedCache = new Map<K, V>();

  return async (key: K): Promise<V> => {
    const cachedValue = resolvedCache.get(key);
    if (cachedValue !== undefined) return cachedValue;

    const runningPromise = pending.get(key);
    if (runningPromise !== undefined) return runningPromise;

    const task = fn(key).then(
      (_value: V): V => {
        resolvedCache.set(key, _value);
        pending.delete(key);
        return _value;
      },
      (_err: unknown): never => {
        pending.delete(key);
        throw _err;
      }
    );

    pending.set(key, task);
    return task;
  };
}
