// FILE: impl.ts
import type { Clock } from '../../lib/clock';

interface CacheEntry<V> {
  value: V;
  expiresAt: number;
  recency: number;
}

function isExpired(entry: CacheEntry<unknown>, now: number): boolean {
  return now >= entry.expiresAt;
}

function evictLeastRecent(entries: Map<string, CacheEntry<unknown>>, now: number): void {
  let targetKey: string | undefined;
  let minRecency = Infinity;

  for (const [key, entry] of entries) {
    const recency = isExpired(entry, now) ? Infinity : entry.recency;
    if (recency < minRecency) {
      minRecency = recency;
      targetKey = key;
    }
  }

  if (targetKey !== undefined) entries.delete(targetKey);
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> {
  const entries = new Map<string, CacheEntry<V>>();
  let recencyCounter = 0;

  const now = (): number => clock.now();
  const touchRecency = (): number => ++recencyCounter;

  const purgeExpired = (nowVal: number): void => {
    for (const [k, e] of entries) if (isExpired(e, nowVal)) entries.delete(k);
  };

  const handleEviction = (nowVal: number): void => {
    if (entries.size < capacity) return;
    purgeExpired(nowVal);
    if (entries.size < capacity) return;
    evictLeastRecent(entries, nowVal);
  };

  const getEntry = (key: string): CacheEntry<V> | undefined => entries.get(key);

  return {
    get(key: K): V | undefined {
      const entry = getEntry(String(key));
      if (!entry || isExpired(entry, now())) {
        if (entry) entries.delete(String(key));
        return undefined;
      }
      entry.recency = touchRecency();
      return entry.value;
    },

    set(key: K, value: V): void {
      const strKey = String(key);
      const currentNow = now();

      if (entries.has(strKey)) {
        entries.set(strKey, { value, expiresAt: currentNow + ttlMs, recency: touchRecency() });
        return;
      }

      handleEviction(currentNow);
      entries.set(strKey, { value, expiresAt: currentNow + ttlMs, recency: touchRecency() });
    },

    has(key: K): boolean {
      const entry = getEntry(String(key));
      return !!entry && !isExpired(entry, now());
    },

    get size(): number {
      const currentNow = now();
      let count = 0;
      for (const entry of entries.values()) {
        if (!isExpired(entry, currentNow)) count++;
      }
      return count;
    }
  };
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import type { Clock } from '../../lib/clock';
import { createLruCache } from './impl';

describe('createLruCache', () => {
  it('should return undefined for missing keys', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 1000, clock);

    expect(cache.get('missing')).toBeUndefined();
  });

  it('should store and retrieve values', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 1000, clock);

    cache.set('a', 1);
    expect(cache.get('a')).toBe(1);
  });

  it('should return false for missing keys via has', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 1000, clock);

    expect(cache.has('missing')).toBe(false);
  });

  it('should return true for present keys via has', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 1000, clock);

    cache.set('a', 1);
    expect(cache.has('a')).toBe(true);
  });

  it('should count size as number of live entries', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 1000, clock);

    cache.set('a', 1);
    cache.set('b', 2);
    expect(cache.size).toBe(2);
  });

  it('should not count expired entries toward size', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 50, clock);

    cache.set('a', 1);
    clock.now = () => 160;
    expect(cache.size).toBe(0);
  });

  it('should return undefined for expired entries via get', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 50, clock);

    cache.set('a', 1);
    clock.now = () => 160;
    expect(cache.get('a')).toBeUndefined();
  });

  it('should return false for expired entries via has', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 50, clock);

    cache.set('a', 1);
    clock.now = () => 160;
    expect(cache.has('a')).toBe(false);
  });

  it('should refresh recency on get to prevent LRU eviction', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set('a', 1);
    cache.set('b', 2);
    cache.get('a'); // refresh 'a'

    clock.now = () => 200;
    cache.set('c', 3); // should evict 'b', not 'a'

    expect(cache.get('a')).toBe(1);
    expect(cache.get('b')).toBeUndefined();
    expect(cache.get('c')).toBe(3);
  });

  it('should evict least recently used entry when capacity is reached', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('c', 3); // evicts 'a'

    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBe(2);
    expect(cache.get('c')).toBe(3);
  });

  it('should evict expired entries before live entries when capacity is reached', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(2, 50, clock);

    cache.set('a', 1); // expires at 150
    cache.set('b', 2); // expires at 150
    cache.set('c', 3); // expires at 150

    clock.now = () => 160;
    cache.set('d', 4); // should evict 'a' (expired), not 'b' (live)

    expect(cache.has('a')).toBe(false);
    expect(cache.has('b')).toBe(true);
    expect(cache.has('c')).toBe(true);
    expect(cache.has('d')).toBe(true);
  });

  it('should overwrite existing key without triggering eviction', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(1, 1000, clock);

    cache.set('a', 1);
    cache.set('a', 2);

    expect(cache.size).toBe(1);
    expect(cache.get('a')).toBe(2);
  });

  it('should handle capacity edge case of zero', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(0, 1000, clock);

    cache.set('a', 1);
    expect(cache.size).toBe(0);
    expect(cache.get('a')).toBeUndefined();
  });

  it('should handle capacity edge case of one', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(1, 1000, clock);

    cache.set('a', 1);
    cache.set('b', 2);

    expect(cache.size).toBe(1);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBe(2);
  });

  it('should return undefined for expired entries via get', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 50, clock);

    cache.set('a', 1);
    clock.now = () => 160;
    expect(cache.get('a')).toBeUndefined();
  });

  it('should update expiry time on set for existing key', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 50, clock);

    cache.set('a', 1);
    clock.now = () => 120;
    cache.set('a', 2); // extends expiry to 170
    clock.now = () => 160;
    expect(cache.get('a')).toBe(2);
  });

  it('should return false for missing keys via has', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 1000, clock);

    expect(cache.has('missing')).toBe(false);
  });

  it('should return true for present keys via has', () => {
    const clock: Clock = { now: () => 100 };

    cache.set('a', 1);
    expect(cache.has('a')).toBe(true);
  });

  it('should count size as number of live entries', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 1000, clock);

    cache.set('a', 1);
    cache.set('b', 2);
    expect(cache.size).toBe(2);
  });

  it('should not count expired entries toward size', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 50, clock);

    cache.set('a', 1);
    clock.now = () => 160;
    expect(cache.size).toBe(0);
  });

  it('should return undefined for expired entries via get', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(10, 50, clock);

    cache.set('a', 1);
    clock.now = () => 160;
    expect(cache.get('a')).toBeUndefined();
  });

  it('should return false for expired entries via has', () => {

  });

    expect(cache.get('c')).toBe(3);
  });

  it('should evict least recently used entry when capacity is reached', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('c', 3); // evicts 'a'

    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBe(2);
    expect(cache.get('c')).toBe(3);
  });

  it('should evict expired entries before live entries when capacity is reached', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(2, 50, clock);

    cache.set('a', 1); // expires at 150
    cache.set('b', 2); // expires at 150
    cache.set('c', 3); // expires at 150

    clock.now = () => 160;
    cache.set('d', 4); // should evict 'a' (expired), not 'b' (live)

    expect(cache.has('a')).toBe(false);
    expect(cache.has('b')).toBe(true);
    expect(cache.has('c')).toBe(true);
    expect(cache.has('d')).toBe(true);
  });

  it('should overwrite existing key without triggering eviction', () => {

    const cache = createLruCache<string, number>(0, 1000, clock);

    cache.set('a', 1);
    expect(cache.size).toBe(0);
    expect(cache.get('a')).toBeUndefined();
  });

  it('should handle capacity edge case of one', () => {
    const clock: Clock = { now: () => 100 };
    const cache = createLruCache<string, number>(1, 1000, clock);

    cache.set('a', 1);
    cache.set('b', 2);

    expect(cache.size).toBe(1);
