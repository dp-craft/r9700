// FILE: impl.ts
import type { Clock } from './clock';

interface Entry<V> {
  value: V;
  expiresAt: number;
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();

  const isExpired = (entry: Entry<V>): boolean => clock.now() >= entry.expiresAt;

  const removeEntry = (key: K): void => {
    store.delete(key);
  };

  const moveToEnd = (key: K): void => {
    const entry = store.get(key);
    if (entry) {
      store.delete(key);
      store.set(key, entry);
    }
  };

  const evictLruLive = (): void => {
    for (const key of store.keys()) {
      const entry = store.get(key);
      if (entry && !isExpired(entry)) {
        removeEntry(key);
        return;
      }
      if (entry) removeEntry(key);
    }
  };

  const countLive = (): number => {
    const now = clock.now();
    let count = 0;
    for (const [key, entry] of store.entries()) {
      if (now < entry.expiresAt) {
        count++;
      } else {
        removeEntry(key);
      }
    }
    return count;
  };

  return {
    get(key: K): V | undefined {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) {
        if (entry) removeEntry(key);
        return undefined;
      }
      moveToEnd(key);
      return entry.value;
    },

    set(key: K, value: V): void {
      if (store.has(key)) {
        store.delete(key);
      } else if (countLive() >= capacity) {
        evictLruLive();
      }
      store.set(key, { value, expiresAt: clock.now() + ttlMs });
    },

    has(key: K): boolean {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) {
        if (entry) removeEntry(key);
        return false;
      }
      return true;
    },

    get size(): number {
      return countLive();
    }
  };
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import type { Clock } from './clock';
import { createLruCache } from './impl';

const buildClock = (initialTime: number = 0): Clock & { advance(ms: number): void } => {
  let time = initialTime;
  return {
    now: () => time,
    advance: (ms: number) => { time += ms; },
  };
};

describe('createLruCache', () => {
  it('should return undefined when key is absent', () => {
    const clock = buildClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    expect(cache.get('a')).toBeUndefined();
  });

  it('should return value and refresh recency on get', () => {
    const clock = buildClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.get('a');
    cache.set('c', 3);
    expect(cache.get('b')).toBeUndefined();
    expect(cache.get('a')).toBe(1);
    expect(cache.get('c')).toBe(3);
  });

  it('should evict least-recently-used live entry when capacity is exceeded', () => {
    const clock = buildClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('c', 3);
    expect(cache.size).toBe(2);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBe(2);
    expect(cache.get('c')).toBe(3);
  });

  it('should treat expired entries as absent and not count them in size', () => {
    const clock = buildClock();
    const cache = createLruCache<string, number>(2, 100, clock);
    cache.set('a', 1);
    clock.advance(101);
    expect(cache.has('a')).toBe(false);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.size).toBe(0);
  });

  it('should allow insertion beyond nominal capacity when entries are expired', () => {
    const clock = buildClock();
    const cache = createLruCache<string, number>(2, 100, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    clock.advance(101);
    cache.set('c', 3);
    cache.set('d', 4);
    expect(cache.size).toBe(2);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBeUndefined();
    expect(cache.get('c')).toBe(3);
    expect(cache.get('d')).toBe(4);
  });

  it('should update value and reset TTL on set for existing key', () => {
    const clock = buildClock();
    const cache = createLruCache<string, number>(2, 100, clock);
    cache.set('a', 1);
    clock.advance(50);
    cache.set('a', 2);
    clock.advance(51);
    expect(cache.get('a')).toBe(2);
    expect(cache.size).toBe(1);
  });

  it('should not resurrect expired entries on has check', () => {
    const clock = buildClock();
    const cache = createLruCache<string, number>(2, 100, clock);
    cache.set('a', 1);
    clock.advance(101);
    expect(cache.has('a')).toBe(false);
    expect(cache.has('a')).toBe(false);
  });

  it('should handle capacity edge case of zero by evicting immediately', () => {
    const clock = buildClock();
    const cache = createLruCache<string, number>(0, 1000, clock);
    cache.set('a', 1);
    expect(cache.size).toBe(0);
    expect(cache.get('a')).toBeUndefined();
  });
});
