// FILE: impl.ts
export type Result =
  | { readonly ok: true; readonly value: number }
  | { readonly ok: false; readonly error: string };

type Token =
  | { readonly type: 'NUM'; readonly value: number }
  | { readonly type: 'PLUS' } | { readonly type: 'MINUS' }
  | { readonly type: 'MULT' } | { readonly type: 'DIV' }
  | { readonly type: 'LPAREN' } | { readonly type: 'RPAREN' }
  | { readonly type: 'EOF' };

type Parser = {
  peek: () => Token;
  advance: () => Token;
  expect: (type: string) => Token;
};

const tokenize = (s: string): { readonly tokens: readonly Token[]; readonly error: string | null } => {
  const out: Token[] = [];
  let i = 0;
  while (i < s.length) {
    const c = s[i];
    if (c === ' ' || c === '\t' || c === '\n' || c === '\r') { i++; continue; }
    if (c >= '0' && c <= '9') {
      let n = '';
      while (i < s.length && s[i] >= '0' && s[i] <= '9') n += s[i++];
      out.push({ type: 'NUM', value: Number(n) }); continue;
    }
    const m: Record<string, Token> = {
      '+': { type: 'PLUS' }, '-': { type: 'MINUS' }, '*': { type: 'MULT' },
      '/': { type: 'DIV' }, '(': { type: 'LPAREN' }, ')': { type: 'RPAREN' },
    };
    if (c in m) out.push(m[c]);
    else return { tokens: [], error: `Unexpected character: '${c}'` };
    i++;
  }
  return { tokens: [...out, { type: 'EOF' }], error: null };
};

const makeParser = (tks: readonly Token[]): Parser => {
  let pos = 0;
  const peek = (): Token => tks[pos];
  const advance = (): Token => tks[pos++];
  const expect = (type: string): Token => {
    const t = peek();
    if (t.type !== type) throw new Error(`Expected ${type}, got ${t.type}`);
    return advance();
  };
  return { peek, advance, expect };
};

const parseExpr = (p: Parser): number => {
  let left = parseTerm(p);
  while (p.peek().type === 'PLUS' || p.peek().type === 'MINUS') {
    const op = p.advance();
    const right = parseTerm(p);
    left = op.type === 'PLUS' ? left + right : left - right;
  }
  return left;
};

const parseTerm = (p: Parser): number => {
  let left = parseFactor(p);
  while (p.peek().type === 'MULT' || p.peek().type === 'DIV') {
    const op = p.advance();
    const right = parseFactor(p);
    if (op.type === 'DIV' && right === 0) throw new Error('Division by zero');
    left = op.type === 'MULT' ? left * right : left / right;
  }
  return left;
};

const parseFactor = (p: Parser): number => {
  const t = p.peek();
  if (t.type === 'NUM') return p.advance().value;
  if (t.type === 'LPAREN') {
    p.advance();
    const val = parseExpr(p);
    p.expect('RPAREN');
    return val;
  }
  throw new Error(`Unexpected token: ${t.type}`);
};

export const evaluate = (expr: string): Result => {
  try {
    const { tokens, error } = tokenize(expr);
    if (error) return { ok: false, error };
    const p = makeParser(tokens);
    const val = parseExpr(p);
    if (p.peek().type !== 'EOF') return { ok: false, error: 'Trailing tokens' };
    return { ok: true, value: val };
  } catch (e) {
    return { ok: false, error: e instanceof Error ? e.message : 'Unknown error' };
  }
};

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import { evaluate } from './impl';

describe('evaluate', () => {
  it('should return correct value for simple addition when input is valid', () => {
    expect(evaluate('1 + 2')).toEqual({ ok: true, value: 3 });
  });

  it('should respect multiplication precedence over addition when mixed', () => {
    expect(evaluate('2 + 3 * 4')).toEqual({ ok: true, value: 14 });
  });

  it('should evaluate left-associative subtraction correctly when chained', () => {
    expect(evaluate('10 - 3 - 2')).toEqual({ ok: true, value: 5 });
  });

  it('should evaluate left-associative division correctly when chained', () => {
    expect(evaluate('8 / 2 / 2')).toEqual({ ok: true, value: 2 });
  });

  it('should handle parentheses to override precedence when present', () => {
    expect(evaluate('(2 + 3) * 4')).toEqual({ ok: true, value: 20 });
  });

  it('should return error for division by zero when divisor is 0', () => {
    expect(evaluate('10 / 0')).toEqual({ ok: false, error: 'Division by zero' });
  });

  it('should return error for unbalanced parentheses when missing closing', () => {
    expect(evaluate('(1 + 2')).toEqual({ ok: false, error: expect.stringContaining('RPAREN') });
  });

  it('should return error for unexpected character when invalid symbol is used', () => {
    expect(evaluate('1 + a')).toEqual({ ok: false, error: expect.stringContaining('Unexpected character') });
  });

  it('should return error for missing operand when operator follows operator', () => {
    expect(evaluate('1 + + 2')).toEqual({ ok: false, error: expect.stringContaining('Unexpected token') });
  });

  it('should handle arbitrary whitespace when padded', () => {
    expect(evaluate('  10  *  5  ')).toEqual({ ok: true, value: 50 });
  });
});
