// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  return (key: K): Promise<V> => {
    const existing = cache.get(key);
    if (existing) {
      return existing;
    }
    const p = fn(key).then(
      (value: V): V => {
        cache.set(key, Promise.resolve(value));
        return value;
      },
      (err: unknown): never => {
        cache.delete(key);
        throw err;
      }
    );
    cache.set(key, p);
    return p;
  };
}

// FILE: impl.test.ts
import { describe, expect, it, vi } from 'vitest';

import { createAsyncMemo } from './impl';

describe('createAsyncMemo', () => {
  it('should cache successes and reuse them when called multiple times', async (): Promise<void> => {
    const fn = vi.fn().mockResolvedValue('ok');
    const memo = createAsyncMemo<string, string>(fn);

    const result1 = await memo('key');
    const result2 = await memo('key');

    expect(result1).toBe('ok');
    expect(result2).toBe('ok');
    expect(fn).toHaveBeenCalledTimes(1);
  });

  it('should not cache failures and retry when called again after rejection', async (): Promise<void> => {
    const fn = vi.fn().mockRejectedValueOnce(new Error('fail'));
    const memo = createAsyncMemo<string, string>(fn);

    await expect(memo('key')).rejects.toThrow('fail');
    await expect(memo('key')).resolves.toBe('ok');

    expect(fn).toHaveBeenCalledTimes(2);
  });

  it('should dedup concurrent calls when invoked simultaneously for the same key', async (): Promise<void> => {
    const fn = vi.fn().mockImplementation(async (): Promise<string> => {
      await new Promise<void>(resolve => setTimeout(resolve, 10));
      return 'done';
    });
    const memo = createAsyncMemo<string, string>(fn);

    const p1 = memo('key');
    const p2 = memo('key');

    const [r1, r2] = await Promise.all([p1, p2]);

    expect(r1).toBe('done');
    expect(r2).toBe('done');
    expect(fn).toHaveBeenCalledTimes(1);
  });

  it('should cache successes after a previous rejection when called again', async (): Promise<void> => {
    const fn = vi.fn().mockRejectedValueOnce(new Error('fail'));
    const memo = createAsyncMemo<string, string>(fn);

    await expect(memo('key')).rejects.toThrow('fail');
    const result = await memo('key');

    expect(result).toBe('ok');
    expect(fn).toHaveBeenCalledTimes(2);
  });
});
