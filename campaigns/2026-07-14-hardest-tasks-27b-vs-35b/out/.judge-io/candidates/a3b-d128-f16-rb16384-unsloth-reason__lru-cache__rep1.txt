import type { Clock } from '../../lib/clock';

interface Entry<V> {
  readonly value: V;
  readonly expiresAt: number;
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

export function createLruCache<K, V>(capacity: number, ttlMs: number, clock: Clock): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();
  let currentLive = 0;

  const isExpired = (entry: Entry<V>): boolean => clock.now() >= entry.expiresAt;

  const evictOldestLive = (): void => {
    for (const key of store.keys()) {
      const existing = store.get(key);
      if (existing && !isExpired(existing)) {
        store.delete(key);
        currentLive--;
        return;
      }
    }
  };

  const touchRecency = (key: K): void => {
    const entry = store.get(key);
    if (entry && !isExpired(entry)) {
      store.delete(key);
      store.set(key, entry);
    }
  };

  return ({
    get(key: K): V | undefined {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) return undefined;
      touchRecency(key);
      return entry.value;
    },
    set(key: K, value: V): void {
      const oldEntry = store.get(key);
      if (oldEntry && !isExpired(oldEntry)) touchRecency(key);
      else {
        store.set(key, { value, expiresAt: clock.now() + ttlMs });
        currentLive++;
      }
      while (currentLive > capacity) evictOldestLive();
    },
    has(key: K): boolean {
      const entry = store.get(key);
      return !!entry && !isExpired(entry);
    },
    get size(): number {
      let count = 0;
      for (const entry of store.values()) if (!isExpired(entry)) count++;
      return count;
