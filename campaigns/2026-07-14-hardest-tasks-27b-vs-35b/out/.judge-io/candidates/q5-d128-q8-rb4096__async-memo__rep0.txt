// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  return (key: K): Promise<V> => {
    const existing = cache.get(key);
    if (existing !== undefined) return existing;
    const p = fn(key);
    cache.set(key, p);
    p.catch(() => cache.delete(key));
    return p;
  };
}

// FILE: impl.test.ts
import { beforeEach, describe, expect, it, vi } from 'vitest';

import { createAsyncMemo } from './impl';

describe('createAsyncMemo', () => {
  let fn: ReturnType<typeof vi.fn>;

  beforeEach(() => {
    fn = vi.fn();
  });

  it('should deduplicate concurrent calls and run fn only once', async () => {
    fn.mockImplementation(async (key: string) => {
      await new Promise(resolve => setTimeout(resolve, 50));
      return `result-${key}`;
    });

    const memoized = createAsyncMemo(fn);

    const p1 = memoized('a');
    const p2 = memoized('a');
    const p3 = memoized('a');

    const [r1, r2, r3] = await Promise.all([p1, p2, p3]);

    expect(r1).toBe('result-a');
    expect(r2).toBe('result-a');
    expect(r3).toBe('result-a');
    expect(fn).toHaveBeenCalledTimes(1);
  });

  it('should retry fn when a previous call rejects', async () => {
    fn.mockRejectedValueOnce(new Error('fail'));
    fn.mockResolvedValue('success');

    const memoized = createAsyncMemo(fn);

    await expect(memoized('a')).rejects.toThrow('fail');

    const result = await memoized('a');

    expect(result).toBe('success');
    expect(fn).toHaveBeenCalledTimes(2);
  });

  it('should cache success and not call fn again for the same key', async () => {
    fn.mockResolvedValue('cached-value');

    const memoized = createAsyncMemo(fn);

    const r1 = await memoized('b');
    const r2 = await memoized('b');
    const r3 = await memoized('b');

    expect(r1).toBe('cached-value');
    expect(r2).toBe('cached-value');
    expect(r3).toBe('cached-value');
    expect(fn).toHaveBeenCalledTimes(1);
  });
});
