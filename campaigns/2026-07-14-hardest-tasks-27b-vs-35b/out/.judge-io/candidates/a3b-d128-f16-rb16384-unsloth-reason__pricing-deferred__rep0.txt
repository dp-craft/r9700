// FILE: types.ts
export type Amount =
  | { readonly kind: "fixed"; readonly value: number }
  | { readonly kind: "percent"; readonly pct: number }
  | { readonly kind: "deferred"; readonly load: () => Promise<number> };

export interface LineItem {
  readonly id: string;
  readonly base: number;
  readonly adjustment: Amount;
}

// FILE: pricing.ts
import type { Amount, LineItem } from "./types";

export const PERCENT_BASE = 100;

function pctOf(amount: Amount): number {
  return amount.kind === "percent" ? amount.pct : 0;
}

async function adjustmentTotal(base: number, amount: Amount): Promise<number> {
  if (amount.kind === "deferred") {
    return await amount.load();
  }
  if (amount.kind === "fixed") {
    return amount.value;
  }
  return (base * pctOf(amount)) / PERCENT_BASE;
}

export async function lineTotal(item: LineItem): Promise<number> {
  return item.base + await adjustmentTotal(item.base, item.adjustment);
}

export async function computeTotal(items: readonly LineItem[]): Promise<number> {
  const rowTotals = await Promise.all(
    items.map(async (i) => lineTotal(i))
  );
  return rowTotals.reduce((sum: number, t: number): number => sum + t, 0);
}

// FILE: report.ts
import type { Amount, LineItem } from "./types";
import { lineTotal } from "./pricing";

const KIND_LABEL: Partial<Record<Amount["kind"], string>> = {
  fixed: "flat",
  percent: "rate",
};
const UNKNOWN_LABEL = "other";

export function amountLabel(amount: Amount): string {
  if (amount.kind === "deferred") return "deferred";
  const label = KIND_LABEL[amount.kind];
  return label ?? UNKNOWN_LABEL;
}

export async function reportLine(item: LineItem): Promise<string> {
  const total = await lineTotal(item);
  return `${item.id} [${amountLabel(item.adjustment)}]: ${String(total)}`;
}

export async function buildReport(items: readonly LineItem[]): Promise<readonly string[]> {
  return Promise.all(
    items.map(async (i) => reportLine(i))
  );
}

// FILE: cache.ts
import type { Clock } from "../../lib/clock";

export interface PriceCache {
  /** Return the cached total for `key` if fresh, else run `compute` and cache its result. */
  get(key: string, compute: () => Promise<number>): Promise<number>;
}

interface CacheEntry {
  readonly value: number;
  readonly expiresAt: number;
}

const DEFAULT_TTL_MS = 60_000;

export function createPriceCache(clock: Clock, ttlMs: number = DEFAULT_TTL_MS): PriceCache {
  const entries = new Map<string, CacheEntry>();
  const pendingComputations = new Map<string, Promise<number>>();

  return {
    async get(key: string, compute: () => Promise<number>): Promise<number> {
      const hit = entries.get(key);
      if (hit !== undefined && hit.expiresAt > clock.now()) {
        return hit.value;
      }

      let pendingPromise = pendingComputations.get(key);
      if (pendingPromise) {
        return await pendingPromise;
      }

      const chain = async (): Promise<number> => {
        try {
          const value = await compute();
          entries.set(key, { value, expiresAt: clock.now() + ttlMs });
          return value;
        } catch (err) {
          entries.delete(key);
          throw err;
        } finally {
          pendingComputations.delete(key);
        }
      };

      pendingPromise = chain();
      pendingComputations.set(key, pendingPromise);
      return pendingPromise;
    },
  };
}

// FILE: pricing.test.ts
import { describe, it, expect } from "vitest";
import { computeTotal, lineTotal } from "./pricing";
import { amountLabel, buildReport, reportLine } from "./report";
import { createPriceCache } from "./cache";
import type { LineItem, Amount } from "./types";

describe("pricing", () => {
  it("should add the fixed value when the adjustment is fixed", async () => {
    expect(
      await lineTotal({ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } })
    ).toBe(120);
  });

  it("should apply the rate when the adjustment is a percent", async () => {
    expect(
      await lineTotal({ id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } })
    ).toBe(220);
  });

  it("should resolve deferred loads correctly", async () => {
    const item: LineItem = {
      id: "c",
      base: 50,
      adjustment: { kind: "deferred", load: () => Promise.resolve(30) },
    };
    expect(await lineTotal(item)).toBe(80);
  });

  it("should sum the line totals when computing an order total", async () => {
    const items: readonly LineItem[] = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } },
    ];
    expect(await computeTotal(items)).toBe(340);
  });

  it("should sum deferred line totals when computing an order total", async () => {
    const items: readonly LineItem[] = [
      { id: "a1", base: 100, adjustment: { kind: "deferred", load: () => Promise.resolve(5) } },
      { id: "b1", base: 200, adjustment: { kind: "fixed", value: 10 } },
    ];
    expect(await computeTotal(items)).toBe(315);
  });
});

describe("report", () => {
  it("should label each adjustment when building the report", () => {
    expect(amountLabel({ kind: "fixed", value: 5 })).toBe("flat");
    expect(amountLabel({ kind: "percent", pct: 5 })).toBe("rate");
    expect(amountLabel({ kind: "deferred", load: async () => 0 })).toBe("deferred");
  });

  it("should resolve deferred values when rendering report lines", async () => {
    const item: LineItem = {
      id: "x",
      base: 10,
      adjustment: { kind: "deferred", load: async () => 40 },
    };
    expect(await reportLine(item)).toBe("x [deferred]: 50");
  });

  it("should render one line per item when building the report", async () => {
    const items: readonly LineItem[] = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
    ];
    expect(await buildReport(items)).toEqual(["a [flat]: 120"]);
  });

  it("should resolve all adjusted values when building the report concurrently", async () => {
    const items: readonly LineItem[] = [
      { id: "r1", base: 50, adjustment: { kind: "deferred", load: async () => 5 } },
      { id: "r2", base: 50, adjustment: { kind: "fixed", value: 5 } },
    ];
    expect(await buildReport(items)).toEqual(["r1 [deferred]: 55", "r2 [flat]: 55"]);
  });
});

describe("createPriceCache", () => {
  it("should reuse the cached total when the entry is still fresh", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = (): Promise<number> => {
      calls += 1;
      return Promise.resolve(42);
    };
    expect(await cache.get("k", compute)).toBe(42);
    expect(await cache.get("k", compute)).toBe(42);
    expect(calls).toBe(1);
  });

  it("should recompute the total when the entry has expired", async () => {
    let t = 0;
    let calls = 0;
    const cache = createPriceCache({ now: (): number => t }, 1000);
    const compute = (): Promise<number> => {
      calls += 1;
      return Promise.resolve(calls);
    };
    expect(await cache.get("k", compute)).toBe(1);
    t = 1500;
    expect(await cache.get("k", compute)).toBe(2);
    expect(calls).toBe(2);
  });

  it("should share ONE computation for concurrent calls to the same fresh key", async () => {
    let callCount = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    
    const compute = (): Promise<number> => {
      callCount += 1;
      return Promise.resolve(callCount * 10);
    };

    const p1 = cache.get("dedup-key", compute);
    const p2 = cache.get("dedup-key", compute);

    expect(await p1).toBe(10);
    expect(await p2).toBe(10);
    expect(callCount).toBe(1);
  });

  it("should retry when a computation fails", async () => {
    const cache = createPriceCache({ now: (): number => 0 }, 5000);
    let attempts = 0;
    const compute = (): Promise<number> => {
      attempts += 1;
      if (attempts === 1) return Promise.reject(new Error("fail"));
      return Promise.resolve(99);
    };

    await expect(cache.get("retry-key", compute)).rejects.toBeDefined();
    expect(await cache.get("retry-key", compute)).toBe(99);
    expect(attempts).toBe(2);
  });

  it("should keep different keys independent when caching totals", async () => {
    let callsA = 0;
    let callsB = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    await cache.get("key-a-calc", () => Promise.resolve(++callsA));
    await cache.get("key-b-calc", () => Promise.resolve(++callsB));
    expect(callsA).toBe(1);
    expect(callsB).toBe(1);
  });
});
