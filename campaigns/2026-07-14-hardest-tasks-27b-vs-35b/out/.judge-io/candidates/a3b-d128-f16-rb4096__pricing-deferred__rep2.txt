// FILE: types.ts
export type Amount =
  | { readonly kind: "fixed"; readonly value: number }
  | { readonly kind: "percent"; readonly pct: number }
  | { readonly kind: "deferred"; readonly load: () => Promise<number> };

export interface LineItem {
  readonly id: string;
  readonly base: number;
  readonly adjustment: Amount;
}

// FILE: pricing.ts
import type { Amount, LineItem } from "./types";

export const PERCENT_BASE = 100;

function pctOf(amount: Amount): number {
  return amount.kind === "percent" ? amount.pct : 0;
}

function syncAdjustmentOf(base: number, amount: Amount): number {
  if (amount.kind === "fixed") {
    return amount.value;
  }
  return (base * pctOf(amount)) / PERCENT_BASE;
}

export async function lineTotal(item: LineItem): Promise<number> {
  const adj =
    item.adjustment.kind === "deferred"
      ? await item.adjustment.load()
      : syncAdjustmentOf(item.base, item.adjustment);
  return item.base + adj;
}

export async function computeTotal(items: readonly LineItem[]): Promise<number> {
  let sum = 0;
  for (const item of items) {
    sum += await lineTotal(item);
  }
  return sum;
}

// FILE: report.ts
import type { Amount, LineItem } from "./types";
import { lineTotal } from "./pricing";

const KIND_LABEL: Partial<Record<Amount["kind"], string>> = {
  fixed: "flat",
  percent: "rate",
  deferred: "deferred",
};
const UNKNOWN_LABEL = "other";

export function amountLabel(amount: Amount): string {
  return KIND_LABEL[amount.kind] ?? UNKNOWN_LABEL;
}

export async function reportLine(item: LineItem): Promise<string> {
  const total = await lineTotal(item);
  return `${item.id} [${amountLabel(item.adjustment)}]: ${String(total)}`;
}

export async function buildReport(items: readonly LineItem[]): Promise<readonly string[]> {
  return Promise.all(items.map(reportLine));
}

// FILE: cache.ts
import type { Clock } from "../../lib/clock";

export interface PriceCache {
  get(key: string, compute: () => Promise<number>): Promise<number>;
}

interface CacheEntry {
  readonly value: number;
  readonly expiresAt: number;
}

const DEFAULT_TTL_MS = 60_000;

export function createPriceCache(clock: Clock, ttlMs: number = DEFAULT_TTL_MS): PriceCache {
  const entries = new Map<string, CacheEntry>();
  const pending = new Map<string, Promise<number>>();

  return {
    async get(key: string, compute: () => Promise<number>): Promise<number> {
      const hit = entries.get(key);
      if (hit !== undefined && hit.expiresAt > clock.now()) {
        return hit.value;
      }

      const existingPending = pending.get(key);
      if (existingPending !== undefined) {
        return existingPending;
      }

      const promise = compute().then(
        (value) => {
          entries.set(key, { value, expiresAt: clock.now() + ttlMs });
          pending.delete(key);
          return value;
        },
        (error) => {
          pending.delete(key);
          throw error;
        }
      );

      pending.set(key, promise);
      return promise;
    },
  };
}

// FILE: pricing.test.ts
import { describe, it, expect } from "vitest";
import { computeTotal, lineTotal } from "./pricing";
import { amountLabel, buildReport, reportLine } from "./report";
import { createPriceCache } from "./cache";

describe("pricing", () => {
  it("should add the fixed value when the adjustment is fixed", async () => {
    expect(
      await lineTotal({ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } })
    ).toBe(120);
  });

  it("should apply the rate when the adjustment is a percent", async () => {
    expect(
      await lineTotal({ id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } })
    ).toBe(220);
  });

  it("should resolve the load function when the adjustment is deferred", async () => {
    expect(
      await lineTotal({ id: "c", base: 50, adjustment: { kind: "deferred", load: async () => 25 } })
    ).toBe(75);
  });

  it("should sum the line totals when computing an order total", async () => {
    const items = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } },
    ] as const;
    expect(await computeTotal(items)).toBe(340);
  });

  it("should compute total with deferred adjustments", async () => {
    const items = [
      { id: "d", base: 10, adjustment: { kind: "deferred", load: async () => 5 } },
    ] as const;
    expect(await computeTotal(items)).toBe(15);
  });
});

describe("report", () => {
  it("should label each adjustment when building the report", () => {
    expect(amountLabel({ kind: "fixed", value: 5 })).toBe("flat");
    expect(amountLabel({ kind: "percent", pct: 5 })).toBe("rate");
    expect(amountLabel({ kind: "deferred", load: async () => 0 })).toBe("deferred");
  });

  it("should render one line per item when building the report", async () => {
    const lines = await buildReport([
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
    ]);
    expect(lines).toEqual(["a [flat]: 120"]);
  });

  it("should render deferred adjustment with resolved total", async () => {
    const lines = await buildReport([
      { id: "b", base: 100, adjustment: { kind: "deferred", load: async () => 50 } },
    ]);
    expect(lines).toEqual(["b [deferred]: 150"]);
  });
});

describe("createPriceCache", () => {
  it("should reuse the cached total when the entry is still fresh", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return 42;
    };
    expect(await cache.get("k", compute)).toBe(42);
    expect(await cache.get("k", compute)).toBe(42);
    expect(calls).toBe(1);
  });

  it("should recompute the total when the entry has expired", async () => {
    let t = 0;
    let calls = 0;
    const cache = createPriceCache({ now: (): number => t }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return calls;
    };
    expect(await cache.get("k", compute)).toBe(1);
    t = 1500;
    expect(await cache.get("k", compute)).toBe(2);
    expect(calls).toBe(2);
  });

  it("should share one computation when multiple concurrent get calls request the same fresh key", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      await new Promise((resolve) => setTimeout(resolve, 10));
      return 99;
    };
    const [r1, r2, r3] = await Promise.all([
      cache.get("shared", compute),
      cache.get("shared", compute),
      cache.get("shared", compute),
    ]);
    expect(r1).toBe(99);
    expect(r2).toBe(99);
    expect(r3).toBe(99);
    expect(calls).toBe(1);
  });

  it("should not cache failed computations and retry on next call", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      if (calls === 1) throw new Error("boom");
      return 42;
    };
    await expect(cache.get("fail", compute)).rejects.toThrow("boom");
    expect(await cache.get("fail", compute)).toBe(42);
    expect(calls).toBe(2);
  });

  it("should keep different keys independent", async () => {
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const computeA = async (): Promise<number> => 10;
    const computeB = async (): Promise<number> => 20;
    expect(await cache.get("a", computeA)).toBe(10);
    expect(await cache.get("b", computeB)).toBe(20);
  });
});
