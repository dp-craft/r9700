// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  return (key: K): Promise<V> => {
    const cached = cache.get(key);
    if (cached) {
      return cached;
    }
    const p = fn(key).catch((err: unknown): never => {
      cache.delete(key);
      throw err;
    });
    cache.set(key, p);
    return p;
  };
}

// FILE: impl.test.ts
import { describe, expect, it, vi } from 'vitest';
import { createAsyncMemo } from './impl';

describe('createAsyncMemo', () => {
  it('should share in-flight computation when key is requested concurrently', async (): Promise<void> => {
    let callCount = 0;
    const fn: (key: string) => Promise<string> = vi.fn((key: string): Promise<string> => {
      callCount += 1;
      return new Promise((resolve): void => {
        setTimeout(() => resolve(`res-${key}`), 50);
      });
    });

    const memoized: (key: string) => Promise<string> = createAsyncMemo(fn);
    const p1: Promise<string> = memoized('key-1');
    const p2: Promise<string> = memoized('key-1');

    await Promise.all([p1, p2]);

    expect(callCount).toBe(1);
    expect(p1).toBe(p2);
  });

  it('should remove failure from cache when async computation rejects', async (): Promise<void> => {
    let callCount = 0;
    const fn: (key: string) => Promise<string> = vi.fn((key: string): Promise<string> => {
      callCount += 1;
      return Promise.reject(new Error('fail'));
    });

    const memoized: (key: string) => Promise<string> = createAsyncMemo(fn);

    await expect(memoized('err-key')).rejects.toThrow('fail');
    await expect(memoized('err-key')).rejects.toThrow('fail');

    expect(callCount).toBe(2);
  });

  it('should reuse resolved value when key is requested again', async (): Promise<void> => {
    let callCount = 0;
    const fn: (key: string) => Promise<string> = vi.fn((key: string): Promise<string> => {
      callCount += 1;
      return Promise.resolve(`val-${key}`);
    });

    const memoized: (key: string) => Promise<string> = createAsyncMemo(fn);

    const r1: string = await memoized('succ-key');
    const r2: string = await memoized('succ-key');

    expect(r1).toBe('val-succ-key');
    expect(r2).toBe('val-succ-key');
    expect(callCount).toBe(1);
  });
});
