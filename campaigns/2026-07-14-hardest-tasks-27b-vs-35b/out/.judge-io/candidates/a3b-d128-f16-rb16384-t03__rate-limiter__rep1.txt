// FILE: impl.ts
import type { Clock } from '../../lib/clock';

const DEFAULT_TOKENS = 1;
const MS_PER_SEC = 1000;

interface Bucket {
  tokens: number;
  lastRefillAt: number;
}

export interface RateLimiter {
  readonly tryAcquire: (key: string, tokens?: number) => boolean;
}

function createRateLimiter(capacity: number, refillPerSec: number, clock: Clock): RateLimiter {
  const buckets = new Map<string, Bucket>();

  const refill = (bucket: Bucket): void => {
    const now = clock.now();
    const elapsedMs = now - bucket.lastRefillAt;
    if (elapsedMs <= 0) return;
    const elapsedSec = elapsedMs / MS_PER_SEC;
    const added = Math.min(elapsedSec * refillPerSec, capacity - bucket.tokens);
    bucket.tokens += added;
    bucket.lastRefillAt = now;
  };

  const acquire = (key: string, tokens: number): boolean => {
    const existing = buckets.get(key);
    if (!existing) {
      buckets.set(key, { tokens: capacity, lastRefillAt: clock.now() });
      return true;
    }
    refill(existing);
    if (existing.tokens < tokens) return false;
    existing.tokens -= tokens;
    return true;
  };

  return {
    tryAcquire: (key: string, tokens?: number): boolean => {
      return acquire(key, tokens ?? DEFAULT_TOKENS);
    },
  };
}

export { createRateLimiter };

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import type { Clock } from '../../lib/clock';

import { createRateLimiter } from './impl';

const createFakeClock = (initialTime: number): Clock & { setTime: (t: number) => void } => {
  let time = initialTime;
  return {
    now: (): number => time,
    setTime: (t: number): void => {
      time = t;
    },
  };
};

describe('createRateLimiter', () => {
  describe('first-burst', () => {
    it('should allow first-burst acquisition when capacity is requested', () => {
      const clock = createFakeClock(0);
      const limiter = createRateLimiter(5, 1, clock);

      expect(limiter.tryAcquire('key-1', 5)).toBe(true);
    });

    it('should reject first-burst acquisition when request exceeds capacity', () => {
      const clock = createFakeClock(0);
      const limiter = createRateLimiter(5, 1, clock);

      expect(limiter.tryAcquire('key-1', 6)).toBe(false);
    });
  });

  describe('lazy refill over time', () => {
    it('should refill tokens lazily when elapsed time passes', () => {
      const clock = createFakeClock(0);
      const limiter = createRateLimiter(10, 2, clock);

      limiter.tryAcquire('key-1', 10);
      clock.setTime(1000);

      expect(limiter.tryAcquire('key-1', 2)).toBe(true);
    });

    it('should cap lazy refill at capacity when time exceeds refill period', () => {
      const clock = createFakeClock(0);
      const limiter = createRateLimiter(5, 1, clock);

      limiter.tryAcquire('key-1', 5);
      clock.setTime(10000);

      expect(limiter.tryAcquire('key-1', 5)).toBe(true);
      expect(limiter.tryAcquire('key-1', 5)).toBe(false);
    });
  });

  describe('empty-bucket rejection', () => {
    it('should reject empty-bucket acquisition when tokens are insufficient', () => {
      const clock = createFakeClock(0);
      const limiter = createRateLimiter(5, 1, clock);

      limiter.tryAcquire('key-1', 5);
      expect(limiter.tryAcquire('key-1', 1)).toBe(false);
    });
  });

  describe('per-key isolation', () => {
    it('should maintain per-key isolation when keys are independent', () => {
      const clock = createFakeClock(0);
      const limiter = createRateLimiter(5, 1, clock);

      limiter.tryAcquire('key-a', 5);
      expect(limiter.tryAcquire('key-b', 5)).toBe(true);
    });

    it('should isolate per-key buckets when one key is exhausted', () => {
      const clock = createFakeClock(0);
      const limiter = createRateLimiter(3, 1, clock);

      limiter.tryAcquire('x', 3);
      expect(limiter.tryAcquire('y', 3)).toBe(true);
    });
  });

  describe('default tokens', () => {
    it('should default to single token acquisition when tokens argument is omitted', () => {
      const clock = createFakeClock(0);
      const limiter = createRateLimiter(5, 1, clock);

      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1', 4)).toBe(false);
    });
  });
});
