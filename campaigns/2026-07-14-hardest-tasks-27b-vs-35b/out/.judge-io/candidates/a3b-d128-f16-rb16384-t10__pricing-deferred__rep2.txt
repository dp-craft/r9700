// FILE: types.ts
export type Amount =
  | { readonly kind: "fixed"; readonly value: number }
  | { readonly kind: "percent"; readonly pct: number }
  | { readonly kind: "deferred"; readonly load: () => Promise<number> };

export interface LineItem {
  readonly id: string;
  readonly base: number;
  readonly adjustment: Amount;
}

// FILE: pricing.ts
import type { Amount, LineItem } from "./types";

export const PERCENT_BASE = 100;

function pctOf(amount: Amount): number {
  return amount.kind === "percent" ? amount.pct : 0;
}

async function adjustmentValue(base: number, amount: Amount): Promise<number> {
  if (amount.kind === "fixed") {
    return amount.value;
  }
  if (amount.kind === "percent") {
    return (base * pctOf(amount)) / PERCENT_BASE;
  }
  return amount.load();
}

export async function lineTotal(item: LineItem): Promise<number> {
  return item.base + await adjustmentValue(item.base, item.adjustment);
}

export async function computeTotal(items: readonly LineItem[]): Promise<number> {
  let sum = 0;
  for (const item of items) {
    sum += await lineTotal(item);
  }
  return sum;
}

// FILE: report.ts
import type { Amount, LineItem } from "./types";
import { lineTotal } from "./pricing";

const KIND_LABEL: Partial<Record<Amount["kind"], string>> = {
  fixed: "flat",
  percent: "rate",
  deferred: "deferred",
};
const UNKNOWN_LABEL = "other";

export function amountLabel(amount: Amount): string {
  return KIND_LABEL[amount.kind] ?? UNKNOWN_LABEL;
}

export async function reportLine(item: LineItem): Promise<string> {
  const total = await lineTotal(item);
  return `${item.id} [${amountLabel(item.adjustment)}]: ${String(total)}`;
}

export async function buildReport(items: readonly LineItem[]): Promise<readonly string[]> {
  return Promise.all(items.map((item: LineItem): Promise<string> => reportLine(item)));
}

// FILE: cache.ts
import type { Clock } from "../../lib/clock";

export interface PriceCache {
  get(key: string, compute: () => Promise<number>): Promise<number>;
}

interface CacheEntry {
  readonly promise: Promise<number>;
  readonly expiresAt: number;
}

const DEFAULT_TTL_MS = 60_000;

export function createPriceCache(clock: Clock, ttlMs: number = DEFAULT_TTL_MS): PriceCache {
  const entries = new Map<string, CacheEntry>();

  return {
    async get(key: string, compute: () => Promise<number>): Promise<number> {
      const now = clock.now();
      const existing = entries.get(key);
      if (existing !== undefined && existing.expiresAt > now) {
        return existing.promise;
      }

      const expiresAt = now + ttlMs;
      const promise = compute().then(
        (value) => {
          entries.set(key, { promise, expiresAt });
          return value;
        },
        () => {
          entries.delete(key);
          throw new Error("Compute failed");
        }
      );

      entries.set(key, { promise, expiresAt });
      return promise;
    },
  };
}

// FILE: pricing.test.ts
import { describe, it, expect, vi } from "vitest";
import { computeTotal, lineTotal } from "./pricing";
import { amountLabel, buildReport } from "./report";
import { createPriceCache } from "./cache";

describe("pricing", () => {
  it("should add the fixed value when the adjustment is fixed", async () => {
    expect(await lineTotal({ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } })).toBe(120);
  });
  it("should apply the rate when the adjustment is a percent", async () => {
    expect(await lineTotal({ id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } })).toBe(220);
  });
  it("should await and use the deferred value", async () => {
    const load = vi.fn().mockResolvedValue(15);
    expect(await lineTotal({ id: "c", base: 50, adjustment: { kind: "deferred", load } })).toBe(65);
    expect(load).toHaveBeenCalledTimes(1);
  });
  it("should sum the line totals when computing an order total", async () => {
    const items = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } },
    ] as const;
    expect(await computeTotal(items)).toBe(340);
  });
  it("should sum line totals including deferred adjustments", async () => {
    const load = vi.fn().mockResolvedValue(30);
    const items = [
      { id: "x", base: 100, adjustment: { kind: "fixed", value: 10 } },
      { id: "y", base: 50, adjustment: { kind: "deferred", load } },
    ] as const;
    expect(await computeTotal(items)).toBe(160);
    expect(load).toHaveBeenCalledTimes(1);
  });
});

describe("report", () => {
  it("should label each adjustment including deferred", () => {
    expect(amountLabel({ kind: "fixed", value: 5 })).toBe("flat");
    expect(amountLabel({ kind: "percent", pct: 5 })).toBe("rate");
    expect(amountLabel({ kind: "deferred", load: async () => 0 })).toBe("deferred");
  });
  it("should render one line per item with resolved deferred total", async () => {
    const load = vi.fn().mockResolvedValue(25);
    const lines = await buildReport([
      { id: "a", base: 100, adjustment: { kind: "deferred", load } }
    ]);
    expect(lines).toEqual(["a [deferred]: 125"]);
  });
});

describe("createPriceCache", () => {
  it("should reuse the cached total when the entry is still fresh", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return 42;
    };
    expect(await cache.get("k", compute)).toBe(42);
    expect(await cache.get("k", compute)).toBe(42);
    expect(calls).toBe(1);
  });
  it("should recompute the total when the entry has expired", async () => {
    let t = 0;
    let calls = 0;
    const cache = createPriceCache({ now: (): number => t }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return calls;
    };
    expect(await cache.get("k", compute)).toBe(1);
    t = 1500;
    expect(await cache.get("k", compute)).toBe(2);
    expect(calls).toBe(2);
  });
  it("should share a single computation for concurrent gets on the same fresh key", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return 99;
    };
    const [r1, r2, r3] = await Promise.all([
      cache.get("shared", compute),
      cache.get("shared", compute),
      cache.get("shared", compute),
    ]);
    expect(r1).toBe(99);
    expect(r2).toBe(99);
    expect(r3).toBe(99);
    expect(calls).toBe(1);
  });
  it("should never cache a failed computation and retry on subsequent get", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      if (calls === 1) throw new Error("fail");
      return 77;
    };
    await expect(cache.get("retry", compute)).rejects.toThrow("fail");
    expect(calls).toBe(1);
    expect(await cache.get("retry", compute)).toBe(77);
    expect(calls).toBe(2);
  });
  it("should keep different keys independent", async () => {
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const computeA = async () => 1;
    const computeB = async () => 2;
    expect(await cache.get("A", computeA)).toBe(1);
    expect(await cache.get("B", computeB)).toBe(2);
  });
  it("should handle deferred load via cache correctly", async () => {
    const load = vi.fn().mockResolvedValue(40);
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async () => load();
    expect(await cache.get("deferred-key", compute)).toBe(40);
    expect(await cache.get("deferred-key", compute)).toBe(40);
    expect(load).toHaveBeenCalledTimes(1);
  });
});
