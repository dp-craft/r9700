// FILE: impl.ts
export type Result =
  | { readonly ok: true; readonly value: number }
  | { readonly ok: false; readonly error: string };

type Token =
  | { readonly type: 'num'; readonly value: number }
  | { readonly type: 'op'; readonly value: '+' | '-' | '*' | '/' }
  | { readonly type: 'lp' }
  | { readonly type: 'rp' }
  | { readonly type: 'eof' };

type Cursor = { pos: number };

function tokenize(input: string): readonly Token[] | Result {
  const tokens: Token[] = []; let i = 0;
  while (i < input.length) {
    const ch = input[i];
    if (ch === ' ') { i++; continue; }
    if (ch === '(') { tokens.push({ type: 'lp' }); i++; continue; }
    if (ch === ')') { tokens.push({ type: 'rp' }); i++; continue; }
    if (ch === '+' || ch === '-' || ch === '*' || ch === '/') { tokens.push({ type: 'op', value: ch }); i++; continue; }
    if (ch !== undefined && ch >= '0' && ch <= '9') {
      let num = '';
      while (i < input.length && input[i] >= '0' && input[i] <= '9') { num += input[i]; i++; }
      const val = Number(num);
      if (Number.isNaN(val)) return { ok: false, error: 'Bad number' };
      tokens.push({ type: 'num', value: val }); continue;
    }
    return { ok: false, error: `Unexpected: ${ch}` };
  }
  tokens.push({ type: 'eof' }); return tokens;
}

function parseExpr(tokens: readonly Token[], cursor: Cursor): Result {
  let res = parseTerm(tokens, cursor);
  if (res.ok === false) return res;
  while (cursor.pos < tokens.length && tokens[cursor.pos].type === 'op' && (tokens[cursor.pos].value === '+' || tokens[cursor.pos].value === '-')) {
    const op = tokens[cursor.pos].value; cursor.pos++;
    const right = parseTerm(tokens, cursor);
    if (right.ok === false) return right;
    if (op === '+') res.value += right.value; else res.value -= right.value;
  }
  return res;
}

function parseTerm(tokens: readonly Token[], cursor: Cursor): Result {
  let res = parseFactor(tokens, cursor);
  if (res.ok === false) return res;
  while (cursor.pos < tokens.length && tokens[cursor.pos].type === 'op' && (tokens[cursor.pos].value === '*' || tokens[cursor.pos].value === '/')) {
    const op = tokens[cursor.pos].value; cursor.pos++;
    const right = parseFactor(tokens, cursor);
    if (right.ok === false) return right;
    if (op === '*') res.value *= right.value;
    else if (right.value === 0) return { ok: false, error: 'Division by zero' };
    else res.value /= right.value;
  }
  return res;
}

function parseFactor(tokens: readonly Token[], cursor: Cursor): Result {
  if (cursor.pos >= tokens.length) return { ok: false, error: 'Missing operand' };
  const token = tokens[cursor.pos];
  if (token.type === 'num') {
    cursor.pos++;
    return { ok: true, value: token.value };
  }
  if (token.type === 'lp') {
    cursor.pos++;
    const res = parseExpr(tokens, cursor);
    if (res.ok === false) return res;
    if (cursor.pos >= tokens.length || tokens[cursor.pos].type !== 'rp') return { ok: false, error: 'Unbalanced parentheses' };
    cursor.pos++;
    return res;
  }
  return { ok: false, error: 'Unexpected token' };
}

export function evaluate(expr: string): Result {
  const tokens = tokenize(expr);
  if ('ok' in tokens && tokens.ok === false) return tokens;
  const cursor = { pos: 0 };
  const res = parseExpr(tokens, cursor);
  if (res.ok === false) return res;
  if (cursor.pos < tokens.length && tokens[cursor.pos].type !== 'eof') return { ok: false, error: 'Unexpected token' };
  return res;
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import { evaluate, type Result } from './impl';

describe('evaluate', () => {
  describe('basic arithmetic', () => {
    it('should return correct value for simple addition when operands are positive', () => {
      const result: Result = evaluate('2 + 3');
      expect(result).toEqual({ ok: true, value: 5 });
    });

    it('should return correct value for simple subtraction when operands are positive', () => {
      const result: Result = evaluate('10 - 4');
      expect(result).toEqual({ ok: true, value: 6 });
    });

    it('should return correct value for simple multiplication when operands are positive', () => {
      const result: Result = evaluate('3 * 4');
      expect(result).toEqual({ ok: true, value: 12 });
    });

    it('should return correct value for simple division when operands divide evenly', () => {
      const result: Result = evaluate('8 / 2');
      expect(result).toEqual({ ok: true, value: 4 });
    });
  });

  describe('precedence', () => {
    it('should multiply before add when mixed operators are present', () => {
      const result: Result = evaluate('2 + 3 * 4');
      expect(result).toEqual({ ok: true, value: 14 });
    });

    it('should divide before subtract when mixed operators are present', () => {
      const result: Result = evaluate('10 - 8 / 2');
      expect(result).toEqual({ ok: true, value: 6 });
    });
  });

  describe('left-associativity', () => {
    it('should evaluate 10-3-2 as 5 when subtraction is chained', () => {
      const result: Result = evaluate('10 - 3 - 2');
      expect(result).toEqual({ ok: true, value: 5 });
    });

    it('should evaluate 8/2/2 as 2 when division is chained', () => {
      const result: Result = evaluate('8 / 2 / 2');
      expect(result).toEqual({ ok: true, value: 2 });
    });
  });

  describe('parentheses', () => {
    it('should override precedence when parentheses are used', () => {
      const result: Result = evaluate('(2 + 3) * 4');
      expect(result).toEqual({ ok: true, value: 20 });
    });

    it('should handle nested parentheses correctly', () => {
      const result: Result = evaluate('((2 + 3) * (4 - 1))');
      expect(result).toEqual({ ok: true, value: 15 });
    });
  });

  describe('whitespace handling', () => {
    it('should ignore arbitrary whitespace when spaces surround operators', () => {
      const result: Result = evaluate('  2   +   3  ');
      expect(result).toEqual({ ok: true, value: 5 });
    });
  });

  describe('error handling', () => {
    it('should return error for division by zero when divisor is zero', () => {
      const result: Result = evaluate('10 / 0');
      expect(result).toEqual({ ok: false, error: 'Division by zero' });
    });

    it('should return error for unbalanced parentheses when closing is missing', () => {
      const result: Result = evaluate('(2 + 3');
      expect(result).toEqual({ ok: false, error: 'Unbalanced parentheses' });
    });

    it('should return error for unexpected characters when non-numeric tokens appear', () => {
      const result: Result = evaluate('2 + a');
      expect(result).toEqual({ ok: false, error: 'Unexpected: a' });
    });

    it('should return error for missing operand when operator lacks right side', () => {
      const result: Result = evaluate('2 +');
      expect(result).toEqual({ ok: false, error: 'Missing operand' });
    });
  });
});
