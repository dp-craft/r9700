// FILE: impl.ts
import type { Clock } from '../../lib/clock';

interface CacheEntry<V> {
  readonly value: V;
  readonly expiresAt: number;
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> {
  const store = new Map<K, CacheEntry<V>>();

  const isExpired = (entry: CacheEntry<V>): boolean => clock.now() >= entry.expiresAt;

  const evictExpired = (): void => {
    const expiredKeys: K[] = [];
    for (const [key, entry] of store) {
      if (isExpired(entry)) {
        expiredKeys.push(key);
      }
    }
    for (const key of expiredKeys) {
      store.delete(key);
    }
  };

  const evictLru = (): void => {
    if (store.size > capacity) {
      const oldestKey = store.keys().next().value;
      if (oldestKey !== undefined) {
        store.delete(oldestKey);
      }
    }
  };

  const touchRecency = (key: K): void => {
    const entry = store.get(key);
    if (entry !== undefined) {
      store.delete(key);
      store.set(key, entry);
    }
  };

  return {
    get(key: K): V | undefined {
      const entry = store.get(key);
      if (entry === undefined || isExpired(entry)) {
        store.delete(key);
        return undefined;
      }
      touchRecency(key);
      return entry.value;
    },
    set(key: K, value: V): void {
      if (store.has(key)) {
        store.delete(key);
      }
      store.set(key, { value, expiresAt: clock.now() + ttlMs });
      evictLru();
    },
    has(key: K): boolean {
      const entry = store.get(key);
      if (entry === undefined || isExpired(entry)) {
        store.delete(key);
        return false;
      }
      return true;
    },
    get size(): number {
      evictExpired();
      return store.size;
    },
  };
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import type { Clock } from '../../lib/clock';
import { createLruCache } from './impl';

const createFakeClock = (initial: number): Clock => {
  let current = initial;
  return {
    now: (): number => current,
    tick: (ms: number): void => {
      current += ms;
    },
  };
};

describe('createLruCache', () => {
  describe('get', () => {
    it('should return the value when the entry is present and not expired', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      expect(cache.get('a')).toBe(1);
    });

    it('should return undefined when the entry is absent', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      expect(cache.get('b')).toBeUndefined();
    });

    it('should return undefined and remove the entry when it is expired', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.tick(100);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.has('a')).toBe(false);
    });

    it('should refresh recency when get is called on a live entry', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.tick(50);
      cache.get('a');
      clock.tick(60);
      expect(cache.get('a')).toBe(1);
    });
  });

  describe('set', () => {
    it('should store the value when capacity is available', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      expect(cache.get('a')).toBe(1);
    });

    it('should overwrite an existing key without triggering eviction', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      cache.set('a', 2);
      expect(cache.get('a')).toBe(2);
      expect(cache.size).toBe(1);
    });

    it('should evict the least-recently-used entry when capacity is exceeded', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      cache.set('c', 3);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
      expect(cache.get('c')).toBe(3);
    });

    it('should evict the correct entry when recency is refreshed', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      cache.get('a');
      cache.set('c', 3);
      expect(cache.get('a')).toBe(1);
      expect(cache.get('b')).toBeUndefined();
      expect(cache.get('c')).toBe(3);
    });
  });

  describe('has', () => {
    it('should return true when the key is present and not expired', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      expect(cache.has('a')).toBe(true);
    });

    it('should return false when the key is expired', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.tick(100);
      expect(cache.has('a')).toBe(false);
    });

    it('should return false when the key is absent', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      expect(cache.has('b')).toBe(false);
    });
  });

  describe('size', () => {
    it('should return the count of live entries', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(3, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      expect(cache.size).toBe(2);
    });

    it('should exclude expired entries from the count', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(3, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      clock.tick(100);
      expect(cache.size).toBe(0);
    });

    it('should return zero when the cache is empty', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(2, 100, clock);
      expect(cache.size).toBe(0);
    });
  });

  describe('capacity edge', () => {
    it('should evict correctly when capacity is 1', () => {
      const clock = createFakeClock(0);
      const cache = createLruCache<string, number>(1, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      expect(cache.size).toBe(1);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
    });
  });
});
