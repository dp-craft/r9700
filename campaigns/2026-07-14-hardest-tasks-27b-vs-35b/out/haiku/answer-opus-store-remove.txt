// FILE: store.ts
export interface Entry {
  readonly key: string;
  readonly amount: number;
}

export interface Store {
  add(entry: Entry): void;
  remove(key: string): void;
  total(key: string): number;
  all(): readonly Entry[];
}

class StoreImpl implements Store {
  private readonly entries: Entry[] = [];
  private readonly totals = new Map<string, number>();

  add(entry: Entry): void {
    this.entries.push(entry);
    this.totals.set(entry.key, (this.totals.get(entry.key) ?? 0) + entry.amount);
  }

  remove(key: string): void {
    for (let i = this.entries.length - 1; i >= 0; i--) {
      if (this.entries[i]?.key === key) {
        this.entries.splice(i, 1);
      }
    }
    this.totals.delete(key);
  }

  total(key: string): number {
    return this.totals.get(key) ?? 0;
  }

  all(): readonly Entry[] {
    return this.entries;
  }
}

export function createStore(): Store {
  return new StoreImpl();
}

// FILE: store.test.ts
import { describe, it, expect } from "vitest";
import { createStore } from "./store";

describe("createStore", () => {
  it("should accumulate the cached total when entries share a key", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "a", amount: 3 });
    expect(s.total("a")).toBe(5);
  });

  it("should reset the cached total to zero when a key is removed", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "a", amount: 3 });
    s.remove("a");
    expect(s.total("a")).toBe(0);
  });

  it("should drop all matching entries from all() when a key is removed", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "a", amount: 3 });
    s.remove("a");
    expect(s.all().some((e) => e.key === "a")).toBe(false);
  });

  it("should leave other keys' totals unchanged when a key is removed", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "b", amount: 7 });
    s.remove("a");
    expect(s.total("b")).toBe(7);
    expect(s.all()).toEqual([{ key: "b", amount: 7 }]);
  });

  it("should be a no-op when removing a key that has no entries", () => {
    const s = createStore();
    s.add({ key: "a", amount: 4 });
    s.remove("missing");
    expect(s.total("a")).toBe(4);
    expect(s.all()).toEqual([{ key: "a", amount: 4 }]);
  });
});
