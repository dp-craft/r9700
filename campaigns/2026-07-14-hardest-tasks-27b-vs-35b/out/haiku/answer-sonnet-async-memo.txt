// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  return (key: K): Promise<V> => {
    const cached = cache.get(key);
    if (cached !== undefined) return cached;
    const p = fn(key).catch((err: unknown): never => {
      cache.delete(key);
      throw err;
    });
    cache.set(key, p);
    return p;
  };
}
// FILE: impl.test.ts
import { describe, it, expect, vi } from "vitest";
import { createAsyncMemo } from "./impl";

describe("createAsyncMemo", () => {
  it("should call fn once when invoked concurrently for the same key", async () => {
    const fn = vi.fn(async (key: string): Promise<string> => {
      await new Promise<void>((resolve) => setTimeout(resolve, 10));
      return `value-${key}`;
    });
    const memo = createAsyncMemo(fn);

    const [a, b] = await Promise.all([memo("k"), memo("k")]);

    expect(a).toBe("value-k");
    expect(b).toBe("value-k");
    expect(fn).toHaveBeenCalledTimes(1);
  });

  it("should not cache a rejection while concurrent callers share the same failing call", async () => {
    const fn = vi.fn(async (): Promise<string> => {
      throw new Error("boom");
    });
    const memo = createAsyncMemo(fn);

    const results = await Promise.allSettled([memo("k"), memo("k")]);

    expect(results[0].status).toBe("rejected");
    expect(results[1].status).toBe("rejected");
    expect(fn).toHaveBeenCalledTimes(1);
  });

  it("should retry fn when the previous call for the same key rejected", async () => {
    let attempt = 0;
    const fn = vi.fn(async (key: string): Promise<string> => {
      attempt += 1;
      if (attempt === 1) throw new Error("boom");
      return `value-${key}`;
    });
    const memo = createAsyncMemo(fn);

    await expect(memo("k")).rejects.toThrow("boom");
    const result = await memo("k");

    expect(result).toBe("value-k");
    expect(fn).toHaveBeenCalledTimes(2);
  });

  it("should reuse the cached value after a successful call instead of calling fn again", async () => {
    const fn = vi.fn(async (key: string): Promise<string> => `value-${key}`);
    const memo = createAsyncMemo(fn);

    const first = await memo("k");
    const second = await memo("k");

    expect(first).toBe("value-k");
    expect(second).toBe("value-k");
    expect(fn).toHaveBeenCalledTimes(1);
  });
});
