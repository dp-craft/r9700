// FILE: impl.ts
import { Clock } from "../../lib/clock";

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

interface Entry<V> {
  value: V;
  insertedAtMs: number;
  lastAccessMs: number;
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();

  function isExpired(entry: Entry<V>): boolean {
    return clock.now() >= entry.insertedAtMs + ttlMs;
  }

  function evictLru(): void {
    let lruKey: K | undefined;
    let lruTime = Infinity;

    for (const [k, entry] of store.entries()) {
      if (!isExpired(entry) && entry.lastAccessMs < lruTime) {
        lruKey = k;
        lruTime = entry.lastAccessMs;
      }
    }

    if (lruKey !== undefined) {
      store.delete(lruKey);
    }
  }

  function cleanupExpired(): void {
    for (const [k, entry] of store.entries()) {
      if (isExpired(entry)) {
        store.delete(k);
      }
    }
  }

  return {
    get(key: K): V | undefined {
      const entry = store.get(key);
      if (!entry) return undefined;
      if (isExpired(entry)) {
        store.delete(key);
        return undefined;
      }
      entry.lastAccessMs = clock.now();
      return entry.value;
    },

    set(key: K, value: V): void {
      const now = clock.now();
      const existing = store.get(key);

      if (existing && !isExpired(existing)) {
        existing.value = value;
        existing.lastAccessMs = now;
        return;
      }

      if (existing) {
        store.delete(key);
      }

      cleanupExpired();

      if (store.size >= capacity) {
        evictLru();
      }

      store.set(key, { value, insertedAtMs: now, lastAccessMs: now });
    },

    has(key: K): boolean {
      const entry = store.get(key);
      if (!entry) return false;
      if (isExpired(entry)) {
        store.delete(key);
        return false;
      }
      return true;
    },

    get size(): number {
      let count = 0;
      for (const entry of store.values()) {
        if (!isExpired(entry)) {
          count++;
        }
      }
      return count;
    },
  };
}
// FILE: impl.test.ts
import { describe, it, expect } from "vitest";
import { createLruCache } from "./impl";
import type { Clock } from "../../lib/clock";

class FakeClock implements Clock {
  private time = 0;

  now(): number {
    return this.time;
  }

  advance(ms: number): void {
    this.time += ms;
  }
}

describe("LruCache", () => {
  it("should return undefined when key is absent", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    expect(cache.get("missing")).toBeUndefined();
    expect(cache.has("missing")).toBe(false);
  });

  it("should store and retrieve values when within capacity", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set("a", 42);
    expect(cache.get("a")).toBe(42);
    expect(cache.has("a")).toBe(true);
    expect(cache.size).toBe(1);
  });

  it("should return undefined and not count expired entries when TTL exceeded", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(2, 100, clock);

    cache.set("key", "value");
    clock.advance(100);

    expect(cache.get("key")).toBeUndefined();
    expect(cache.has("key")).toBe(false);
    expect(cache.size).toBe(0);
  });

  it("should not count multiple expired entries toward size", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(5, 100, clock);

    cache.set("a", "1");
    cache.set("b", "2");
    cache.set("c", "3");
    expect(cache.size).toBe(3);

    clock.advance(100);
    expect(cache.size).toBe(0);
  });

  it("should refresh recency on get and preserve recently-accessed entries during eviction", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    cache.set("a", "1");
    clock.advance(10);
    cache.set("b", "2");
    clock.advance(10);
    cache.get("a");
    clock.advance(10);
    cache.set("c", "3");

    expect(cache.has("a")).toBe(true);
    expect(cache.has("b")).toBe(false);
    expect(cache.has("c")).toBe(true);
  });

  it("should evict least-recently-used entry when capacity exceeded", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, number>(3, 1000, clock);

    cache.set("a", 1);
    clock.advance(10);
    cache.set("b", 2);
    clock.advance(10);
    cache.set("c", 3);
    clock.advance(10);
    cache.set("d", 4);

    expect(cache.has("a")).toBe(false);
    expect(cache.has("b")).toBe(true);
    expect(cache.has("c")).toBe(true);
    expect(cache.has("d")).toBe(true);
  });

  it("should skip expired entries during eviction and only evict live LRU", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(2, 50, clock);

    cache.set("a", "1");
    clock.advance(10);
    cache.set("b", "2");
    clock.advance(50);
    clock.advance(10);
    cache.set("c", "3");

    expect(cache.has("a")).toBe(false);
    expect(cache.has("b")).toBe(true);
    expect(cache.has("c")).toBe(true);
    expect(cache.size).toBe(2);
  });

  it("should update value and refresh recency when setting existing live key", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(2, 1000, clock);

    cache.set("a", "old");
    clock.advance(10);
    cache.set("a", "new");

    expect(cache.get("a")).toBe("new");
    expect(cache.size).toBe(1);
  });

  it("should evict oldest entry when at capacity edge of 1", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(1, 1000, clock);

    cache.set("a", "1");
    expect(cache.size).toBe(1);

    clock.advance(10);
    cache.set("b", "2");

    expect(cache.size).toBe(1);
    expect(cache.has("a")).toBe(false);
    expect(cache.has("b")).toBe(true);
  });

  it("should not resurrect expired entries and remove them from store via has()", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(2, 50, clock);

    cache.set("a", "value");
    clock.advance(50);

    expect(cache.has("a")).toBe(false);
    expect(cache.get("a")).toBeUndefined();
    expect(cache.size).toBe(0);
  });

  it("should handle staggered expiry of multiple entries correctly", () => {
    const clock = new FakeClock();
    const cache = createLruCache<string, string>(5, 100, clock);

    cache.set("a", "1");
    clock.advance(50);
    cache.set("b", "2");
    clock.advance(50);
    cache.set("c", "3");

    expect(cache.size).toBe(3);
    clock.advance(50);
    expect(cache.size).toBe(2);
    expect(cache.has("a")).toBe(false);
    expect(cache.has("b")).toBe(true);
    expect(cache.has("c")).toBe(true);
  });
});
