// FILE: impl.ts
import type { Clock } from '../../lib/clock';

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

interface CacheEntry<V> {
  value: V;
  insertedAt: number;
  accessedAt: number;
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> {
  const cache = new Map<K, CacheEntry<V>>();

  function isExpired(entry: CacheEntry<V>): boolean {
    return clock.now() >= entry.insertedAt + ttlMs;
  }

  function cleanExpired(): void {
    for (const [key, entry] of cache.entries()) {
      if (isExpired(entry)) {
        cache.delete(key);
      }
    }
  }

  function getLiveSize(): number {
    let count = 0;
    for (const entry of cache.values()) {
      if (!isExpired(entry)) {
        count++;
      }
    }
    return count;
  }

  function evictLruLive(): void {
    let oldestKey: K | undefined;
    let oldestAccessTime = Infinity;

    for (const [key, entry] of cache.entries()) {
      if (!isExpired(entry) && entry.accessedAt < oldestAccessTime) {
        oldestAccessTime = entry.accessedAt;
        oldestKey = key;
      }
    }

    if (oldestKey !== undefined) {
      cache.delete(oldestKey);
    }
  }

  return {
    get(key: K): V | undefined {
      const entry = cache.get(key);
      if (!entry) return undefined;

      if (isExpired(entry)) {
        cache.delete(key);
        return undefined;
      }

      entry.accessedAt = clock.now();
      return entry.value;
    },

    set(key: K, value: V): void {
      const now = clock.now();
      const existing = cache.get(key);

      if (existing && !isExpired(existing)) {
        existing.value = value;
        existing.accessedAt = now;
        return;
      }

      cache.delete(key);
      cleanExpired();

      const liveCount = getLiveSize();
      if (liveCount >= capacity) {
        evictLruLive();
      }

      cache.set(key, {
        value,
        insertedAt: now,
        accessedAt: now,
      });
    },

    has(key: K): boolean {
      const entry = cache.get(key);
      if (!entry) return false;

      if (isExpired(entry)) {
        cache.delete(key);
        return false;
      }

      return true;
    },

    get size(): number {
      return getLiveSize();
    },
  };
}

// FILE: impl.test.ts
import { describe, it, expect } from 'vitest';
import type { Clock } from '../../lib/clock';
import { createLruCache } from './impl';

describe('createLruCache', () => {
  let time = 0;
  const fakeClock: Clock = {
    now: () => time,
  };

  describe('should store and retrieve values when not expired', () => {
    it('should get a value before expiry', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 1000, fakeClock);
      cache.set('a', 1);
      expect(cache.get('a')).toBe(1);
    });

    it('should handle multiple entries', () => {
      time = 0;
      const cache = createLruCache<string, string>(3, 1000, fakeClock);
      cache.set('x', 'hello');
      cache.set('y', 'world');
      expect(cache.get('x')).toBe('hello');
      expect(cache.get('y')).toBe('world');
    });
  });

  describe('should expire entries when TTL exceeded', () => {
    it('should return undefined for expired entries', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 100, fakeClock);
      cache.set('a', 42);
      time = 100;
      expect(cache.get('a')).toBeUndefined();
    });

    it('should not count expired entries in size', () => {
      time = 0;
      const cache = createLruCache<string, number>(5, 100, fakeClock);
      cache.set('a', 1);
      cache.set('b', 2);
      expect(cache.size).toBe(2);
      time = 100;
      expect(cache.size).toBe(0);
    });

    it('should return false for has() when entry expired', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 50, fakeClock);
      cache.set('key', 123);
      time = 50;
      expect(cache.has('key')).toBe(false);
    });
  });

  describe('should evict LRU live entry when capacity exceeded', () => {
    it('should evict least recently used entry when inserting beyond capacity', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 1000, fakeClock);
      cache.set('a', 1);
      time = 10;
      cache.set('b', 2);
      time = 20;
      cache.set('c', 3);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
      expect(cache.get('c')).toBe(3);
    });

    it('should skip expired entries during eviction', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 100, fakeClock);
      cache.set('a', 1);
      time = 50;
      cache.set('b', 2);
      time = 101;
      cache.set('c', 3);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
      expect(cache.get('c')).toBe(3);
      expect(cache.size).toBe(2);
    });
  });

  describe('should refresh recency on get()', () => {
    it('should update access time to prevent eviction', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 1000, fakeClock);
      cache.set('a', 1);
      time = 10;
      cache.set('b', 2);
      time = 20;
      cache.get('a');
      time = 30;
      cache.set('c', 3);
      expect(cache.get('a')).toBe(1);
      expect(cache.get('b')).toBeUndefined();
      expect(cache.get('c')).toBe(3);
    });

    it('should not refresh recency in has()', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 1000, fakeClock);
      cache.set('a', 1);
      time = 10;
      cache.set('b', 2);
      time = 20;
      cache.has('a');
      time = 30;
      cache.set('c', 3);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
      expect(cache.get('c')).toBe(3);
    });
  });

  describe('should handle capacity edge cases', () => {
    it('should handle capacity of 1', () => {
      time = 0;
      const cache = createLruCache<string, number>(1, 1000, fakeClock);
      cache.set('a', 1);
      expect(cache.size).toBe(1);
      time = 10;
      cache.set('b', 2);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
      expect(cache.size).toBe(1);
    });

    it('should not evict when under capacity', () => {
      time = 0;
      const cache = createLruCache<string, number>(3, 1000, fakeClock);
      cache.set('a', 1);
      cache.set('b', 2);
      expect(cache.size).toBe(2);
      expect(cache.get('a')).toBe(1);
      expect(cache.get('b')).toBe(2);
    });
  });

  describe('should handle updating existing keys', () => {
    it('should update value and refresh recency when key exists', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 1000, fakeClock);
      cache.set('a', 1);
      time = 10;
      cache.set('b', 2);
      time = 20;
      cache.set('a', 10);
      time = 30;
      cache.set('c', 3);
      expect(cache.get('a')).toBe(10);
      expect(cache.get('b')).toBeUndefined();
      expect(cache.get('c')).toBe(3);
    });
  });

  describe('should handle absent keys', () => {
    it('should return undefined for missing keys in get()', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 1000, fakeClock);
      expect(cache.get('nonexistent')).toBeUndefined();
    });

    it('should return false for missing keys in has()', () => {
      time = 0;
      const cache = createLruCache<string, number>(2, 1000, fakeClock);
      expect(cache.has('nonexistent')).toBe(false);
    });
  });
});
