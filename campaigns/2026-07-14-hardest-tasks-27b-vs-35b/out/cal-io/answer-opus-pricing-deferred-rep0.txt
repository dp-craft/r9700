// FILE: types.ts
export type Amount =
  | { readonly kind: "fixed"; readonly value: number }
  | { readonly kind: "percent"; readonly pct: number }
  | { readonly kind: "deferred"; readonly load: () => Promise<number> };

export interface LineItem {
  readonly id: string;
  readonly base: number;
  readonly adjustment: Amount;
}

// FILE: pricing.ts
import type { Amount, LineItem } from "./types";

export const PERCENT_BASE = 100;

const NO_PERCENT = 0;
const EMPTY_TOTAL = 0;

/** Percentage carried by an amount; non-percent amounts contribute no rate. */
function pctOf(amount: Amount): number {
  return amount.kind === "percent" ? amount.pct : NO_PERCENT;
}

/** Resolved adjustment for `base`; a deferred amount is loaded and then treated exactly like a fixed one. */
function adjustmentOf(base: number, amount: Amount): Promise<number> {
  switch (amount.kind) {
    case "fixed":
      return Promise.resolve(amount.value);
    case "percent":
      return Promise.resolve((base * pctOf(amount)) / PERCENT_BASE);
    case "deferred":
      return amount.load();
  }
}

export async function lineTotal(item: LineItem): Promise<number> {
  const adjustment = await adjustmentOf(item.base, item.adjustment);
  return item.base + adjustment;
}

export async function computeTotal(items: readonly LineItem[]): Promise<number> {
  const totals = await Promise.all(items.map((item: LineItem): Promise<number> => lineTotal(item)));
  return totals.reduce((sum: number, total: number): number => sum + total, EMPTY_TOTAL);
}

// FILE: report.ts
import type { Amount, LineItem } from "./types";
import { lineTotal } from "./pricing";

const KIND_LABEL: Partial<Record<Amount["kind"], string>> = {
  fixed: "flat",
  percent: "rate",
  deferred: "deferred",
};
const UNKNOWN_LABEL = "other";

export function amountLabel(amount: Amount): string {
  return KIND_LABEL[amount.kind] ?? UNKNOWN_LABEL;
}

export async function reportLine(item: LineItem): Promise<string> {
  const total = await lineTotal(item);
  return `${item.id} [${amountLabel(item.adjustment)}]: ${String(total)}`;
}

export function buildReport(items: readonly LineItem[]): Promise<readonly string[]> {
  return Promise.all(items.map((item: LineItem): Promise<string> => reportLine(item)));
}

// FILE: cache.ts
import type { Clock } from "../../lib/clock";

export interface PriceCache {
  /** Return the cached total for `key` if fresh, else run `compute` and cache its result. */
  get(key: string, compute: () => Promise<number>): Promise<number>;
}

/** A cache slot holds the in-flight promise, not the settled value, so concurrent
 *  callers for the same fresh key share ONE computation. */
interface CacheEntry {
  readonly value: Promise<number>;
  readonly expiresAt: number;
}

const DEFAULT_TTL_MS = 60_000;

/** Run `compute`, turning a synchronous throw into a rejection so failures are uniform. */
function runCompute(compute: () => Promise<number>): Promise<number> {
  try {
    return compute();
  } catch (error: unknown) {
    return Promise.reject(error instanceof Error ? error : new Error(String(error)));
  }
}

/** Start a computation, publish its promise under `key`, and evict the slot if it fails. */
function start(
  entries: Map<string, CacheEntry>,
  key: string,
  compute: () => Promise<number>,
  expiresAt: number,
): Promise<number> {
  const value = runCompute(compute);
  const entry: CacheEntry = { value, expiresAt };
  entries.set(key, entry);
  // A failed computation must never stay cached: drop the slot so a later `get` retries.
  value.catch((): void => {
    if (entries.get(key) === entry) {
      entries.delete(key);
    }
  });
  return value;
}

export function createPriceCache(clock: Clock, ttlMs: number = DEFAULT_TTL_MS): PriceCache {
  const entries = new Map<string, CacheEntry>();
  return {
    get(key: string, compute: () => Promise<number>): Promise<number> {
      const now = clock.now();
      const hit = entries.get(key);
      if (hit !== undefined && hit.expiresAt > now) {
        return hit.value;
      }
      return start(entries, key, compute, now + ttlMs);
    },
  };
}

// FILE: pricing.test.ts
import { describe, it, expect } from "vitest";
import { computeTotal, lineTotal } from "./pricing";
import { amountLabel, buildReport, reportLine } from "./report";
import { createPriceCache } from "./cache";

describe("pricing", () => {
  it("should add the fixed value when the adjustment is fixed", async () => {
    expect(await lineTotal({ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } })).toBe(120);
  });
  it("should apply the rate when the adjustment is a percent", async () => {
    expect(await lineTotal({ id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } })).toBe(220);
  });
  it("should add the resolved value when the adjustment is deferred", async () => {
    const load = async (): Promise<number> => 20;
    expect(await lineTotal({ id: "c", base: 100, adjustment: { kind: "deferred", load } })).toBe(120);
  });
  it("should match the fixed result when a deferred adjustment resolves to the same value", async () => {
    const deferred = await lineTotal({
      id: "c",
      base: 100,
      adjustment: { kind: "deferred", load: async (): Promise<number> => 20 },
    });
    expect(deferred).toBe(await lineTotal({ id: "c", base: 100, adjustment: { kind: "fixed", value: 20 } }));
  });
  it("should reject when a deferred load fails", async () => {
    const load = async (): Promise<number> => {
      throw new Error("upstream down");
    };
    await expect(lineTotal({ id: "c", base: 100, adjustment: { kind: "deferred", load } })).rejects.toThrow(
      "upstream down",
    );
  });
  it("should sum the line totals when computing an order total", async () => {
    const items = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } },
    ] as const;
    expect(await computeTotal(items)).toBe(340);
  });
  it("should include the resolved deferred totals when computing an order total", async () => {
    const items = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } },
      { id: "c", base: 50, adjustment: { kind: "deferred", load: async (): Promise<number> => 5 } },
    ] as const;
    expect(await computeTotal(items)).toBe(395);
  });
  it("should return zero when computing the total of no items", async () => {
    expect(await computeTotal([])).toBe(0);
  });
});

describe("report", () => {
  it("should label each adjustment when building the report", () => {
    expect(amountLabel({ kind: "fixed", value: 5 })).toBe("flat");
    expect(amountLabel({ kind: "percent", pct: 5 })).toBe("rate");
    expect(amountLabel({ kind: "deferred", load: async (): Promise<number> => 5 })).toBe("deferred");
  });
  it("should render one line per item when building the report", async () => {
    const lines = await buildReport([{ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } }]);
    expect(lines).toEqual(["a [flat]: 120"]);
  });
  it("should render the resolved total when the adjustment is deferred", async () => {
    const load = async (): Promise<number> => 20;
    expect(await reportLine({ id: "c", base: 100, adjustment: { kind: "deferred", load } })).toBe(
      "c [deferred]: 120",
    );
  });
  it("should never render a promise when the report contains a deferred item", async () => {
    const items = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "c", base: 50, adjustment: { kind: "deferred", load: async (): Promise<number> => 5 } },
    ] as const;
    const lines = await buildReport(items);
    expect(lines).toEqual(["a [flat]: 120", "c [deferred]: 55"]);
    expect(lines.join("\n")).not.toContain("[object Promise]");
  });
});

describe("createPriceCache", () => {
  it("should reuse the cached total when the entry is still fresh", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return 42;
    };
    expect(await cache.get("k", compute)).toBe(42);
    expect(await cache.get("k", compute)).toBe(42);
    expect(calls).toBe(1);
  });

  it("should recompute the total when the entry has expired", async () => {
    let t = 0;
    let calls = 0;
    const cache = createPriceCache({ now: (): number => t }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return calls;
    };
    expect(await cache.get("k", compute)).toBe(1);
    t = 1500;
    expect(await cache.get("k", compute)).toBe(2);
    expect(calls).toBe(2);
  });

  it("should share one computation when concurrent gets hit the same fresh key", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      await Promise.resolve();
      return 7;
    };
    const results = await Promise.all([cache.get("k", compute), cache.get("k", compute), cache.get("k", compute)]);
    expect(results).toEqual([7, 7, 7]);
    expect(calls).toBe(1);
  });

  it("should invoke a deferred load exactly once when concurrent gets share a key", async () => {
    let loads = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const item = {
      id: "c",
      base: 100,
      adjustment: {
        kind: "deferred",
        load: async (): Promise<number> => {
          loads += 1;
          return 20;
        },
      },
    } as const;
    const compute = (): Promise<number> => lineTotal(item);
    expect(await Promise.all([cache.get(item.id, compute), cache.get(item.id, compute)])).toEqual([120, 120]);
    expect(loads).toBe(1);
  });

  it("should retry the computation when the previous one failed", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      if (calls === 1) {
        throw new Error("boom");
      }
      return 9;
    };
    await expect(cache.get("k", compute)).rejects.toThrow("boom");
    expect(await cache.get("k", compute)).toBe(9);
    expect(calls).toBe(2);
  });

  it("should reject every concurrent caller when the shared computation fails", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      await Promise.resolve();
      throw new Error("boom");
    };
    const first = cache.get("k", compute);
    const second = cache.get("k", compute);
    await expect(first).rejects.toThrow("boom");
    await expect(second).rejects.toThrow("boom");
    expect(calls).toBe(1);
  });

  it("should keep the entries independent when different keys are cached", async () => {
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    expect(await cache.get("a", async (): Promise<number> => 1)).toBe(1);
    expect(await cache.get("b", async (): Promise<number> => 2)).toBe(2);
    expect(await cache.get("a", async (): Promise<number> => 99)).toBe(1);
    expect(await cache.get("b", async (): Promise<number> => 99)).toBe(2);
  });

  it("should give the same total when the pipeline runs with and without the cache", async () => {
    const items = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "c", base: 50, adjustment: { kind: "deferred", load: async (): Promise<number> => 5 } },
    ] as const;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const cached = await cache.get("order", (): Promise<number> => computeTotal(items));
    expect(cached).toBe(await computeTotal(items));
  });
});
