// FILE: types.ts
export type Amount =
  | { readonly kind: "fixed"; readonly value: number }
  | { readonly kind: "percent"; readonly pct: number }
  | { readonly kind: "deferred"; readonly load: () => Promise<number> };

export interface LineItem {
  readonly id: string;
  readonly base: number;
  readonly adjustment: Amount;
}

// FILE: pricing.ts
import type { Amount, LineItem } from "./types";

export const PERCENT_BASE = 100;

/** Percentage carried by an amount; non-percent amounts contribute no rate. */
function pctOf(amount: Amount): number {
  return amount.kind === "percent" ? amount.pct : 0;
}

async function adjustmentOf(base: number, amount: Amount): Promise<number> {
  if (amount.kind === "fixed") {
    return amount.value;
  }
  if (amount.kind === "deferred") {
    return amount.load();
  }
  return (base * pctOf(amount)) / PERCENT_BASE;
}

export async function lineTotal(item: LineItem): Promise<number> {
  return item.base + (await adjustmentOf(item.base, item.adjustment));
}

export async function computeTotal(items: readonly LineItem[]): Promise<number> {
  const totals = await Promise.all(items.map((item: LineItem): Promise<number> => lineTotal(item)));
  return totals.reduce((sum: number, total: number): number => sum + total, 0);
}

// FILE: report.ts
import type { Amount, LineItem } from "./types";
import { lineTotal } from "./pricing";

const KIND_LABEL: Partial<Record<Amount["kind"], string>> = {
  fixed: "flat",
  percent: "rate",
  deferred: "deferred",
};
const UNKNOWN_LABEL = "other";

export function amountLabel(amount: Amount): string {
  return KIND_LABEL[amount.kind] ?? UNKNOWN_LABEL;
}

export async function reportLine(item: LineItem): Promise<string> {
  const total = await lineTotal(item);
  return `${item.id} [${amountLabel(item.adjustment)}]: ${String(total)}`;
}

export async function buildReport(items: readonly LineItem[]): Promise<readonly string[]> {
  return Promise.all(items.map((item: LineItem): Promise<string> => reportLine(item)));
}

// FILE: cache.ts
import type { Clock } from "../../lib/clock";

export interface PriceCache {
  /** Return the cached total for `key` if fresh, else run `compute` and cache its result. */
  get(key: string, compute: () => Promise<number>): Promise<number>;
}

interface CacheEntry {
  readonly value: number;
  readonly expiresAt: number;
}

const DEFAULT_TTL_MS = 60_000;

async function computeAndCache(
  key: string,
  compute: () => Promise<number>,
  entries: Map<string, CacheEntry>,
  pending: Map<string, Promise<number>>,
  clock: Clock,
  ttlMs: number,
): Promise<number> {
  try {
    const value = await compute();
    entries.set(key, { value, expiresAt: clock.now() + ttlMs });
    return value;
  } finally {
    pending.delete(key);
  }
}

export function createPriceCache(clock: Clock, ttlMs: number = DEFAULT_TTL_MS): PriceCache {
  const entries = new Map<string, CacheEntry>();
  const pending = new Map<string, Promise<number>>();
  return {
    async get(key: string, compute: () => Promise<number>): Promise<number> {
      const hit = entries.get(key);
      if (hit !== undefined && hit.expiresAt > clock.now()) {
        return hit.value;
      }
      const inFlight = pending.get(key);
      if (inFlight !== undefined) {
        return inFlight;
      }
      const promise = computeAndCache(key, compute, entries, pending, clock, ttlMs);
      pending.set(key, promise);
      return promise;
    },
  };
}

// FILE: pricing.test.ts
import { describe, it, expect } from "vitest";
import { computeTotal, lineTotal } from "./pricing";
import { amountLabel, buildReport, reportLine } from "./report";
import { createPriceCache } from "./cache";
import type { LineItem } from "./types";

describe("pricing", () => {
  it("should add the fixed value when the adjustment is fixed", async () => {
    expect(await lineTotal({ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } })).toBe(120);
  });
  it("should apply the rate when the adjustment is a percent", async () => {
    expect(await lineTotal({ id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } })).toBe(220);
  });
  it("should resolve the load when the adjustment is deferred", async () => {
    const item: LineItem = {
      id: "c",
      base: 50,
      adjustment: { kind: "deferred", load: async (): Promise<number> => 30 },
    };
    expect(await lineTotal(item)).toBe(80);
  });
  it("should sum the line totals when computing an order total", async () => {
    const items: readonly LineItem[] = [
      { id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } },
      { id: "b", base: 200, adjustment: { kind: "percent", pct: 10 } },
      { id: "c", base: 50, adjustment: { kind: "deferred", load: async (): Promise<number> => 30 } },
    ];
    expect(await computeTotal(items)).toBe(420);
  });
});

describe("report", () => {
  it("should label each adjustment when building the report", () => {
    expect(amountLabel({ kind: "fixed", value: 5 })).toBe("flat");
    expect(amountLabel({ kind: "percent", pct: 5 })).toBe("rate");
    expect(amountLabel({ kind: "deferred", load: async (): Promise<number> => 1 })).toBe("deferred");
  });
  it("should render one line per item when building the report", async () => {
    const lines = await buildReport([{ id: "a", base: 100, adjustment: { kind: "fixed", value: 20 } }]);
    expect(lines).toEqual(["a [flat]: 120"]);
  });
  it("should render the resolved total, not a stringified promise, when the adjustment is deferred", async () => {
    const item: LineItem = {
      id: "d",
      base: 10,
      adjustment: { kind: "deferred", load: async (): Promise<number> => 5 },
    };
    expect(await reportLine(item)).toBe("d [deferred]: 15");
  });
});

describe("createPriceCache", () => {
  it("should reuse the cached total when the entry is still fresh", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return 42;
    };
    expect(await cache.get("k", compute)).toBe(42);
    expect(await cache.get("k", compute)).toBe(42);
    expect(calls).toBe(1);
  });
  it("should recompute the total when the entry has expired", async () => {
    let t = 0;
    let calls = 0;
    const cache = createPriceCache({ now: (): number => t }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return calls;
    };
    expect(await cache.get("k", compute)).toBe(1);
    t = 1500;
    expect(await cache.get("k", compute)).toBe(2);
    expect(calls).toBe(2);
  });
  it("should share one computation when concurrent gets race on the same fresh key", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      return 7;
    };
    const [a, b] = await Promise.all([cache.get("k", compute), cache.get("k", compute)]);
    expect(a).toBe(7);
    expect(b).toBe(7);
    expect(calls).toBe(1);
  });
  it("should not cache a failed computation and should retry on the next get", async () => {
    let calls = 0;
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const compute = async (): Promise<number> => {
      calls += 1;
      if (calls === 1) {
        throw new Error("boom");
      }
      return 9;
    };
    await expect(cache.get("k", compute)).rejects.toThrow("boom");
    expect(await cache.get("k", compute)).toBe(9);
    expect(calls).toBe(2);
  });
  it("should keep different keys independent", async () => {
    const cache = createPriceCache({ now: (): number => 0 }, 1000);
    const a = await cache.get("a", async (): Promise<number> => 1);
    const b = await cache.get("b", async (): Promise<number> => 2);
    expect(a).toBe(1);
    expect(b).toBe(2);
  });
});
