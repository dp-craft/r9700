// FILE: impl.ts
import type { Clock } from "../../lib/clock";

const MILLIS_PER_SECOND = 1000;
const DEFAULT_ACQUIRE_TOKENS = 1;

export interface RateLimiter {
  tryAcquire(key: string, tokens?: number): boolean;
}

interface Bucket {
  tokens: number;
  lastRefillMs: number;
}

function createBucket(capacity: number, nowMs: number): Bucket {
  return { tokens: capacity, lastRefillMs: nowMs };
}

function refill(
  bucket: Bucket,
  nowMs: number,
  capacity: number,
  refillPerSec: number,
): void {
  const elapsedSeconds = (nowMs - bucket.lastRefillMs) / MILLIS_PER_SECOND;
  if (elapsedSeconds <= 0) {
    return;
  }
  const replenished = bucket.tokens + elapsedSeconds * refillPerSec;
  bucket.tokens = Math.min(capacity, replenished);
  bucket.lastRefillMs = nowMs;
}

export function createRateLimiter(
  capacity: number,
  refillPerSec: number,
  clock: Clock,
): RateLimiter {
  const buckets = new Map<string, Bucket>();

  const bucketFor = (key: string, nowMs: number): Bucket => {
    const existing = buckets.get(key);
    if (existing !== undefined) {
      return existing;
    }
    const created = createBucket(capacity, nowMs);
    buckets.set(key, created);
    return created;
  };

  return {
    tryAcquire(key: string, tokens: number = DEFAULT_ACQUIRE_TOKENS): boolean {
      const nowMs = clock.now();
      const bucket = bucketFor(key, nowMs);
      refill(bucket, nowMs, capacity, refillPerSec);
      if (bucket.tokens < tokens) {
        return false;
      }
      bucket.tokens -= tokens;
      return true;
    },
  };
}

// FILE: impl.test.ts
import { describe, expect, it } from "vitest";
import type { Clock } from "../../lib/clock";
import { createRateLimiter } from "./impl";

const CAPACITY = 3;
const REFILL_PER_SEC = 2;
const START_MS = 1_000;
const MILLIS_PER_SECOND = 1000;

interface FakeClock extends Clock {
  advanceSeconds(seconds: number): void;
}

function createFakeClock(startMs: number = START_MS): FakeClock {
  let currentMs = startMs;
  return {
    now: (): number => currentMs,
    advanceSeconds: (seconds: number): void => {
      currentMs += seconds * MILLIS_PER_SECOND;
    },
  };
}

describe("createRateLimiter", () => {
  it("should allow a full burst up to capacity when the bucket is untouched", () => {
    const clock = createFakeClock();
    const limiter = createRateLimiter(CAPACITY, REFILL_PER_SEC, clock);

    const results: readonly boolean[] = [
      limiter.tryAcquire("a"),
      limiter.tryAcquire("a"),
      limiter.tryAcquire("a"),
    ];

    expect(results).toEqual([true, true, true]);
  });

  it("should reject the request when the bucket is empty and no time has passed", () => {
    const clock = createFakeClock();
    const limiter = createRateLimiter(CAPACITY, REFILL_PER_SEC, clock);

    for (let i = 0; i < CAPACITY; i += 1) {
      limiter.tryAcquire("a");
    }

    expect(limiter.tryAcquire("a")).toBe(false);
  });

  it("should deduct nothing when the request is rejected", () => {
    const clock = createFakeClock();
    const limiter = createRateLimiter(CAPACITY, REFILL_PER_SEC, clock);

    expect(limiter.tryAcquire("a", CAPACITY)).toBe(true);
    expect(limiter.tryAcquire("a", 2)).toBe(false);

    clock.advanceSeconds(1);

    expect(limiter.tryAcquire("a", 2)).toBe(true);
  });

  it("should lazily refill tokens when time elapses between calls", () => {
    const clock = createFakeClock();
    const limiter = createRateLimiter(CAPACITY, REFILL_PER_SEC, clock);

    for (let i = 0; i < CAPACITY; i += 1) {
      limiter.tryAcquire("a");
    }
    expect(limiter.tryAcquire("a")).toBe(false);

    clock.advanceSeconds(0.5);

    expect(limiter.tryAcquire("a")).toBe(true);
    expect(limiter.tryAcquire("a")).toBe(false);
  });

  it("should cap the refill at capacity when a long idle period elapses", () => {
    const clock = createFakeClock();
    const limiter = createRateLimiter(CAPACITY, REFILL_PER_SEC, clock);

    limiter.tryAcquire("a", CAPACITY);
    clock.advanceSeconds(60);

    for (let i = 0; i < CAPACITY; i += 1) {
      expect(limiter.tryAcquire("a")).toBe(true);
    }
    expect(limiter.tryAcquire("a")).toBe(false);
  });

  it("should accept a multi-token request when the bucket holds enough tokens", () => {
    const clock = createFakeClock();
    const limiter = createRateLimiter(CAPACITY, REFILL_PER_SEC, clock);

    expect(limiter.tryAcquire("a", 2)).toBe(true);
    expect(limiter.tryAcquire("a", 2)).toBe(false);
    expect(limiter.tryAcquire("a", 1)).toBe(true);
  });

  it("should keep buckets independent when different keys are used", () => {
    const clock = createFakeClock();
    const limiter = createRateLimiter(CAPACITY, REFILL_PER_SEC, clock);

    for (let i = 0; i < CAPACITY; i += 1) {
      limiter.tryAcquire("a");
    }

    expect(limiter.tryAcquire("a")).toBe(false);
    expect(limiter.tryAcquire("b")).toBe(true);
  });

  it("should start a newly seen key full when it is first used after a delay", () => {
    const clock = createFakeClock();
    const limiter = createRateLimiter(CAPACITY, REFILL_PER_SEC, clock);

    clock.advanceSeconds(120);

    for (let i = 0; i < CAPACITY; i += 1) {
      expect(limiter.tryAcquire("late")).toBe(true);
    }
    expect(limiter.tryAcquire("late")).toBe(false);
  });
});
