// FILE: impl.ts
import type { Clock } from "../../lib/clock";

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

interface Entry<V> {
  readonly value: V;
  /** Absolute deadline; the entry is expired once `clock.now() >= expiresAt`. */
  readonly expiresAt: number;
}

function isLive<V>(entry: Entry<V> | undefined, now: number): entry is Entry<V> {
  return entry !== undefined && now < entry.expiresAt;
}

/** Reads an entry, dropping it if it has expired. */
function takeLive<K, V>(store: Map<K, Entry<V>>, key: K, now: number): Entry<V> | undefined {
  const entry = store.get(key);
  if (isLive(entry, now)) return entry;
  store.delete(key);
  return undefined;
}

/** Map iteration order is insertion order, so re-inserting marks `key` most-recently-used. */
function touch<K, V>(store: Map<K, Entry<V>>, key: K, entry: Entry<V>): void {
  store.delete(key);
  store.set(key, entry);
}

function purgeExpired<K, V>(store: Map<K, Entry<V>>, now: number): void {
  for (const [key, entry] of store) {
    if (!isLive(entry, now)) store.delete(key);
  }
}

function evictLru<K, V>(store: Map<K, Entry<V>>, capacity: number): void {
  while (store.size > capacity) {
    const oldest = store.keys().next();
    if (oldest.done === true) return;
    store.delete(oldest.value);
  }
}

function insert<K, V>(
  store: Map<K, Entry<V>>,
  key: K,
  value: V,
  capacity: number,
  now: number,
  ttlMs: number,
): void {
  store.delete(key);
  purgeExpired(store, now);
  store.set(key, { value, expiresAt: now + ttlMs });
  evictLru(store, capacity);
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock,
): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();
  const readLive = (key: K): Entry<V> | undefined => takeLive(store, key, clock.now());
  return {
    get: (key: K): V | undefined => {
      const entry = readLive(key);
      if (entry === undefined) return undefined;
      touch(store, key, entry);
      return entry.value;
    },
    set: (key: K, value: V): void =>
      insert(store, key, value, Math.max(capacity, 0), clock.now(), ttlMs),
    has: (key: K): boolean => readLive(key) !== undefined,
    get size(): number {
      purgeExpired(store, clock.now());
      return store.size;
    },
  };
}

// FILE: impl.test.ts
import { describe, expect, it } from "vitest";
import type { Clock } from "../../lib/clock";
import { createLruCache } from "./impl";

interface FakeClock extends Clock {
  advance(ms: number): void;
}

function createFakeClock(start: number = 0): FakeClock {
  let current = start;
  return {
    now: (): number => current,
    advance: (ms: number): void => {
      current += ms;
    },
  };
}

describe("createLruCache", () => {
  it("should return the stored value when the entry is live", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set("a", 1);
    expect(cache.get("a")).toBe(1);
    expect(cache.has("a")).toBe(true);
    expect(cache.size).toBe(1);
  });

  it("should return undefined when the key was never set", () => {
    const cache = createLruCache<string, number>(2, 1000, createFakeClock());
    expect(cache.get("missing")).toBeUndefined();
    expect(cache.has("missing")).toBe(false);
  });

  it("should evict the least-recently-used entry when capacity is exceeded", () => {
    const cache = createLruCache<string, number>(2, 1000, createFakeClock());
    cache.set("a", 1);
    cache.set("b", 2);
    cache.set("c", 3);
    expect(cache.get("a")).toBeUndefined();
    expect(cache.get("b")).toBe(2);
    expect(cache.get("c")).toBe(3);
    expect(cache.size).toBe(2);
  });

  it("should evict the other key when a live get refreshes recency", () => {
    const cache = createLruCache<string, number>(2, 1000, createFakeClock());
    cache.set("a", 1);
    cache.set("b", 2);
    expect(cache.get("a")).toBe(1);
    cache.set("c", 3);
    expect(cache.get("a")).toBe(1);
    expect(cache.get("b")).toBeUndefined();
  });

  it("should refresh recency when an existing key is overwritten", () => {
    const cache = createLruCache<string, number>(2, 1000, createFakeClock());
    cache.set("a", 1);
    cache.set("b", 2);
    cache.set("a", 10);
    cache.set("c", 3);
    expect(cache.get("a")).toBe(10);
    expect(cache.get("b")).toBeUndefined();
  });

  it("should expire an entry when now reaches insertion time plus ttl", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set("a", 1);
    clock.advance(999);
    expect(cache.get("a")).toBe(1);
    clock.advance(1);
    expect(cache.get("a")).toBeUndefined();
  });

  it("should report false from has when the entry is expired", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set("a", 1);
    clock.advance(1000);
    expect(cache.has("a")).toBe(false);
    expect(cache.has("a")).toBe(false);
    expect(cache.get("a")).toBeUndefined();
  });

  it("should not count expired entries when reporting size", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(3, 1000, clock);
    cache.set("a", 1);
    clock.advance(600);
    cache.set("b", 2);
    clock.advance(500);
    expect(cache.size).toBe(1);
  });

  it("should reset the ttl when an existing key is overwritten", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set("a", 1);
    clock.advance(900);
    cache.set("a", 2);
    clock.advance(900);
    expect(cache.get("a")).toBe(2);
  });

  it("should not let expired entries block eviction of live ones", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set("a", 1);
    clock.advance(1000);
    cache.set("b", 2);
    cache.set("c", 3);
    expect(cache.get("b")).toBe(2);
    expect(cache.get("c")).toBe(3);
    expect(cache.size).toBe(2);
  });

  it("should keep only the newest entry when capacity is one", () => {
    const cache = createLruCache<string, number>(1, 1000, createFakeClock());
    cache.set("a", 1);
    cache.set("b", 2);
    expect(cache.get("a")).toBeUndefined();
    expect(cache.get("b")).toBe(2);
    expect(cache.size).toBe(1);
  });

  it("should store nothing when capacity is zero", () => {
    const cache = createLruCache<string, number>(0, 1000, createFakeClock());
    cache.set("a", 1);
    expect(cache.get("a")).toBeUndefined();
    expect(cache.size).toBe(0);
  });

  it("should expire immediately when ttl is zero", () => {
    const cache = createLruCache<string, number>(2, 0, createFakeClock());
    cache.set("a", 1);
    expect(cache.get("a")).toBeUndefined();
    expect(cache.size).toBe(0);
  });
});
