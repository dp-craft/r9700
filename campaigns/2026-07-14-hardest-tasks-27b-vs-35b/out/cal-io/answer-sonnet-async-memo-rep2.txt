// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  return (key: K): Promise<V> => {
    const cached = cache.get(key);
    if (cached !== undefined) return cached;
    const pending = fn(key).catch((err: unknown): never => {
      cache.delete(key);
      throw err;
    });
    cache.set(key, pending);
    return pending;
  };
}

// FILE: impl.test.ts
import { describe, it, expect, vi } from "vitest";
import { createAsyncMemo } from "./impl";

describe("createAsyncMemo", () => {
  it("should call fn only once when concurrent calls share the same key", async (): Promise<void> => {
    let resolveFn: (value: string) => void = (): void => {};
    const fn = vi.fn((key: string): Promise<string> => {
      return new Promise<string>((resolve: (value: string) => void): void => {
        resolveFn = resolve;
      });
    });
    const memo = createAsyncMemo(fn);

    const p1 = memo("a");
    const p2 = memo("a");
    resolveFn("value");
    const [r1, r2] = await Promise.all([p1, p2]);

    expect(fn).toHaveBeenCalledTimes(1);
    expect(r1).toBe("value");
    expect(r2).toBe("value");
  });

  it("should retry fn when the previous call rejected", async (): Promise<void> => {
    let callCount = 0;
    const fn = vi.fn((key: string): Promise<string> => {
      callCount += 1;
      if (callCount === 1) return Promise.reject(new Error("fail"));
      return Promise.resolve(`resolved-${key}`);
    });
    const memo = createAsyncMemo(fn);

    await expect(memo("a")).rejects.toThrow("fail");
    const result = await memo("a");

    expect(result).toBe("resolved-a");
    expect(fn).toHaveBeenCalledTimes(2);
  });

  it("should cache hit when a call resolved successfully before", async (): Promise<void> => {
    const fn = vi.fn((key: string): Promise<string> => Promise.resolve(`resolved-${key}`));
    const memo = createAsyncMemo(fn);

    const first = await memo("a");
    const second = await memo("a");

    expect(first).toBe("resolved-a");
    expect(second).toBe("resolved-a");
    expect(fn).toHaveBeenCalledTimes(1);
  });
});
