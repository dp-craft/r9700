// FILE: impl.ts
export type Result =
  | { readonly ok: true; readonly value: number }
  | { readonly ok: false; readonly error: string };

type Token =
  | { readonly kind: "num"; readonly value: number }
  | { readonly kind: "op"; readonly text: string };

interface Cursor {
  readonly tokens: readonly Token[];
  index: number;
}

const DIGIT = /[0-9]/;
const WHITESPACE = /\s/;
const PUNCTUATION = "+-*/()";
const ADD_OPS: readonly string[] = ["+", "-"];
const MUL_OPS: readonly string[] = ["*", "/"];
const CLOSE_PAREN: readonly string[] = [")"];

/** Internal-only signal; caught in `evaluate` so it never escapes to the caller. */
class ParseError extends Error {}

function fail(message: string): never {
  throw new ParseError(message);
}

function scanNumber(expr: string, start: number): { readonly end: number; readonly value: number } {
  let end = start;
  while (end < expr.length && DIGIT.test(expr.charAt(end))) {
    end += 1;
  }
  return { end, value: Number(expr.slice(start, end)) };
}

function tokenize(expr: string): readonly Token[] {
  const tokens: Token[] = [];
  let i = 0;
  while (i < expr.length) {
    const ch = expr.charAt(i);
    if (WHITESPACE.test(ch)) {
      i += 1;
    } else if (DIGIT.test(ch)) {
      const scanned = scanNumber(expr, i);
      tokens.push({ kind: "num", value: scanned.value });
      i = scanned.end;
    } else if (PUNCTUATION.includes(ch)) {
      tokens.push({ kind: "op", text: ch });
      i += 1;
    } else {
      fail(`unexpected character '${ch}'`);
    }
  }
  return tokens;
}

function peek(cur: Cursor): Token | undefined {
  return cur.tokens[cur.index];
}

function matchOp(cur: Cursor, ops: readonly string[]): string | undefined {
  const tok = peek(cur);
  if (tok !== undefined && tok.kind === "op" && ops.includes(tok.text)) {
    cur.index += 1;
    return tok.text;
  }
  return undefined;
}

function divide(left: number, right: number): number {
  if (right === 0) {
    fail("division by zero");
  }
  return Math.trunc(left / right);
}

function parseExpr(cur: Cursor): number {
  let left = parseTerm(cur);
  for (;;) {
    const op = matchOp(cur, ADD_OPS);
    if (op === undefined) {
      return left;
    }
    const right = parseTerm(cur);
    left = op === "+" ? left + right : left - right;
  }
}

function parseTerm(cur: Cursor): number {
  let left = parseFactor(cur);
  for (;;) {
    const op = matchOp(cur, MUL_OPS);
    if (op === undefined) {
      return left;
    }
    const right = parseFactor(cur);
    left = op === "*" ? left * right : divide(left, right);
  }
}

function parseFactor(cur: Cursor): number {
  const sign = matchOp(cur, ADD_OPS);
  if (sign !== undefined) {
    const operand = parseFactor(cur);
    return sign === "-" ? -operand : operand;
  }
  return parsePrimary(cur);
}

function parsePrimary(cur: Cursor): number {
  const tok = peek(cur);
  if (tok === undefined) {
    fail("unexpected end of input");
  }
  cur.index += 1;
  if (tok.kind === "num") {
    return tok.value;
  }
  if (tok.text === "(") {
    return parseGroup(cur);
  }
  fail(`unexpected token '${tok.text}'`);
}

function parseGroup(cur: Cursor): number {
  const value = parseExpr(cur);
  if (matchOp(cur, CLOSE_PAREN) === undefined) {
    fail("unbalanced parentheses");
  }
  return value;
}

export function evaluate(expr: string): Result {
  try {
    const cur: Cursor = { tokens: tokenize(expr), index: 0 };
    const value = parseExpr(cur);
    if (cur.index !== cur.tokens.length) {
      fail("unexpected trailing input");
    }
    if (!Number.isFinite(value)) {
      fail("result is not a finite number");
    }
    return { ok: true, value };
  } catch (err: unknown) {
    if (err instanceof ParseError) {
      return { ok: false, error: err.message };
    }
    return { ok: false, error: "invalid expression" };
  }
}

// FILE: impl.test.ts
import { describe, expect, it } from "vitest";
import { evaluate, type Result } from "./impl";

function valueOf(result: Result): number | undefined {
  return result.ok ? result.value : undefined;
}

describe("evaluate", () => {
  it("should return the number itself when the expression is a single literal", () => {
    expect(evaluate("42")).toEqual({ ok: true, value: 42 });
  });

  it("should apply multiplication before addition when precedence differs", () => {
    expect(valueOf(evaluate("2+3*4"))).toBe(14);
  });

  it("should apply division before subtraction when precedence differs", () => {
    expect(valueOf(evaluate("20-8/4"))).toBe(18);
  });

  it("should subtract left to right when several minus operators chain", () => {
    expect(valueOf(evaluate("10-3-2"))).toBe(5);
  });

  it("should divide left to right when several divide operators chain", () => {
    expect(valueOf(evaluate("8/2/2"))).toBe(2);
  });

  it("should override precedence when parentheses group an addition", () => {
    expect(valueOf(evaluate("(2+3)*4"))).toBe(20);
  });

  it("should evaluate correctly when parentheses are nested", () => {
    expect(valueOf(evaluate("((1+2)*(3+4))-5"))).toBe(16);
  });

  it("should ignore whitespace when it surrounds tokens and operators", () => {
    expect(valueOf(evaluate("  12 *  ( 3 +\t1 ) \n"))).toBe(48);
  });

  it("should negate the operand when a leading unary minus is used", () => {
    expect(valueOf(evaluate("-(4+1)*2"))).toBe(-10);
  });

  it("should truncate toward zero when division is inexact", () => {
    expect(valueOf(evaluate("7/2"))).toBe(3);
  });

  it("should return an error when parentheses are unbalanced", () => {
    const result = evaluate("(1+2");
    expect(result.ok).toBe(false);
    expect(result).toEqual({ ok: false, error: "unbalanced parentheses" });
  });

  it("should return an error when a closing parenthesis is unmatched", () => {
    expect(evaluate("1+2)").ok).toBe(false);
  });

  it("should return an error when an operand is missing", () => {
    expect(evaluate("1+").ok).toBe(false);
  });

  it("should return an error when the input is empty", () => {
    expect(evaluate("   ").ok).toBe(false);
  });

  it("should return an error when the input contains an unexpected character", () => {
    expect(evaluate("2 $ 3")).toEqual({ ok: false, error: "unexpected character '$'" });
  });

  it("should return an error when dividing by zero", () => {
    expect(evaluate("1/0")).toEqual({ ok: false, error: "division by zero" });
  });

  it("should return an error when dividing by a parenthesized zero", () => {
    expect(evaluate("8/(4-4)")).toEqual({ ok: false, error: "division by zero" });
  });

  it("should not throw when the input is malformed", () => {
    expect(() => evaluate("((*)/")).not.toThrow();
  });
});
