// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  return (key: K): Promise<V> => {
    const cached = cache.get(key);
    if (cached !== undefined) return cached;
    const pending = fn(key);
    cache.set(key, pending);
    pending.catch((): void => {
      cache.delete(key);
    });
    return pending;
  };
}

// FILE: impl.test.ts
import { describe, it, expect, vi } from "vitest";
import { createAsyncMemo } from "./impl";

function createDeferred<T>(): { promise: Promise<T>; resolve: (value: T) => void } {
  let resolve: (value: T) => void = (): void => {};
  const promise = new Promise<T>((res): void => {
    resolve = res;
  });
  return { promise, resolve };
}

describe("createAsyncMemo", (): void => {
  it("should call fn only once when called concurrently for the same key", async (): Promise<void> => {
    const deferred = createDeferred<string>();
    const fn = vi.fn((): Promise<string> => deferred.promise);
    const memo = createAsyncMemo(fn);

    const p1 = memo("a");
    const p2 = memo("a");
    deferred.resolve("value-a");

    const [r1, r2] = await Promise.all([p1, p2]);

    expect(fn).toHaveBeenCalledTimes(1);
    expect(r1).toBe("value-a");
    expect(r2).toBe("value-a");
  });

  it("should retry fn when the previous call rejected", async (): Promise<void> => {
    let callCount = 0;
    const fn = vi.fn((): Promise<string> => {
      callCount += 1;
      if (callCount === 1) return Promise.reject(new Error("boom"));
      return Promise.resolve("recovered");
    });
    const memo = createAsyncMemo(fn);

    await expect(memo("a")).rejects.toThrow("boom");
    const result = await memo("a");

    expect(result).toBe("recovered");
    expect(fn).toHaveBeenCalledTimes(2);
  });

  it("should reuse the cached value when a previous call succeeded", async (): Promise<void> => {
    const fn = vi.fn((key: string): Promise<string> => Promise.resolve(`resolved-${key}`));
    const memo = createAsyncMemo(fn);

    const first = await memo("a");
    const second = await memo("a");

    expect(first).toBe("resolved-a");
    expect(second).toBe("resolved-a");
    expect(fn).toHaveBeenCalledTimes(1);
  });
});
