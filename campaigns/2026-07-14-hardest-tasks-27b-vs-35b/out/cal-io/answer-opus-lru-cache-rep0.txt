// FILE: impl.ts
import type { Clock } from "../../lib/clock";

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

interface Entry<V> {
  readonly value: V;
  readonly expiresAt: number;
}

/** An entry is live while `now` has not yet reached its expiry instant. */
function isLive<V>(entry: Entry<V> | undefined, now: number): entry is Entry<V> {
  return entry !== undefined && now < entry.expiresAt;
}

/** Drops every expired entry so it can neither be seen nor block eviction. */
function purgeExpired<K, V>(store: Map<K, Entry<V>>, now: number): void {
  for (const [key, entry] of store) {
    if (!isLive(entry, now)) {
      store.delete(key);
    }
  }
}

function readLive<K, V>(store: Map<K, Entry<V>>, key: K, now: number): Entry<V> | undefined {
  const entry = store.get(key);
  if (isLive(entry, now)) {
    return entry;
  }
  store.delete(key);
  return undefined;
}

/** Map iteration order is insertion order, so re-inserting marks the key most-recently-used. */
function touch<K, V>(store: Map<K, Entry<V>>, key: K, entry: Entry<V>): void {
  store.delete(key);
  store.set(key, entry);
}

function readAndTouch<K, V>(store: Map<K, Entry<V>>, key: K, now: number): V | undefined {
  const entry = readLive(store, key, now);
  if (entry === undefined) {
    return undefined;
  }
  touch(store, key, entry);
  return entry.value;
}

function evictLru<K, V>(store: Map<K, Entry<V>>, capacity: number): void {
  while (store.size > capacity) {
    const oldest = store.keys().next();
    if (oldest.done === true) {
      return;
    }
    store.delete(oldest.value);
  }
}

function insert<K, V>(
  store: Map<K, Entry<V>>,
  key: K,
  value: V,
  capacity: number,
  ttlMs: number,
  now: number,
): void {
  purgeExpired(store, now);
  store.delete(key);
  store.set(key, { value, expiresAt: now + ttlMs });
  evictLru(store, capacity);
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock,
): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();
  return {
    get: (key: K): V | undefined => readAndTouch(store, key, clock.now()),
    set: (key: K, value: V): void => insert(store, key, value, capacity, ttlMs, clock.now()),
    has: (key: K): boolean => readLive(store, key, clock.now()) !== undefined,
    get size(): number {
      purgeExpired(store, clock.now());
      return store.size;
    },
  };
}

// FILE: impl.test.ts
import { describe, expect, it } from "vitest";
import type { Clock } from "../../lib/clock";
import { createLruCache } from "./impl";

interface FakeClock extends Clock {
  advance(ms: number): void;
}

function createFakeClock(start: number = 0): FakeClock {
  let current = start;
  return {
    now: (): number => current,
    advance: (ms: number): void => {
      current += ms;
    },
  };
}

describe("createLruCache", () => {
  it("should return the stored value when the entry is live", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set("a", 1);

    expect(cache.get("a")).toBe(1);
    expect(cache.has("a")).toBe(true);
    expect(cache.size).toBe(1);
  });

  it("should evict the least-recently-used entry when capacity is exceeded", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set("a", 1);
    cache.set("b", 2);
    cache.set("c", 3);

    expect(cache.get("a")).toBeUndefined();
    expect(cache.get("b")).toBe(2);
    expect(cache.get("c")).toBe(3);
    expect(cache.size).toBe(2);
  });

  it("should keep an entry alive when a live get refreshes its recency", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set("a", 1);
    cache.set("b", 2);
    cache.get("a");
    cache.set("c", 3);

    expect(cache.get("a")).toBe(1);
    expect(cache.get("b")).toBeUndefined();
  });

  it("should refresh recency when an existing key is overwritten by set", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set("a", 1);
    cache.set("b", 2);
    cache.set("a", 10);
    cache.set("c", 3);

    expect(cache.get("a")).toBe(10);
    expect(cache.get("b")).toBeUndefined();
    expect(cache.size).toBe(2);
  });

  it("should return undefined when the entry has reached exactly its ttl", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set("a", 1);
    clock.advance(999);
    expect(cache.get("a")).toBe(1);

    clock.advance(1);
    expect(cache.get("a")).toBeUndefined();
  });

  it("should report false when has is called on an expired entry", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set("a", 1);
    clock.advance(1000);

    expect(cache.has("a")).toBe(false);
    expect(cache.has("a")).toBe(false);
    expect(cache.get("a")).toBeUndefined();
  });

  it("should exclude expired entries when size is read", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(3, 1000, clock);

    cache.set("a", 1);
    clock.advance(600);
    cache.set("b", 2);
    clock.advance(500);

    expect(cache.size).toBe(1);
    expect(cache.get("b")).toBe(2);
  });

  it("should evict an expired entry before a live one when capacity is reached", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);

    cache.set("a", 1);
    clock.advance(900);
    cache.set("b", 2);
    clock.advance(200);
    cache.set("c", 3);

    expect(cache.get("a")).toBeUndefined();
    expect(cache.get("b")).toBe(2);
    expect(cache.get("c")).toBe(3);
    expect(cache.size).toBe(2);
  });

  it("should keep only the newest entry when capacity is one", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(1, 1000, clock);

    cache.set("a", 1);
    cache.set("b", 2);

    expect(cache.has("a")).toBe(false);
    expect(cache.get("b")).toBe(2);
    expect(cache.size).toBe(1);
  });

  it("should store nothing when capacity is zero", () => {
    const clock = createFakeClock();
    const cache = createLruCache<string, number>(0, 1000, clock);

    cache.set("a", 1);

    expect(cache.get("a")).toBeUndefined();
    expect(cache.size).toBe(0);
  });

  it("should keep distinct generic key and value types when structured data is cached", () => {
    const clock = createFakeClock();
    const cache = createLruCache<number, { readonly label: string }>(2, 500, clock);

    cache.set(7, { label: "seven" });

    expect(cache.get(7)).toEqual({ label: "seven" });
    expect(cache.get(8)).toBeUndefined();
  });
});
