// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  
  return (key: K): Promise<V> => {
    const cached = cache.get(key);
    if (cached !== undefined) return cached;
    
    const p = fn(key);
    const result = p.catch((error: unknown) => {
      cache.delete(key);
      throw error;
    });
    
    cache.set(key, result);
    return result;
  };
}

// FILE: impl.test.ts
import { describe, expect, it, vi } from 'vitest';

import { createAsyncMemo } from './impl';

describe('createAsyncMemo', () => {
  describe('concurrent dedup', () => {
    it('should invoke fn once when multiple concurrent calls share the same key', async (): Promise<void> => {
      let callCount = 0;
      const fn = vi.fn(async (): Promise<string> => {
        callCount++;
        await new Promise<void>(resolve => setTimeout(resolve, 10));
        return 'value';
      });
      
      const memoized = createAsyncMemo(fn);
      
      const [r1, r2, r3] = await Promise.all([
        memoized('key'),
        memoized('key'),
        memoized('key'),
      ]);
      
      expect(callCount).toBe(1);
      expect(r1).toBe('value');
      expect(r2).toBe('value');
      expect(r3).toBe('value');
    });
  });

  describe('retry-after-rejection', () => {
    it('should retry fn when a previous call rejected for the same key', async (): Promise<void> => {
      let callCount = 0;
      const fn = vi.fn(async (): Promise<string> => {
        callCount++;
        if (callCount === 1) throw new Error('fail');
        return 'recovered';
      });
      
      const memoized = createAsyncMemo(fn);
      
      await expect(memoized('key')).rejects.toThrow('fail');
      
      const result = await memoized('key');
      
      expect(callCount).toBe(2);
      expect(result).toBe('recovered');
    });
  });

  describe('cache-hit-after-success', () => {
    it('should reuse cached value when fn resolves successfully for the same key', async (): Promise<void> => {
      let callCount = 0;
      const fn = vi.fn(async (): Promise<string> => {
        callCount++;
        return 'cached';
      });
      
      const memoized = createAsyncMemo(fn);
      
      const r1 = await memoized('key');
      const r2 = await memoized('key');
      
      expect(callCount).toBe(1);
      expect(r1).toBe('cached');
      expect(r2).toBe('cached');
    });
  });
});
