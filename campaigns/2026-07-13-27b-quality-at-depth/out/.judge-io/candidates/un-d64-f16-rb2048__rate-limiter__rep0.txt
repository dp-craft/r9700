// FILE: impl.ts
import type { Clock } from '../../lib/clock';

export interface RateLimiter {
  tryAcquire(key: string, tokens?: number): boolean;
}

interface BucketState {
  tokens: number;
  lastRefillAt: number;
}

const DEFAULT_TOKENS = 1;

export function createRateLimiter(
  capacity: number,
  refillPerSec: number,
  clock: Clock
): RateLimiter {
  const buckets = new Map<string, BucketState>();

  const refillBucket = (bucket: BucketState): void => {
    const now = clock.now();
    const elapsedSec = (now - bucket.lastRefillAt) / 1000;
    bucket.tokens = Math.min(capacity, bucket.tokens + elapsedSec * refillPerSec);
    bucket.lastRefillAt = now;
  };

  return {
    tryAcquire(key: string, tokens: number = DEFAULT_TOKENS): boolean {
      const bucket = buckets.get(key) ?? { tokens: capacity, lastRefillAt: clock.now() };
      refillBucket(bucket);
      const acquired = bucket.tokens >= tokens;
      if (acquired) {
        bucket.tokens -= tokens;
      }
      buckets.set(key, bucket);
      return acquired;
    },
  };
}

// FILE: impl.test.ts
import { beforeEach, describe, expect, it, vi } from 'vitest';

import type { Clock } from '../../lib/clock';
import { createRateLimiter } from './impl';

const buildFakeClock = (initialMs: number): Clock & { advanceByMs: (ms: number) => void } => {
  let current = initialMs;
  return {
    now: () => current,
    advanceByMs: (ms: number) => { current += ms; },
  };
};

describe('createRateLimiter', () => {
  let clock: Clock & { advanceByMs: (ms: number) => void };

  beforeEach(() => {
    clock = buildFakeClock(0);
  });

  it('should allow first-burst up to capacity when bucket is full', () => {
    const limiter = createRateLimiter(3, 1, clock);
    expect(limiter.tryAcquire('key-a')).toBe(true);
    expect(limiter.tryAcquire('key-a')).toBe(true);
    expect(limiter.tryAcquire('key-a')).toBe(true);
  });

  it('should reject acquisition when bucket is empty', () => {
    const limiter = createRateLimiter(2, 1, clock);
    limiter.tryAcquire('key-a');
    limiter.tryAcquire('key-a');
    expect(limiter.tryAcquire('key-a')).toBe(false);
  });

  it('should lazily refill tokens over time when clock advances', () => {
    const limiter = createRateLimiter(2, 2, clock);
    limiter.tryAcquire('key-a');
    limiter.tryAcquire('key-a');
    expect(limiter.tryAcquire('key-a')).toBe(false);

    clock.advanceByMs(1000);
    expect(limiter.tryAcquire('key-a')).toBe(true);
    expect(limiter.tryAcquire('key-a')).toBe(true);
  });

  it('should cap refilled tokens at capacity when exceeding limit', () => {
    const limiter = createRateLimiter(2, 10, clock);
    limiter.tryAcquire('key-a');
    limiter.tryAcquire('key-a');

    clock.advanceByMs(1000);
    expect(limiter.tryAcquire('key-a')).toBe(true);
    expect(limiter.tryAcquire('key-a')).toBe(true);
    expect(limiter.tryAcquire('key-a')).toBe(false);
  });

  it('should maintain per-key isolation when different keys are used', () => {
    const limiter = createRateLimiter(1, 1, clock);
    expect(limiter.tryAcquire('key-a')).toBe(true);
    expect(limiter.tryAcquire('key-b')).toBe(true);
    expect(limiter.tryAcquire('key-a')).toBe(false);
    expect(limiter.tryAcquire('key-b')).toBe(false);
  });

  it('should support custom token cost when tokens parameter is provided', () => {
    const limiter = createRateLimiter(5, 1, clock);
    expect(limiter.tryAcquire('key-a', 3)).toBe(true);
    expect(limiter.tryAcquire('key-a', 3)).toBe(false);
    expect(limiter.tryAcquire('key-a', 2)).toBe(true);
  });
});
