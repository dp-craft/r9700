// FILE: impl.ts
import type { Clock } from '../../lib/clock';

export interface RateLimiter {
  tryAcquire(key: string, tokens?: number): boolean;
}

interface BucketState {
  tokens: number;
  lastRefillAt: number;
}

const DEFAULT_TOKENS = 1;

export function createRateLimiter(
  capacity: number,
  refillPerSec: number,
  clock: Clock
): RateLimiter {
  const buckets = new Map<string, BucketState>();

  const refillBucket = (bucket: BucketState): void => {
    const now = clock.now();
    const elapsedSec = (now - bucket.lastRefillAt) / 1000;
    bucket.tokens = Math.min(capacity, bucket.tokens + elapsedSec * refillPerSec);
    bucket.lastRefillAt = now;
  };

  return {
    tryAcquire(key: string, tokens: number = DEFAULT_TOKENS): boolean {
      const bucket = buckets.get(key) ?? { tokens: capacity, lastRefillAt: clock.now() };
      refillBucket(bucket);
      const acquired = bucket.tokens >= tokens;
      if (acquired) {
        bucket.tokens -= tokens;
      }
      buckets.set(key, bucket);
      return acquired;
    },
  };
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import type { Clock } from '../../lib/clock';
import { createRateLimiter } from './impl';

describe('createRateLimiter', () => {
  const buildFakeClock = (initialMs: number): Clock => {
    let time = initialMs;
    return {
      now: (): number => time,
      tick: (deltaMs: number): void => { time += deltaMs; },
    };
  };

  it('should allow first-burst requests up to capacity when bucket is full', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(3, 1, clock);

    expect(limiter.tryAcquire('user-1')).toBe(true);
    expect(limiter.tryAcquire('user-1')).toBe(true);
    expect(limiter.tryAcquire('user-1')).toBe(true);
  });

  it('should reject requests when bucket is empty', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(2, 1, clock);

    limiter.tryAcquire('user-2');
    limiter.tryAcquire('user-2');

    expect(limiter.tryAcquire('user-2')).toBe(false);
  });

  it('should lazily refill tokens over elapsed time', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(2, 2, clock);

    limiter.tryAcquire('user-3');
    limiter.tryAcquire('user-3');

    expect(limiter.tryAcquire('user-3')).toBe(false);

    clock.tick(500);
    expect(limiter.tryAcquire('user-3')).toBe(true);
  });

  it('should cap refilled tokens at bucket capacity', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(2, 10, clock);

    limiter.tryAcquire('user-4');
    clock.tick(1000);
    limiter.tryAcquire('user-4');
    clock.tick(1000);

    expect(limiter.tryAcquire('user-4')).toBe(true);
    expect(limiter.tryAcquire('user-4')).toBe(false);
  });

  it('should maintain per-key isolation between different keys', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(1, 0, clock);

    expect(limiter.tryAcquire('key-a')).toBe(true);
    expect(limiter.tryAcquire('key-b')).toBe(true);
    expect(limiter.tryAcquire('key-a')).toBe(false);
  });

  it('should deduct custom token amounts when requested', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(5, 0, clock);

    expect(limiter.tryAcquire('user-5', 3)).toBe(true);
    expect(limiter.tryAcquire('user-5', 3)).toBe(false);
    expect(limiter.tryAcquire('user-5', 2)).toBe(true);
  });

  it('should reject custom token requests exceeding remaining balance', () => {
    const clock = buildFakeClock(0);
    const limiter = createRateLimiter(4, 0, clock);

    expect(limiter.tryAcquire('user-6', 3)).toBe(true);
    expect(limiter.tryAcquire('user-6', 2)).toBe(false);
    expect(limiter.tryAcquire('user-6', 1)).toBe(true);
  });
});
