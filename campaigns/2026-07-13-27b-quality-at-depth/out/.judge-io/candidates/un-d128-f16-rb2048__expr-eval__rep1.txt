// FILE: impl.ts
export type Result =
  | { readonly ok: true; readonly value: number }
  | { readonly ok: false; readonly error: string };

const err = (msg: string): Result => ({ ok: false, error: msg });
const ok = (val: number): Result => ({ ok: true, value: val });

interface Parser {
  readonly src: string;
  pos: number;
}

function skipWs(p: Parser): void {
  while (p.pos < p.src.length && /\s/.test(p.src[p.pos])) {
    p.pos++;
  }
}

function peek(p: Parser): string | undefined {
  return p.pos < p.src.length ? p.src[p.pos] : undefined;
}

function parseNum(p: Parser): Result {
  if (p.pos >= p.src.length || !/\d/.test(p.src[p.pos])) {
    return err('Missing operand');
  }
  let n = '';
  while (p.pos < p.src.length && /\d/.test(p.src[p.pos])) {
    n += p.src[p.pos++];
  }
  return ok(Number(n));
}

function parseFact(p: Parser): Result {
  skipWs(p);
  const ch = peek(p);
  if (ch === '(') {
    p.pos++;
    const r = parseExpr(p);
    if (!r.ok) return r;
    skipWs(p);
    if (peek(p) !== ')') return err('Unbalanced parentheses');
    p.pos++;
    return r;
  }
  if (ch === '-' || ch === '+') {
    p.pos++;
    const r = parseFact(p);
    return r.ok ? ok(ch === '-' ? -r.value : r.value) : r;
  }
  return parseNum(p);
}

function parseTerm(p: Parser): Result {
  let r = parseFact(p);
  if (!r.ok) return r;
  while (true) {
    skipWs(p);
    const op = peek(p);
    if (op !== '*' && op !== '/') break;
    p.pos++;
    const rhs = parseFact(p);
    if (!rhs.ok) return rhs;
    if (op === '/' && rhs.value === 0) return err('Division by zero');
    r = ok(op === '/' ? r.value / rhs.value : r.value * rhs.value);
  }
  return r;
}

function parseExpr(p: Parser): Result {
  let r = parseTerm(p);
  if (!r.ok) return r;
  while (true) {
    skipWs(p);
    const op = peek(p);
    if (op !== '+' && op !== '-') break;
    p.pos++;
    const rhs = parseTerm(p);
    if (!rhs.ok) return rhs;
    r = ok(op === '-' ? r.value - rhs.value : r.value + rhs.value);
  }
  return r;
}

export function evaluate(expr: string): Result {
  try {
    const p: Parser = { src: expr, pos: 0 };
    const res = parseExpr(p);
    if (!res.ok) return res;
    skipWs(p);
    if (p.pos < p.src.length) return err('Unexpected character');
    return res;
  } catch (e: unknown) {
    return err(e instanceof Error ? e.message : 'Evaluation failed');
  }
}

import { describe, expect, it } from 'vitest';
import { evaluate } from './impl';

describe('evaluate', () => {
  it('should compute addition correctly when given valid expression', () => {
    expect(evaluate('2 + 3')).toEqual({ ok: true, value: 5 });
  });

  it('should compute subtraction correctly when given valid expression', () => {
    expect(evaluate('10 - 4')).toEqual({ ok: true, value: 6 });
  });
