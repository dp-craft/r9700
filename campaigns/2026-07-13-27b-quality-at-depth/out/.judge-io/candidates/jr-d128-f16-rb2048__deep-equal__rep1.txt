// FILE: impl.ts
function deepEqual(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (a === null || b === null) return false;
  if (typeof a !== typeof b) return false;

  if (typeof a === 'number') {
    return Number.isNaN(a) && Number.isNaN(b);
  }

  const aIsArray = Array.isArray(a);
  const bIsArray = Array.isArray(b);

  if (aIsArray !== bIsArray) return false;

  if (aIsArray) {
    const arrA = a as unknown[];
    const arrB = b as unknown[];
    if (arrA.length !== arrB.length) return false;
    return arrA.every((item, i) => deepEqual(item, arrB[i]));
  }

  if (typeof a === 'object') {
    const keysA = Object.keys(a);
    const keysB = Object.keys(b);
    if (keysA.length !== keysB.length) return false;
    const objA = a as Record<string, unknown>;
    const objB = b as Record<string, unknown>;
    return keysA.every(key => deepEqual(objA[key], objB[key]));
  }

  return false;
}

export { deepEqual };

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import { deepEqual } from './impl';

describe('deepEqual', () => {
  describe('primitives', () => {
    it('should return true when both values are NaN', () => {
      expect(deepEqual(NaN, NaN)).toBe(true);
    });

    it('should return true when both strings are equal', () => {
      expect(deepEqual('hello', 'hello')).toBe(true);
    });

    it('should return false when strings differ', () => {
      expect(deepEqual('hello', 'world')).toBe(false);
    });

    it('should return true when both numbers are equal', () => {
      expect(deepEqual(42, 42)).toBe(true);
    });

    it('should return false when numbers differ', () => {
      expect(deepEqual(42, 43)).toBe(false);
    });

    it('should return true when both booleans are equal', () => {
      expect(deepEqual(true, true)).toBe(true);
    });

    it('should return false when booleans differ', () => {
      expect(deepEqual(true, false)).toBe(false);
    });

    it('should return true when both are null', () => {
      expect(deepEqual(null, null)).toBe(true);
    });

    it('should return true when both are undefined', () => {
      expect(deepEqual(undefined, undefined)).toBe(true);
    });
  });

  describe('null vs object', () => {
    it('should return false when null is compared to empty object', () => {
      expect(deepEqual(null, {})).toBe(false);
    });

    it('should return false when empty object is compared to null', () => {
      expect(deepEqual({}, null)).toBe(false);
    });
  });

  describe('array vs object', () => {
    it('should return false when array is compared to object with matching indices', () => {
      expect(deepEqual([1, 2], { 0: 1, 1: 2 })).toBe(false);
    });

    it('should return false when object with matching indices is compared to array', () => {
      expect(deepEqual({ 0: 1, 1: 2 }, [1, 2])).toBe(false);
    });
  });

  describe('arrays', () => {
    it('should return true when arrays have same length and equal elements', () => {
      expect(deepEqual([1, 2, 3], [1, 2, 3])).toBe(true);
    });

    it('should return false when arrays have different lengths', () => {
      expect(deepEqual([1, 2], [1, 2, 3])).toBe(false);
    });

    it('should return true for deeply nested arrays', () => {
      expect(deepEqual([[1, 2], [3]], [[1, 2], [3]])).toBe(true);
    });

    it('should return false for deeply nested arrays with mismatch', () => {
      expect(deepEqual([[1, 2], [3]], [[1, 2], [3, 4]])).toBe(false);
    });
  });

  describe('objects', () => {
    it('should return true when objects have same keys and equal values', () => {
      expect(deepEqual({ a: 1, b: 2 }, { a: 1, b: 2 })).toBe(true);
    });

    it('should return false when key-count mismatches (more keys)', () => {
      expect(deepEqual({ a: 1 }, { a: 1, b: 2 })).toBe(false);
    });

    it('should return false when key-count mismatches (fewer keys)', () => {
      expect(deepEqual({ a: 1, b: 2 }, { a: 1 })).toBe(false);
    });

    it('should return false when keys differ but same count', () => {
      expect(deepEqual({ a: 1 }, { b: 1 })).toBe(false);
    });

    it('should return true for deeply nested objects', () => {
      expect(deepEqual({ a: { b: [1, 2] } }, { a: { b: [1, 2] } })).toBe(true);
    });

    it('should return false for deeply nested objects with mismatch', () => {
      expect(deepEqual({ a: { b: [1, 2] } }, { a: { b: [1, 3] } })).toBe(false);
    });
  });

  describe('mixed types', () => {
    it('should return false when comparing array to non-array object', () => {
      expect(deepEqual([], {})).toBe(false);
    });

    it('should return false when comparing number to string', () => {
      expect(deepEqual(1, '1')).toBe(false);
    });

    it('should return false when comparing boolean to number', () => {
      expect(deepEqual(true, 1)).toBe(false);
    });
  });
});
