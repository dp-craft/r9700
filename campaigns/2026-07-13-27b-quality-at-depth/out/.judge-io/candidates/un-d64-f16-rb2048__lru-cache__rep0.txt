// FILE: impl.ts
import type { Clock } from '../../lib/clock';

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

interface Entry<V> {
  value: V;
  insertedAt: number;
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();

  function isExpired(entry: Entry<V>): boolean {
    return clock.now() >= entry.insertedAt + ttlMs;
  }

  function refreshRecency(key: K): void {
    const entry = store.get(key);
    if (entry !== undefined) {
      store.delete(key);
      store.set(key, entry);
    }
  }

  function evictOldest(): void {
    if (store.size <= capacity) return;
    const oldestKey = store.keys().next().value;
    if (oldestKey !== undefined) {
      store.delete(oldestKey);
      evictOldest();
    }
  }

  function cleanExpired(): void {
    const now = clock.now();
    for (const [key, entry] of store.entries()) {
      if (now >= entry.insertedAt + ttlMs) {
        store.delete(key);
      }
    }
  }

  function getLiveValue(key: K): V | undefined {
    const entry = store.get(key);
    if (entry === undefined || isExpired(entry)) {
      if (entry !== undefined) store.delete(key);
      return undefined;
    }
    refreshRecency(key);
    return entry.value;
  }

  return {
    get(key: K): V | undefined {
      return getLiveValue(key);
    },
    set(key: K, value: V): void {
      store.delete(key);
      store.set(key, { value, insertedAt: clock.now() });
      evictOldest();
    },
    has(key: K): boolean {
      const entry = store.get(key);
      if (entry === undefined || isExpired(entry)) {
        if (entry !== undefined) store.delete(key);
        return false;
      }
      return true;
    },
    get size(): number {
      cleanExpired();
      return store.size;
    }
  };
}

// FILE: impl.test.ts
import { beforeEach, describe, expect, it, vi } from 'vitest';

import { createLruCache } from './impl';
import type { Clock } from '../../lib/clock';

describe('createLruCache', () => {
  let clock: Clock;
  let advanceTime: (ms: number) => void;

  beforeEach(() => {
    let now = 0;
    clock = { now: () => now };
    advanceTime = (ms: number) => { now += ms; };
  });

  it('should return undefined for absent keys', () => {
    const cache = createLruCache<string, number>(2, 1000, clock);
    expect(cache.get('missing')).toBeUndefined();
  });

  it('should store and retrieve values within TTL', () => {
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    expect(cache.get('a')).toBe(1);
  });

  it('should return undefined for expired entries', () => {
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    advanceTime(1000);
    expect(cache.get('a')).toBeUndefined();
  });

  it('should not resurrect expired entries via has()', () => {
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    advanceTime(1000);
    expect(cache.has('a')).toBe(false);
    expect(cache.get('a')).toBeUndefined();
  });

  it('should exclude expired entries from size', () => {
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    expect(cache.size).toBe(2);
    advanceTime(1000);
    expect(cache.size).toBe(0);
  });

  it('should refresh recency on live get()', () => {
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.get('a');
    cache.set('c', 3);
    expect(cache.get('a')).toBe(1);
    expect(cache.get('b')).toBeUndefined();
  });

  it('should evict least-recently-used entry when capacity exceeded', () => {
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('c', 3);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBe(2);
    expect(cache.get('c')).toBe(3);
  });

  it('should not evict live entries when expired ones fill capacity', () => {
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    advanceTime(1000);
    cache.set('b', 2);
    expect(cache.size).toBe(1);
    expect(cache.get('b')).toBe(2);
  });

  it('should handle capacity edge case of 1', () => {
    const cache = createLruCache<string, number>(1, 1000, clock);
    cache.set('x', 10);
    cache.set('y', 20);
    expect(cache.get('x')).toBeUndefined();
    expect(cache.get('y')).toBe(20);
    expect(cache.size).toBe(1);
  });

  it('should correctly track size with mixed live and expired entries', () => {
    const cache = createLruCache<string, number>(5, 500, clock);
    cache.set('1', 1);
    advanceTime(600);
    cache.set('2', 2);
    advanceTime(600);
    cache.set('3', 3);
    expect(cache.size).toBe(1);
  });
});
