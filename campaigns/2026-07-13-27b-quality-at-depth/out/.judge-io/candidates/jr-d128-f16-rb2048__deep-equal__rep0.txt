// FILE: impl.ts
export function deepEqual(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (a === null || b === null) return false;
  if (typeof a !== typeof b) return false;
  if (typeof a === 'number') return Number.isNaN(a) && Number.isNaN(b);
  if (typeof a !== 'object') return false;
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  if (Array.isArray(a)) return arrayEqual(a, b);
  return objectEqual(a, b);
}

function arrayEqual(a: unknown[], b: unknown[]): boolean {
  if (a.length !== b.length) return false;
  for (let i = 0; i < a.length; i++) {
    if (!deepEqual(a[i], b[i])) return false;
  }
  return true;
}

function objectEqual(
  a: Record<string, unknown>,
  b: Record<string, unknown>
): boolean {
  const keysA = Object.keys(a);
  const keysB = Object.keys(b);
  if (keysA.length !== keysB.length) return false;
  for (const key of keysA) {
    if (!keysB.includes(key)) return false;
    if (!deepEqual(a[key], b[key])) return false;
  }
  return true;
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import { deepEqual } from './impl';

describe('deepEqual', () => {
  describe('primitives', () => {
    it('should return true when both arguments are the same number', () => {
      expect(deepEqual(42, 42)).toBe(true);
    });

    it('should return true when both arguments are the same string', () => {
      expect(deepEqual('hello', 'hello')).toBe(true);
    });

    it('should return true when both arguments are the same boolean', () => {
      expect(deepEqual(true, true)).toBe(true);
    });

    it('should return true when both arguments are NaN', () => {
      expect(deepEqual(NaN, NaN)).toBe(true);
    });

    it('should return false when one argument is NaN and the other is a number', () => {
      expect(deepEqual(NaN, 0)).toBe(false);
      expect(deepEqual(0, NaN)).toBe(false);
    });

    it('should return false when arguments are different primitives', () => {
      expect(deepEqual(1, '1')).toBe(false);
      expect(deepEqual(true, 1)).toBe(false);
    });
  });

  describe('null and undefined', () => {
    it('should return true when both arguments are null', () => {
      expect(deepEqual(null, null)).toBe(true);
    });

    it('should return true when both arguments are undefined', () => {
      expect(deepEqual(undefined, undefined)).toBe(true);
    });

    it('should return false when null is compared to an empty object', () => {
      expect(deepEqual(null, {})).toBe(false);
      expect(deepEqual({}, null)).toBe(false);
    });

    it('should return false when null is compared to an empty array', () => {
      expect(deepEqual(null, [])).toBe(false);
      expect(deepEqual([], null)).toBe(false);
    });

    it('should return false when null is compared to undefined', () => {
      expect(deepEqual(null, undefined)).toBe(false);
    });
  });

  describe('arrays', () => {
    it('should return true when both arguments are empty arrays', () => {
      expect(deepEqual([], [])).toBe(true);
    });

    it('should return true when both arguments are deeply equal arrays', () => {
      expect(deepEqual([1, 2, [3, 4]], [1, 2, [3, 4]])).toBe(true);
    });

    it('should return false when array lengths differ', () => {
      expect(deepEqual([1, 2], [1, 2, 3])).toBe(false);
    });

    it('should return false when array elements differ at the same index', () => {
      expect(deepEqual([1, 2], [1, 3])).toBe(false);
    });
  });

  describe('objects', () => {
    it('should return true when both arguments are empty objects', () => {
      expect(deepEqual({}, {})).toBe(true);
    });

    it('should return true when both arguments are deeply equal objects', () => {
      expect(deepEqual({ a: 1, b: { c: 2 } }, { a: 1, b: { c: 2 } })).toBe(true);
    });

    it('should return false when key counts differ', () => {
      expect(deepEqual({ a: 1 }, { a: 1, b: 2 })).toBe(false);
    });

    it('should return false when objects have different keys', () => {
      expect(deepEqual({ a: 1 }, { b: 1 })).toBe(false);
    });

    it('should return false when values differ for the same key', () => {
      expect(deepEqual({ a: 1 }, { a: 2 })).toBe(false);
    });
  });

  describe('array vs object', () => {
    it('should return false when an array is compared to a plain object with matching indices', () => {
      expect(deepEqual([1, 2], { 0: 1, 1: 2 })).toBe(false);
      expect(deepEqual({ 0: 1, 1: 2 }, [1, 2])).toBe(false);
    });

    it('should return false when an empty array is compared to an empty object', () => {
      expect(deepEqual([], {})).toBe(false);
      expect(deepEqual({}, [])).toBe(false);
    });
  });

  describe('nested structures', () => {
    it('should return true for deeply nested arrays of objects', () => {
      const a = [{ x: [1, 2] }, { y: [3, 4] }];
      const b = [{ x: [1, 2] }, { y: [3, 4] }];
      expect(deepEqual(a, b)).toBe(true);
    });

    it('should return true for deeply nested objects with NaN values', () => {
      const a = { outer: { inner: NaN } };
      const b = { outer: { inner: NaN } };
      expect(deepEqual(a, b)).toBe(true);
    });

    it('should return false for nested structures that differ deep inside', () => {
      const a = { outer: { inner: [1, 2] } };
      const b = { outer: { inner: [1, 3] } };
      expect(deepEqual(a, b)).toBe(false);
    });
  });
});
