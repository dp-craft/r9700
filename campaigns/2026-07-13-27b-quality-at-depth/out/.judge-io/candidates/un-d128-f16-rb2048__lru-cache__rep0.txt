// FILE: impl.ts
import type { Clock } from '../../lib/clock';

interface Entry<V> {
  readonly value: V;
  readonly insertedAt: number;
  lastAccessedAt: number;
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

export function createLruCache<K, V>(capacity: number, ttlMs: number, clock: Clock): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();

  function isExpired(entry: Entry<V>): boolean {
    return clock.now() >= entry.insertedAt + ttlMs;
  }

  function refreshRecency(key: K, entry: Entry<V>): void {
    entry.lastAccessedAt = clock.now();
    store.delete(key);
    store.set(key, entry);
  }

  function evictExpired(): void {
    for (const [key, entry] of store) {
      if (isExpired(entry)) {
        store.delete(key);
      }
    }
  }

  function evictOldest(): void {
    evictExpired();
    const oldestKey = store.keys().next().value;
    if (oldestKey !== undefined) {
      store.delete(oldestKey);
    }
  }

  function countLive(): number {
    let count = 0;
    for (const entry of store.values()) {
      if (!isExpired(entry)) {
        count += 1;
      }
    }
    return count;
  }

  const cache: LruCache<K, V> = {
    get(key: K): V | undefined {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) {
        if (entry) {
          store.delete(key);
        }
        return undefined;
      }
      refreshRecency(key, entry);
      return entry.value;
    },

    set(key: K, value: V): void {
      const now = clock.now();
      if (store.has(key)) {
        store.delete(key);
      }
      if (countLive() >= capacity) {
        evictOldest();
      }
      store.set(key, { value, insertedAt: now, lastAccessedAt: now });
    },

    has(key: K): boolean {
      const entry = store.get(key);
      if (!entry || isExpired(entry)) {
        if (entry) {
          store.delete(key);
        }
        return false;
      }
      return true;
    },

    get size(): number {
      return countLive();
    },
  };

  return cache;
}

// FILE: impl.test.ts
import { beforeEach, describe, expect, it } from 'vitest';

import type { Clock } from '../../lib/clock';
import { createLruCache } from './impl';

function buildFakeClock(initialTime: number = 0): Clock & { tick(ms: number): void } {
  let now = initialTime;
  return {
    now: () => now,
    tick: (ms: number) => { now += ms; },
  };
}

describe('createLruCache', () => {
  let clock: Clock & { tick(ms: number): void };
  let cache: ReturnType<typeof createLruCache<string, number>>;

  beforeEach(() => {
    clock = buildFakeClock(0);
    cache = createLruCache(3, 100, clock);
  });

  it('should return undefined when key is absent', () => {
    expect(cache.get('missing')).toBeUndefined();
  });

  it('should store and retrieve a value within TTL', () => {
    cache.set('a', 1);
    expect(cache.get('a')).toBe(1);
  });

  it('should return undefined and remove entry when TTL expires', () => {
    cache.set('a', 1);
    clock.tick(100);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.size).toBe(0);
  });

  it('should report false for has() on expired entry without resurrecting', () => {
    cache.set('b', 2);
    clock.tick(100);
    expect(cache.has('b')).toBe(false);
    expect(cache.get('b')).toBeUndefined();
  });

  it('should refresh recency on get() and delay eviction', () => {
    cache.set('old', 1);
    clock.tick(50);
    cache.set('new', 2);
    cache.get('old'); // refreshes 'old'
    clock.tick(50);
    cache.set('evict', 3);
    expect(cache.get('new')).toBeUndefined();
    expect(cache.get('old')).toBe(1);
  });

  it('should evict least-recently-used live entry when capacity is exceeded', () => {
    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('c', 3);
    cache.set('d', 4);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('d')).toBe(4);
    expect(cache.size).toBe(3);
  });

  it('should skip expired entries during eviction and remove a live one', () => {
    cache.set('exp', 1);
    clock.tick(100);
    cache.set('live', 2);
    cache.set('overflow', 3);
    expect(cache.get('exp')).toBeUndefined();
    expect(cache.get('live')).toBe(2);
    expect(cache.size).toBe(1);
  });

  it('should overwrite existing key without counting toward capacity', () => {
    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('c', 3);
    cache.set('a', 10);
    expect(cache.get('a')).toBe(10);
    expect(cache.size).toBe(3);
  });

  it('should handle capacity edge of 1 correctly', () => {
    const singleCache = createLruCache(1, 100, clock);
    singleCache.set('x', 1);
    singleCache.set('y', 2);
    expect(singleCache.get('x')).toBeUndefined();
    expect(singleCache.get('y')).toBe(2);
  });
});
