// FILE: impl.ts
import type { Clock } from '../../lib/clock';

interface Entry<V> {
  value: V;
  expiresAt: number;
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

function isExpired(entry: Entry<unknown>, now: number): boolean {
  return now >= entry.expiresAt;
}

function createLruCache<K, V>(capacity: number, ttlMs: number, clock: Clock): LruCache<K, V> {
  const store = new Map<K, Entry<V>>();

  function purgeExpired(): void {
    const now = clock.now();
    const toDelete: K[] = [];
    for (const [key, entry] of store) {
      if (isExpired(entry, now)) {
        toDelete.push(key);
      }
    }
    for (const key of toDelete) {
      store.delete(key);
    }
  }

  function evictOldest(): void {
    const firstKey = store.keys().next().value;
    if (firstKey !== undefined) {
      store.delete(firstKey);
    }
  }

  function refreshRecency(key: K): void {
    const entry = store.get(key);
    if (entry !== undefined) {
      store.delete(key);
      store.set(key, entry);
    }
  }

  return {
    get(key: K): V | undefined {
      const now = clock.now();
      const entry = store.get(key);
      if (entry === undefined || isExpired(entry, now)) {
        store.delete(key);
        return undefined;
      }
      refreshRecency(key);
      return entry.value;
    },

    set(key: K, value: V): void {
      const now = clock.now();
      if (store.has(key)) {
        refreshRecency(key);
        store.set(key, { value, expiresAt: now + ttlMs });
        return;
      }
      purgeExpired();
      if (store.size >= capacity) {
        evictOldest();
      }
      store.set(key, { value, expiresAt: now + ttlMs });
    },

    has(key: K): boolean {
      const now = clock.now();
      const entry = store.get(key);
      if (entry === undefined || isExpired(entry, now)) {
        store.delete(key);
        return false;
      }
      return true;
    },

    get size(): number {
      const now = clock.now();
      let count = 0;
      for (const entry of store.values()) {
        if (!isExpired(entry, now)) {
          count += 1;
        }
      }
      return count;
    },
  };
}

export { createLruCache };

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import { createLruCache } from './impl';
import type { Clock } from '../../lib/clock';

const buildFakeClock = (initial: number = 0): Clock & { tick: (ms: number) => void } => {
  let time = initial;
  return {
    now: () => time,
    tick: (ms: number) => { time += ms; },
  };
};

describe('createLruCache', () => {
  describe('basic operations', () => {
    it('should return undefined for absent key when cache is empty', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      expect(cache.get('a')).toBeUndefined();
    });

    it('should return value for present key when within TTL', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      expect(cache.get('a')).toBe(1);
    });

    it('should return true for has when key exists and is within TTL', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      expect(cache.has('a')).toBe(true);
    });

    it('should return false for has when key is expired', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.tick(101);
      expect(cache.has('a')).toBe(false);
    });
  });

  describe('TTL and expiry', () => {
    it('should expire entry when clock.now exceeds insertion time plus ttlMs', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.tick(100);
      expect(cache.get('a')).toBeUndefined();
    });

    it('should not count expired entries towards size', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      clock.tick(101);
      expect(cache.size).toBe(0);
    });
  });

  describe('recency refresh', () => {
    it('should refresh recency on get when entry is accessed', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      cache.get('a');
      cache.set('c', 3);
      expect(cache.get('b')).toBeUndefined();
      expect(cache.get('a')).toBe(1);
    });

    it('should not resurrect expired entry on get', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.tick(101);
      expect(cache.get('a')).toBeUndefined();
    });
  });

  describe('eviction order', () => {
    it('should evict least recently used entry when capacity is exceeded', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      cache.set('c', 3);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
      expect(cache.get('c')).toBe(3);
    });

    it('should evict expired entries before evicting live entries when capacity is exceeded', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(3, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      cache.set('c', 3);
      clock.tick(101);
      cache.set('d', 4);
      expect(cache.size).toBe(1);
      expect(cache.get('d')).toBe(4);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBeUndefined();
      expect(cache.get('c')).toBeUndefined();
    });
  });

  describe('capacity edge', () => {
    it('should handle capacity 1 correctly', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(1, 1000, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
      expect(cache.size).toBe(1);
    });

    it('should update value and refresh TTL when set is called on existing key', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.tick(50);
      cache.set('a', 10);
      clock.tick(60);
      expect(cache.get('a')).toBe(10);
    });
  });
});
