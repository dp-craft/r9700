// FILE: impl.ts
export function deepEqual(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (typeof a !== typeof b) return false;
  if (a === null || b === null) return false;
  if (typeof a !== 'object') return Number.isNaN(a as number) && Number.isNaN(b as number);

  const aIsArray = Array.isArray(a);
  const bIsArray = Array.isArray(b);
  if (aIsArray !== bIsArray) return false;

  if (aIsArray) return compareArrays(a as readonly unknown[], b as readonly unknown[]);
  return compareObjects(a as Record<string, unknown>, b as Record<string, unknown>);
}

function compareArrays(a: readonly unknown[], b: readonly unknown[]): boolean {
  if (a.length !== b.length) return false;
  return a.every((val, idx) => deepEqual(val, b[idx]));
}

function compareObjects(a: Record<string, unknown>, b: Record<string, unknown>): boolean {
  const keysA = Object.keys(a);
  const keysB = Object.keys(b);
  if (keysA.length !== keysB.length) return false;
  return keysA.every(key => Object.hasOwn(b, key) && deepEqual(a[key], b[key]));
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import { deepEqual } from './impl';

describe('deepEqual', () => {
  it('should return true when both values are identical primitives', () => {
    expect(deepEqual(42, 42)).toBe(true);
    expect(deepEqual('hello', 'hello')).toBe(true);
    expect(deepEqual(true, true)).toBe(true);
    expect(deepEqual(false, false)).toBe(true);
  });

  it('should return true when both values are NaN', () => {
    expect(deepEqual(NaN, NaN)).toBe(true);
  });

  it('should return false when one value is NaN and the other is a number', () => {
    expect(deepEqual(NaN, 42)).toBe(false);
    expect(deepEqual(0, NaN)).toBe(false);
  });

  it('should return true when nested structures are structurally equal', () => {
    const a = { x: { y: [1, { z: true }] }, w: 'hello' };
    const b = { x: { y: [1, { z: true }] }, w: 'hello' };
    expect(deepEqual(a, b)).toBe(true);
  });

  it('should return false when nested structures differ in depth', () => {
    expect(deepEqual({ a: { b: 1 } }, { a: 1 })).toBe(false);
  });

  it('should return false when left object has more keys than right object', () => {
    expect(deepEqual({ a: 1, b: 2 }, { a: 1 })).toBe(false);
  });

  it('should return false when right object has more keys than left object', () => {
    expect(deepEqual({ a: 1 }, { a: 1, b: 2 })).toBe(false);
  });

  it('should return false when comparing an array to a plain object with matching indices', () => {
    expect(deepEqual([1, 2], { 0: 1, 1: 2 })).toBe(false);
    expect(deepEqual({ 0: 1, 1: 2 }, [1, 2])).toBe(false);
  });

  it('should return false when comparing null to an empty object', () => {
    expect(deepEqual(null, {})).toBe(false);
    expect(deepEqual({}, null)).toBe(false);
  });

  it('should return true when both values are null', () => {
    expect(deepEqual(null, null)).toBe(true);
  });

  it('should return false when array lengths differ', () => {
    expect(deepEqual([1, 2], [1, 2, 3])).toBe(false);
  });

  it('should return true when both values are empty arrays', () => {
    expect(deepEqual([], [])).toBe(true);
  });

  it('should return true when both values are empty objects', () => {
    expect(deepEqual({}, {})).toBe(true);
  });

  it('should return false when object values differ despite same keys', () => {
    expect(deepEqual({ a: 1 }, { a: 2 })).toBe(false);
  });
});
