import type { Clock } from '../../lib/clock';

interface CacheEntry<V> {
  readonly value: V;
  readonly expiresAt: number;
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

const isExpired = (entry: CacheEntry<unknown>, now: number): boolean =>
  now >= entry.expiresAt;

const refreshRecency = <K, V>(store: Map<K, CacheEntry<V>>, key: K): void => {
  const entry = store.get(key);
  if (entry !== undefined) {
    store.delete(key);
    store.set(key, entry);
  }
};

const evictOldest = <K, V>(store: Map<K, CacheEntry<V>>): void => {
  const oldestKey = store.keys().next().value;
  if (oldestKey !== undefined) {
    store.delete(oldestKey);
  }
};

export const createLruCache = <K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> => {
  const store = new Map<K, CacheEntry<V>>();

  return {
    get(key: K): V | undefined {
      const entry = store.get(key);
      if (entry === undefined || isExpired(entry, clock.now())) {
        store.delete(key);
        return undefined;
      }
      refreshRecency(store, key);
      return entry.value;
    },

    set(key: K, value: V): void {
      const now = clock.now();
      const existing = store.get(key);
      if (existing !== undefined) {
        store.set(key, { value, expiresAt: now + ttlMs });
        refreshRecency(store, key);
        return;
      }
      if (store.size >= capacity) {
        evictOldest(store);
      }
      store.set(key, { value, expiresAt: now + ttlMs });
    },

    has(key: K): boolean {
      const entry = store.get(key);
      return entry !== undefined && !isExpired(entry, clock.now());
    },

    get size(): number {
      const now = clock.now();
      let count = 0;
      for (const entry of store.values()) {
        if (!isExpired(entry, now)) count += 1;
      }
      return count;
    },
  };
};

// FILE: impl.test.ts
import { describe, expect, it, vi } from 'vitest';

import type { Clock } from '../../lib/clock';
import { createLruCache } from './impl';

const buildFakeClock = (initialTime: number = 0): Clock & { advance(ms: number): void } => {
  let time = initialTime;
  return {
    now: () => time,
    advance(ms: number): void { time += ms; },
  };
};

describe('createLruCache', () => {
  it('should return undefined when key is absent', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    expect(cache.get('missing')).toBeUndefined();
  });

  it('should return value when key is present and within TTL', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    expect(cache.get('a')).toBe(1);
  });

  it('should return undefined when key is present but expired', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    clock.advance(1000);
    expect(cache.get('a')).toBeUndefined();
  });

  it('should refresh recency on successful get', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.get('a'); // refreshes 'a'
    cache.set('c', 3); // evicts 'b' (oldest)
    expect(cache.get('b')).toBeUndefined();
    expect(cache.get('a')).toBe(1);
  });

  it('should evict least-recently-used entry when capacity is exceeded', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('c', 3);
    expect(cache.get('a')).toBeUndefined();
    expect(cache.get('b')).toBe(2);
    expect(cache.get('c')).toBe(3);
  });

  it('should return false for has when key is expired', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    clock.advance(1000);
    expect(cache.has('a')).toBe(false);
  });

  it('should return true for has when key is live', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    expect(cache.has('a')).toBe(true);
  });

  it('should not resurrect expired entries via has', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    clock.advance(1000);
    cache.has('a');
    expect(cache.get('a')).toBeUndefined();
  });

  it('should count only live entries in size', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(3, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    cache.set('c', 3);
    expect(cache.size).toBe(3);
    clock.advance(1000);
    expect(cache.size).toBe(0);
  });

  it('should handle capacity edge case with zero capacity', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(0, 1000, clock);
    cache.set('a', 1);
    expect(cache.size).toBe(0);
    expect(cache.get('a')).toBeUndefined();
  });

  it('should update value and reset TTL on set for existing key', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    clock.advance(500);
    cache.set('a', 2);
    clock.advance(500);
    expect(cache.get('a')).toBe(2);
  });

  it('should evict expired entries before capacity check during set', () => {
    const clock = buildFakeClock();
    const cache = createLruCache<string, number>(2, 1000, clock);
    cache.set('a', 1);
    cache.set('b', 2);
    clock.advance(1000);
    cache.set('c', 3);
    expect(cache.size).toBe(2);
    expect(cache.get('c')).toBe(3);
  });
});
