// FILE: impl.ts
export type Result =
  | { readonly ok: true; readonly value: number }
  | { readonly ok: false; readonly error: string };

interface ParserState {
  pos: number;
  input: string;
  error?: string;
}

function fail(state: ParserState, msg: string): Result {
  state.error = msg;
  return { ok: false, error: msg };
}

function skipWhitespace(state: ParserState): void {
  while (state.pos < state.input.length && /\s/.test(state.input[state.pos])) {
    state.pos++;
  }
}

function parseNumber(state: ParserState): Result {
  skipWhitespace(state);
  const start = state.pos;
  while (state.pos < state.input.length && /\d/.test(state.input[state.pos])) {
    state.pos++;
  }
  if (start === state.pos) {
    return fail(state, 'Expected number');
  }
  return { ok: true, value: Number(state.input.slice(start, state.pos)) };
}

function parseFactor(state: ParserState): Result {
  skipWhitespace(state);
  if (state.pos >= state.input.length) return fail(state, 'Unexpected end');
  const char = state.input[state.pos];
  if (char === '(') {
    state.pos++;
    const res = parseExpr(state);
    if (!res.ok) return res;
    skipWhitespace(state);
    if (state.pos >= state.input.length || state.input[state.pos] !== ')') {
      return fail(state, 'Missing closing parenthesis');
    }
    state.pos++;
    return res;
  }
  return parseNumber(state);
}

function parseTerm(state: ParserState): Result {
  let left = parseFactor(state);
  if (!left.ok) return left;
  while (true) {
    skipWhitespace(state);
    if (state.pos >= state.input.length) break;
    const op = state.input[state.pos];
    if (op !== '*' && op !== '/') break;
    state.pos++;
    const right = parseFactor(state);
    if (!right.ok) return right;
    if (op === '/' && right.value === 0) return fail(state, 'Division by zero');
    left = { ok: true, value: op === '*' ? left.value * right.value : left.value / right.value };
  }
  return left;
}

function parseExpr(state: ParserState): Result {
  let left = parseTerm(state);
  if (!left.ok) return left;
  while (true) {
    skipWhitespace(state);
    if (state.pos >= state.input.length) break;
    const op = state.input[state.pos];
    if (op !== '+' && op !== '-') break;
    state.pos++;
    const right = parseTerm(state);
    if (!right.ok) return right;
    left = { ok: true, value: op === '+' ? left.value + right.value : left.value - right.value };
  }
  return left;
}

export function evaluate(expr: string): Result {
  const state: ParserState = { pos: 0, input: expr };
  const res = parseExpr(state);
  if (!res.ok) return res;
  skipWhitespace(state);
  if (state.pos < state.input.length) {
    return fail(state, `Unexpected character: '${state.input[state.pos]}'`);
  }
  return res;
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import { evaluate, type Result } from './impl';

function assertOk(res: Result, expected: number): void {
  expect(res.ok).toBe(true);
  if (res.ok) expect(res.value).toBe(expected);
}

function assertFail(res: Result, msg: string): void {
  expect(res.ok).toBe(false);
  if (!res.ok) expect(res.error).toBe(msg);
}

describe('evaluate', () => {
  it('should return correct value for simple addition', () => {
    assertOk(evaluate('2 + 3'), 5);
  });

  it('should return correct value for simple subtraction', () => {
    assertOk(evaluate('10 - 4'), 6);
  });

  it('should return correct value for simple multiplication', () => {
    assertOk(evaluate('3 * 7'), 21);
  });

  it('should return correct value for simple division', () => {
    assertOk(evaluate('20 / 5'), 4);
  });

  it('should respect precedence when mixing operators', () => {
    assertOk(evaluate('2 + 3 * 4'), 14);
  });

  it('should evaluate left-associative subtraction correctly', () => {
    assertOk(evaluate('10 - 3 - 2'), 5);
  });

  it('should evaluate left-associative division correctly', () => {
    assertOk(evaluate('8 / 2 / 2'), 2);
  });

  it('should handle parentheses correctly', () => {
    assertOk(evaluate('(2 + 3) * 4'), 20);
  });

  it('should handle nested parentheses correctly', () => {
    assertOk(evaluate('((2 + 3) * (4 - 1)) / 5'), 3);
  });

  it('should ignore arbitrary whitespace', () => {
    assertOk(evaluate('  10   +   20  '), 30);
  });

  it('should return error on division by zero', () => {
    assertFail(evaluate('5 / 0'), 'Division by zero');
  });

  it('should return error on missing operand', () => {
    assertFail(evaluate('5 +'), 'Expected number');
  });

  it('should return error on unbalanced parentheses', () => {
    assertFail(evaluate('(5 + 3'), 'Missing closing parenthesis');
  });

  it('should return error on unexpected character', () => {
    assertFail(evaluate('10 + 5@'), "Unexpected character: '@'");
  });

  it('should return error on empty string', () => {
    assertFail(evaluate(''), 'Unexpected end');
  });

  it('should return error on only whitespace', () => {
    assertFail(evaluate('   '), 'Unexpected end');
  });
});
