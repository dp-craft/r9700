// FILE: impl.ts
export function deepEqual(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (a === null || b === null) return false;
  if (typeof a !== typeof b) return false;
  if (typeof a !== 'object') return a !== a && b !== b;
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  if (Array.isArray(a)) return arrayEqual(a, b as unknown[]);
  return objectEqual(a as Record<string, unknown>, b as Record<string, unknown>);
}

function arrayEqual(a: unknown[], b: unknown[]): boolean {
  if (a.length !== b.length) return false;
  return a.every((item, index) => deepEqual(item, b[index]));
}

function objectEqual(a: Record<string, unknown>, b: Record<string, unknown>): boolean {
  const keysA = Object.keys(a);
  const keysB = Object.keys(b);
  if (keysA.length !== keysB.length) return false;
  return keysA.every(key => Object.prototype.hasOwnProperty.call(b, key) && deepEqual(a[key], b[key]));
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';
import { deepEqual } from './impl';

describe('deepEqual', () => {
  describe('should return true when primitives compare by value', () => {
    it('should return true for equal numbers', () => {
      expect(deepEqual(42, 42)).toBe(true);
    });

    it('should return true for equal strings', () => {
      expect(deepEqual('hello', 'hello')).toBe(true);
    });

    it('should return true for equal booleans', () => {
      expect(deepEqual(true, true)).toBe(true);
    });

    it('should return true for the same reference', () => {
      const obj = { x: 1 };
      expect(deepEqual(obj, obj)).toBe(true);
    });
  });

  describe('should return false for mismatched primitives', () => {
    it('should return false for different numbers', () => {
      expect(deepEqual(1, 2)).toBe(false);
    });

    it('should return false for different string types', () => {
      expect(deepEqual(1, '1')).toBe(false);
    });

    it('should return false for null vs undefined', () => {
      expect(deepEqual(null, undefined)).toBe(false);
    });
  });

  describe('should treat NaN as equal to NaN', () => {
    it('should return true for NaN equals NaN', () => {
      expect(deepEqual(NaN, NaN)).toBe(true);
    });

    it('should return false for NaN vs a number', () => {
      expect(deepEqual(NaN, 42)).toBe(false);
    });

    it('should return true for objects containing NaN', () => {
      expect(deepEqual({ a: NaN }, { a: NaN })).toBe(true);
    });
  });

  describe('should handle null correctly', () => {
    it('should return false for null vs empty object', () => {
      expect(deepEqual(null, {})).toBe(false);
    });

    it('should return false for empty object vs null', () => {
      expect(deepEqual({}, null)).toBe(false);
    });

    it('should return true for null equals null', () => {
      expect(deepEqual(null, null)).toBe(true);
    });
  });

  describe('should compare nested structures recursively', () => {
    it('should return true for equal nested objects', () => {
      expect(deepEqual({ a: { b: 1 } }, { a: { b: 1 } })).toBe(true);
    });

    it('should return true for equal nested arrays', () => {
      expect(deepEqual([1, [2, 3]], [1, [2, 3]])).toBe(true);
    });

    it('should return true for mixed nested structures', () => {
      expect(deepEqual({ a: [1, { b: 2 }] }, { a: [1, { b: 2 }] })).toBe(true);
    });

    it('should return false for different nested values', () => {
      expect(deepEqual({ a: { b: 1 } }, { a: { b: 2 } })).toBe(false);
    });
  });

  describe('should detect key-count mismatch in objects', () => {
    it('should return false when left object has fewer keys', () => {
      expect(deepEqual({ a: 1 }, { a: 1, b: 2 })).toBe(false);
    });

    it('should return false when right object has fewer keys', () => {
      expect(deepEqual({ a: 1, b: 2 }, { a: 1 })).toBe(false);
    });
  });

  describe('should treat arrays as different from plain objects', () => {
    it('should return false for array vs object with matching indices', () => {
      expect(deepEqual([1, 2, 3], { '0': 1, '1': 2, '2': 3 })).toBe(false);
    });

    it('should return false for object vs array with matching indices', () => {
      expect(deepEqual({ '0': 1, '1': 2, '2': 3 }, [1, 2, 3])).toBe(false);
    });
  });

  describe('should compare arrays by length and element equality', () => {
    it('should return true for equal arrays', () => {
      expect(deepEqual([1, 2, 3], [1, 2, 3])).toBe(true);
    });

    it('should return false for arrays of different lengths', () => {
      expect(deepEqual([1, 2], [1, 2, 3])).toBe(false);
    });

    it('should return false for arrays with different elements', () => {
      expect(deepEqual([1, 2], [1, 3])).toBe(false);
    });

    it('should return true for empty arrays', () => {
      expect(deepEqual([], [])).toBe(true);
    });
  });

  describe('should handle edge cases', () => {
    it('should return true for empty objects', () => {
      expect(deepEqual({}, {})).toBe(true);
    });

    it('should return true for deeply nested empty structures', () => {
      expect(deepEqual({ a: { b: {} } }, { a: { b: {} } })).toBe(true);
    });

    it('should return false for object with missing nested key', () => {
      expect(deepEqual({ a: { b: 1 } }, { a: {} })).toBe(false);
    });
  });
});
