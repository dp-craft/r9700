// FILE: impl.ts
export function deepEqual(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (a === null || b === null) return false;
  if (typeof a !== typeof b) return false;
  if (typeof a !== 'object') return a !== a && b !== b;

  if (Array.isArray(a) && Array.isArray(b)) {
    return compareArrays(a, b);
  }

  if (!Array.isArray(a) && !Array.isArray(b)) {
    return compareObjects(a, b);
  }

  return false;
}

function compareArrays(a: unknown[], b: unknown[]): boolean {
  if (a.length !== b.length) return false;
  for (let i = 0; i < a.length; i++) {
    if (!deepEqual(a[i], b[i])) return false;
  }
  return true;
}

function compareObjects(a: Record<string, unknown>, b: Record<string, unknown>): boolean {
  const keysA = Object.keys(a);
  const keysB = Object.keys(b);
  if (keysA.length !== keysB.length) return false;
  for (const key of keysA) {
    if (!(key in b)) return false;
    if (!deepEqual(a[key], b[key])) return false;
  }
  return true;
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import { deepEqual } from './impl';

describe('deepEqual', () => {
  describe('primitives', () => {
    it('should return true when strings are identical', () => {
      expect(deepEqual('hello', 'hello')).toBe(true);
    });

    it('should return false when strings differ', () => {
      expect(deepEqual('hello', 'world')).toBe(false);
    });

    it('should return true when numbers are identical', () => {
      expect(deepEqual(42, 42)).toBe(true);
    });

    it('should return true when both values are NaN', () => {
      expect(deepEqual(NaN, NaN)).toBe(true);
    });

    it('should return false when one value is NaN and the other is a number', () => {
      expect(deepEqual(NaN, 42)).toBe(false);
    });

    it('should return true when booleans are identical', () => {
      expect(deepEqual(true, true)).toBe(true);
      expect(deepEqual(false, false)).toBe(true);
    });

    it('should return false when booleans differ', () => {
      expect(deepEqual(true, false)).toBe(false);
    });

    it('should return true when both values are null', () => {
      expect(deepEqual(null, null)).toBe(true);
    });

    it('should return true when both values are undefined', () => {
      expect(deepEqual(undefined, undefined)).toBe(true);
    });
  });

  describe('type mismatch', () => {
    it('should return false when comparing null to an empty object', () => {
      expect(deepEqual(null, {})).toBe(false);
    });

    it('should return false when comparing array to object with matching indices', () => {
      expect(deepEqual([1, 2], { 0: 1, 1: 2 })).toBe(false);
    });

    it('should return false when comparing string to number', () => {
      expect(deepEqual('42', 42)).toBe(false);
    });

    it('should return false when comparing number to boolean', () => {
      expect(deepEqual(1, true)).toBe(false);
    });
  });

  describe('arrays', () => {
    it('should return true when arrays have identical elements', () => {
      expect(deepEqual([1, 2, 3], [1, 2, 3])).toBe(true);
    });

    it('should return false when arrays have different lengths', () => {
      expect(deepEqual([1, 2], [1, 2, 3])).toBe(false);
    });

    it('should return false when arrays contain different elements', () => {
      expect(deepEqual([1, 2], [1, 3])).toBe(false);
    });

    it('should return true when nested arrays are structurally equal', () => {
      expect(deepEqual([[1, 2], [3, 4]], [[1, 2], [3, 4]])).toBe(true);
    });

    it('should return true when both arrays are empty', () => {
      expect(deepEqual([], [])).toBe(true);
    });
  });

  describe('objects', () => {
    it('should return true when objects have identical key-value pairs', () => {
      expect(deepEqual({ a: 1, b: 2 }, { a: 1, b: 2 })).toBe(true);
    });

    it('should return true when both objects are empty', () => {
      expect(deepEqual({}, {})).toBe(true);
    });

    it('should return false when objects have a key-count mismatch', () => {
      expect(deepEqual({ a: 1 }, { a: 1, b: 2 })).toBe(false);
    });

    it('should return false when objects share keys but differ in values', () => {
      expect(deepEqual({ a: 1 }, { a: 2 })).toBe(false);
    });

    it('should return true when nested objects are structurally equal', () => {
      expect(deepEqual({ a: { b: 1 } }, { a: { b: 1 } })).toBe(true);
    });

    it('should return true when deeply nested structures match', () => {
      const a = { x: [1, { y: [2, 3] }], z: null };
      const b = { x: [1, { y: [2, 3] }], z: null };
      expect(deepEqual(a, b)).toBe(true);
    });

    it('should return false when deeply nested structures diverge', () => {
      const a = { x: [1, { y: [2, 3] }] };
      const b = { x: [1, { y: [2, 4] }] };
      expect(deepEqual(a, b)).toBe(false);
    });
  });

  describe('mixed structures', () => {
    it('should return true when arrays of objects are structurally equal', () => {
      expect(deepEqual([{ a: 1 }], [{ a: 1 }])).toBe(true);
    });

    it('should return false when arrays of objects differ in nested values', () => {
      expect(deepEqual([{ a: 1 }], [{ a: 2 }])).toBe(false);
    });

    it('should return true when objects containing arrays are structurally equal', () => {
      expect(deepEqual({ arr: [1, 2] }, { arr: [1, 2] })).toBe(true);
    });

    it('should return true when structures contain NaN in nested positions', () => {
      expect(deepEqual({ x: NaN }, { x: NaN })).toBe(true);
    });

    it('should return true when structures contain null and undefined mixed', () => {
      expect(deepEqual({ a: null, b: undefined }, { a: null, b: undefined })).toBe(true);
    });
  });
});
