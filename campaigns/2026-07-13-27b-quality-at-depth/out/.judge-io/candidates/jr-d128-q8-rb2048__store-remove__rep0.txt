// FILE: store.ts
export interface Entry {
  readonly key: string;
  readonly amount: number;
}

export interface Store {
  add(entry: Entry): void;
  total(key: string): number;
  all(): readonly Entry[];
  remove(key: string): void;
}

class StoreImpl implements Store {
  private readonly entries: Entry[] = [];
  private readonly totals = new Map<string, number>();

  add(entry: Entry): void {
    this.entries.push(entry);
    this.totals.set(entry.key, (this.totals.get(entry.key) ?? 0) + entry.amount);
  }

  total(key: string): number {
    return this.totals.get(key) ?? 0;
  }

  all(): readonly Entry[] {
    return this.entries;
  }

  remove(key: string): void {
    if (!this.totals.has(key)) return;
    const filtered = this.entries.filter(e => e.key !== key);
    this.entries.length = 0;
    this.entries.push(...filtered);
    this.totals.set(key, 0);
  }
}

export function createStore(): Store {
  return new StoreImpl();
}

// FILE: store.test.ts
import { describe, it, expect } from "vitest";
import { createStore } from "./store";

describe("createStore", () => {
  it("should accumulate the cached total when entries share a key", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "a", amount: 3 });
    expect(s.total("a")).toBe(5);
  });

  describe("remove", () => {
    it("should remove all entries and reset total to 0 when key is removed", () => {
      const s = createStore();
      s.add({ key: "a", amount: 2 });
      s.add({ key: "a", amount: 3 });
      s.add({ key: "b", amount: 10 });
      
      s.remove("a");
      
      expect(s.total("a")).toBe(0);
      expect(s.total("b")).toBe(10);
      expect(s.all()).toEqual([{ key: "b", amount: 10 }]);
    });

    it("should be a no-op when the key has no entries", () => {
      const s = createStore();
      s.add({ key: "a", amount: 5 });
      
      s.remove("z");
      
      expect(s.total("a")).toBe(5);
      expect(s.all()).toEqual([{ key: "a", amount: 5 }]);
    });

    it("should remove all entries with the key even if amounts cancel out to 0", () => {
      const s = createStore();
      s.add({ key: "a", amount: 10 });
      s.add({ key: "a", amount: -10 });
      s.add({ key: "b", amount: 5 });
      
      s.remove("a");
      
      expect(s.total("a")).toBe(0);
      expect(s.total("b")).toBe(5);
      expect(s.all()).toEqual([{ key: "b", amount: 5 }]);
    });

    it("should leave an empty list and zero total when removing the only key", () => {
      const s = createStore();
      s.add({ key: "only", amount: 1 });
      
      s.remove("only");
      
      expect(s.total("only")).toBe(0);
      expect(s.all()).toEqual([]);
    });
  });
});
