// FILE: impl.ts
import type { Clock } from '../../lib/clock';

export interface RateLimiter {
  tryAcquire(key: string, tokens?: number): boolean;
}

const DEFAULT_TOKENS = 1;

interface BucketState {
  tokens: number;
  lastRefillAt: number;
}

export function createRateLimiter(
  capacity: number,
  refillPerSec: number,
  clock: Clock
): RateLimiter {
  const buckets = new Map<string, BucketState>();

  const refill = (state: BucketState): void => {
    const now = clock.now();
    const elapsedSeconds = (now - state.lastRefillAt) / 1000;
    const addedTokens = elapsedSeconds * refillPerSec;
    state.tokens = Math.min(capacity, state.tokens + addedTokens);
    state.lastRefillAt = now;
  };

  return {
    tryAcquire: (key: string, tokens: number = DEFAULT_TOKENS): boolean => {
      let state = buckets.get(key);
      if (state === undefined) {
        state = { tokens: capacity, lastRefillAt: clock.now() };
        buckets.set(key, state);
      }

      refill(state);

      if (state.tokens >= tokens) {
        state.tokens -= tokens;
        return true;
      }

      return false;
    },
  };
}

// FILE: impl.test.ts
import { describe, expect, it, vi } from 'vitest';
import type { Clock } from '../../lib/clock';
import { createRateLimiter } from './impl';

describe('createRateLimiter', () => {
  const makeFakeClock = (initialMs: number = 0): Clock => ({
    now: vi.fn(() => initialMs),
  });

  describe('first-burst', () => {
    it('should allow acquiring up to capacity when bucket is full', () => {
      const clock = makeFakeClock();
      const limiter = createRateLimiter(5, 1, clock);

      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(true);
    });

    it('should reject acquiring more than capacity without time passing', () => {
      const clock = makeFakeClock();
      const limiter = createRateLimiter(3, 10, clock);

      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(false);
    });
  });

  describe('lazy refill over time', () => {
    it('should refill tokens based on elapsed time and allow acquisition', () => {
      const clock = makeFakeClock(0);
      const limiter = createRateLimiter(5, 10, clock);

      limiter.tryAcquire('key-1', 5); // empty bucket

      // Advance time by 200ms -> 200ms * 10 tokens/sec = 2 tokens
      (clock.now as ReturnType<typeof vi.fn>).mockReturnValue(200);
      expect(limiter.tryAcquire('key-1', 2)).toBe(true);
    });

    it('should cap refilled tokens at capacity', () => {
      const clock = makeFakeClock(0);
      const limiter = createRateLimiter(5, 10, clock);

      // Advance time by 1 second -> 10 tokens, but cap is 5
      (clock.now as ReturnType<typeof vi.fn>).mockReturnValue(1000);
      expect(limiter.tryAcquire('key-1', 5)).toBe(true);
      expect(limiter.tryAcquire('key-1')).toBe(false);
    });
  });

  describe('empty-bucket rejection', () => {
    it('should return false and deduct nothing when insufficient tokens', () => {
      const clock = makeFakeClock();
      const limiter = createRateLimiter(2, 1, clock);

      limiter.tryAcquire('key-1', 2); // empty bucket
      expect(limiter.tryAcquire('key-1', 1)).toBe(false);

      // Advance time by 500ms -> 0.5 tokens
      (clock.now as ReturnType<typeof vi.fn>).mockReturnValue(500);
      expect(limiter.tryAcquire('key-1', 1)).toBe(false);

      // Advance time to 1500ms -> 1.5 tokens
      (clock.now as ReturnType<typeof vi.fn>).mockReturnValue(1500);
      expect(limiter.tryAcquire('key-1', 1)).toBe(true);
    });
  });

  describe('per-key isolation', () => {
    it('should maintain independent buckets for different keys', () => {
      const clock = makeFakeClock();
      const limiter = createRateLimiter(2, 1, clock);

      expect(limiter.tryAcquire('key-a', 2)).toBe(true);
      expect(limiter.tryAcquire('key-b', 2)).toBe(true);

      expect(limiter.tryAcquire('key-a', 1)).toBe(false);
      expect(limiter.tryAcquire('key-b', 1)).toBe(false);
    });

    it('should not affect other keys when one key is refilled by time', () => {
      const clock = makeFakeClock(0);
      const limiter = createRateLimiter(2, 10, clock);

      limiter.tryAcquire('key-a', 2); // empty key-a
      limiter.tryAcquire('key-b', 2); // empty key-b

      // Advance time by 200ms -> 2 tokens for both
      (clock.now as ReturnType<typeof vi.fn>).mockReturnValue(200);
      expect(limiter.tryAcquire('key-a', 2)).toBe(true);
      expect(limiter.tryAcquire('key-b', 2)).toBe(true);
    });
  });
});
