// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  return (key: K): Promise<V> => {
    const existing = cache.get(key);
    if (existing !== undefined) return existing;
    const p = fn(key);
    p.catch(() => cache.delete(key));
    cache.set(key, p);
    return p;
  };
}

// FILE: impl.test.ts
import { describe, expect, it, vi } from 'vitest';

import { createAsyncMemo } from './impl';

describe('createAsyncMemo', () => {
  it('should dedup concurrent calls when the same key is requested simultaneously', async () => {
    const fn = vi.fn(async (key: string): Promise<string> => {
      await new Promise(resolve => setTimeout(resolve, 10));
      return key;
    });
    const memoized = createAsyncMemo(fn);
    const p1 = memoized('a');
    const p2 = memoized('a');
    const [r1, r2] = await Promise.all([p1, p2]);
    expect(r1).toBe('a');
    expect(r2).toBe('a');
    expect(fn).toHaveBeenCalledTimes(1);
  });

  it('should retry fn when a previous call for the key rejects', async () => {
    let callCount = 0;
    const fn = vi.fn(async (): Promise<string> => {
      callCount++;
      if (callCount === 1) throw new Error('fail');
      return 'success';
    });
    const memoized = createAsyncMemo(fn);
    await expect(memoized('a')).rejects.toThrow('fail');
    const result = await memoized('a');
    expect(result).toBe('success');
    expect(fn).toHaveBeenCalledTimes(2);
  });

  it('should cache success when fn resolves and reuse the value on subsequent calls', async () => {
    const fn = vi.fn(async (key: string): Promise<string> => key);
    const memoized = createAsyncMemo(fn);
    const r1 = await memoized('b');
    const r2 = await memoized('b');
    expect(r1).toBe('b');
    expect(r2).toBe('b');
    expect(fn).toHaveBeenCalledTimes(1);
  });
});
