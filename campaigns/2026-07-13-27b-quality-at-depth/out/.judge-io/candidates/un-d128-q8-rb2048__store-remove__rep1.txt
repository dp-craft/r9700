// FILE: store.ts
export interface Entry {
  readonly key: string;
  readonly amount: number;
}

export interface Store {
  add(entry: Entry): void;
  remove(key: string): void;
  total(key: string): number;
  all(): readonly Entry[];
}

class StoreImpl implements Store {
  private readonly entries: Entry[] = [];
  private readonly totals = new Map<string, number>();

  add(entry: Entry): void {
    this.entries.push(entry);
    this.totals.set(entry.key, (this.totals.get(entry.key) ?? 0) + entry.amount);
  }

  remove(key: string): void {
    let i = this.entries.length;
    while (i-- > 0) {
      if (this.entries[i].key === key) {
        this.entries.splice(i, 1);
      }
    }
    this.totals.delete(key);
  }

  total(key: string): number {
    return this.totals.get(key) ?? 0;
  }

  all(): readonly Entry[] {
    return this.entries;
  }
}

export function createStore(): Store {
  return new StoreImpl();
}

// FILE: store.test.ts
import { describe, it, expect } from "vitest";
import { createStore } from "./store";

describe("createStore", () => {
  it("should accumulate the cached total when entries share a key", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "a", amount: 3 });
    expect(s.total("a")).toBe(5);
  });

  describe("remove", () => {
    it("should remove all matching entries when called with an existing key", () => {
      const s = createStore();
      s.add({ key: "a", amount: 1 });
      s.add({ key: "b", amount: 2 });
      s.add({ key: "a", amount: 3 });
      s.remove("a");
      expect(s.all().map(e => e.key)).toEqual(["b"]);
    });

    it("should reset the cached total to 0 when the key is removed", () => {
      const s = createStore();
      s.add({ key: "a", amount: 10 });
      s.remove("a");
      expect(s.total("a")).toBe(0);
    });

    it("should leave other keys totals unchanged when a different key is removed", () => {
      const s = createStore();
      s.add({ key: "a", amount: 5 });
      s.add({ key: "b", amount: 10 });
      s.remove("a");
      expect(s.total("b")).toBe(10);
    });

    it("should be a no-op when the key does not exist", () => {
      const s = createStore();
      s.add({ key: "a", amount: 1 });
      s.remove("z");
      expect(s.all().length).toBe(1);
      expect(s.total("a")).toBe(1);
    });
  });
});
