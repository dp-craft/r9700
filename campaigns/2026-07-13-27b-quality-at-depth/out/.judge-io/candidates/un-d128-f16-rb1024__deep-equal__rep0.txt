// FILE: impl.ts
export function deepEqual(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (a === null || b === null) return false;
  if (typeof a !== typeof b) return false;
  if (typeof a !== 'object') return a !== a && b !== b;

  const aIsArray = Array.isArray(a);
  const bIsArray = Array.isArray(b);
  if (aIsArray !== bIsArray) return false;

  if (aIsArray) return compareArrays(a as unknown[], b as unknown[]);
  return compareObjects(a as Record<string, unknown>, b as Record<string, unknown>);
}

function compareArrays(a: unknown[], b: unknown[]): boolean {
  if (a.length !== b.length) return false;
  for (let i = 0; i < a.length; i++) {
    if (!deepEqual(a[i], b[i])) return false;
  }
  return true;
}

function compareObjects(a: Record<string, unknown>, b: Record<string, unknown>): boolean {
  const keysA = Object.keys(a);
  const keysB = Object.keys(b);
  if (keysA.length !== keysB.length) return false;
  for (const key of keysA) {
    if (!(key in b)) return false;
    if (!deepEqual(a[key], b[key])) return false;
  }
  return true;
}

// FILE: impl.test.ts
import { describe, expect, it } from 'vitest';

import { deepEqual } from './impl';

describe('deepEqual', () => {
  it('should return true when both inputs are identical primitives', () => {
    expect(deepEqual(42, 42)).toBe(true);
    expect(deepEqual('hello', 'hello')).toBe(true);
    expect(deepEqual(true, true)).toBe(true);
    expect(deepEqual(false, false)).toBe(true);
    expect(deepEqual(undefined, undefined)).toBe(true);
  });

  it('should return false when primitives differ in value', () => {
    expect(deepEqual(1, 2)).toBe(false);
    expect(deepEqual('a', 'b')).toBe(false);
    expect(deepEqual(true, false)).toBe(false);
  });

  it('should return true when both inputs are NaN', () => {
    expect(deepEqual(NaN, NaN)).toBe(true);
  });

  it('should return false when one is NaN and the other is a number', () => {
    expect(deepEqual(NaN, 1)).toBe(false);
    expect(deepEqual(1, NaN)).toBe(false);
  });

  it('should return true when both inputs are null', () => {
    expect(deepEqual(null, null)).toBe(true);
  });

  it('should return false when null is compared to an empty object', () => {
    expect(deepEqual(null, {})).toBe(false);
    expect(deepEqual({}, null)).toBe(false);
  });

  it('should return true when both inputs are the same reference', () => {
    const obj = { x: 1 };
    expect(deepEqual(obj, obj)).toBe(true);
  });

  it('should return true when arrays have same length and deeply equal values', () => {
    expect(deepEqual([1, 2, 3], [1, 2, 3])).toBe(true);
    expect(deepEqual([], [])).toBe(true);
  });

  it('should return false when arrays differ in length', () => {
    expect(deepEqual([1, 2], [1, 2, 3])).toBe(false);
  });

  it('should return false when an array is compared to a plain object with matching indices', () => {
    expect(deepEqual([1, 2], { 0: 1, 1: 2 })).toBe(false);
    expect(deepEqual({ 0: 1, 1: 2 }, [1, 2])).toBe(false);
  });

  it('should return true when objects have same keys and deeply equal values', () => {
    expect(deepEqual({ a: 1, b: 2 }, { a: 1, b: 2 })).toBe(true);
    expect(deepEqual({}, {})).toBe(true);
  });

  it('should return false when objects have a key-count mismatch', () => {
    expect(deepEqual({ a: 1 }, { a: 1, b: 2 })).toBe(false);
    expect(deepEqual({ a: 1, b: 2 }, { a: 1 })).toBe(false);
  });

  it('should return false when objects have same keys but different values', () => {
    expect(deepEqual({ a: 1 }, { a: 2 })).toBe(false);
  });

  it('should return true for nested structures with matching shape and values', () => {
    const a = { x: [1, { y: 'hello', z: NaN }], w: true };
    const b = { x: [1, { y: 'hello', z: NaN }], w: true };
    expect(deepEqual(a, b)).toBe(true);
  });

  it('should return false for nested structures with differing values', () => {
    const a = { x: [1, { y: 'hello' }] };
    const b = { x: [1, { y: 'world' }] };
    expect(deepEqual(a, b)).toBe(false);
  });

  it('should return true when both inputs are arrays of objects', () => {
    expect(deepEqual([{ a: 1 }, { b: 2 }], [{ a: 1 }, { b: 2 }])).toBe(true);
  });

  it('should return false when object values differ in a deeply nested array', () => {
    const a = { items: [{ val: 1 }] };
    const b = { items: [{ val: 2 }] };
    expect(deepEqual(a, b)).toBe(false);
  });

  it('should return true for objects with nested null values', () => {
    expect(deepEqual({ a: null, b: { c: null } }, { a: null, b: { c: null } })).toBe(true);
  });

  it('should return false when one nested value is null and the other is an object', () => {
    expect(deepEqual({ a: null }, { a: {} })).toBe(false);
  });

  it('should return true for arrays containing NaN at the same position', () => {
    expect(deepEqual([NaN, 1], [NaN, 1])).toBe(true);
  });

  it('should return false for arrays containing NaN at different positions', () => {
    expect(deepEqual([NaN, 1], [1, NaN])).toBe(false);
  });

  it('should return true for deeply nested arrays', () => {
    expect(deepEqual([[1, [2, [3]]]], [[1, [2, [3]]]])).toBe(true);
  });

  it('should return false when nested array structures differ', () => {
    expect(deepEqual([[1, 2]], [[1], [2]])).toBe(false);
  });
});
