// FILE: impl.ts
import type { Clock } from '../../lib/clock';

interface CacheEntry<V> {
  value: V;
  expiresAt: number;
}

export interface LruCache<K, V> {
  get(key: K): V | undefined;
  set(key: K, value: V): void;
  has(key: K): boolean;
  readonly size: number;
}

function isExpired(entry: CacheEntry<unknown>, now: number): boolean {
  return now >= entry.expiresAt;
}

function purgeExpired<K, V>(store: Map<K, CacheEntry<V>>, now: number): void {
  for (const [key, entry] of store) {
    if (isExpired(entry, now)) {
      store.delete(key);
    }
  }
}

function evictOldest<K, V>(store: Map<K, CacheEntry<V>>): void {
  const oldestKey = store.keys().next().value;
  if (oldestKey !== undefined) {
    store.delete(oldestKey);
  }
}

function refreshRecency<K, V>(store: Map<K, CacheEntry<V>>, key: K): void {
  const entry = store.get(key);
  if (entry !== undefined) {
    store.delete(key);
    store.set(key, entry);
  }
}

export function createLruCache<K, V>(
  capacity: number,
  ttlMs: number,
  clock: Clock
): LruCache<K, V> {
  const store = new Map<K, CacheEntry<V>>();

  return {
    get(key: K): V | undefined {
      const now = clock.now();
      const entry = store.get(key);
      if (entry === undefined || isExpired(entry, now)) {
        store.delete(key);
        return undefined;
      }
      refreshRecency(store, key);
      return entry.value;
    },

    set(key: K, value: V): void {
      const now = clock.now();
      if (store.has(key)) {
        refreshRecency(store, key);
        store.set(key, { value, expiresAt: now + ttlMs });
        return;
      }
      purgeExpired(store, now);
      if (store.size >= capacity) {
        evictOldest(store);
      }
      store.set(key, { value, expiresAt: now + ttlMs });
    },

    has(key: K): boolean {
      const now = clock.now();
      const entry = store.get(key);
      if (entry === undefined || isExpired(entry, now)) {
        store.delete(key);
        return false;
      }
      return true;
    },

    get size(): number {
      const now = clock.now();
      let count = 0;
      for (const entry of store.values()) {
        if (!isExpired(entry, now)) {
          count += 1;
        }
      }
      return count;
    },
  };
}

// FILE: impl.test.ts
import { describe, expect, it, vi } from 'vitest';

import type { Clock } from '../../lib/clock';
import { createLruCache } from './impl';

function buildFakeClock(initialTime: number = 0): Clock & { advanceBy(ms: number): void } {
  let now = initialTime;
  return {
    now: () => now,
    advanceBy: (ms: number) => { now += ms; },
  };
}

describe('createLruCache', () => {
  describe('basic operations', () => {
    it('should return undefined for a key that was never set', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      expect(cache.get('a')).toBeUndefined();
    });

    it('should return the value for a recently set key', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      expect(cache.get('a')).toBe(1);
    });

    it('should report size as the count of live entries', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(3, 1000, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      expect(cache.size).toBe(2);
    });

    it('should return true for has when key exists and is not expired', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      expect(cache.has('a')).toBe(true);
    });

    it('should return false for has when key does not exist', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      expect(cache.has('missing')).toBe(false);
    });
  });

  describe('TTL expiry', () => {
    it('should return undefined when an entry expires', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.advanceBy(101);
      expect(cache.get('a')).toBeUndefined();
    });

    it('should remove expired entries from size count', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      clock.advanceBy(101);
      expect(cache.size).toBe(1);
    });

    it('should return false for has when an entry expires', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.advanceBy(101);
      expect(cache.has('a')).toBe(false);
    });

    it('should not resurrect an expired entry via has', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      clock.advanceBy(101);
      expect(cache.has('a')).toBe(false);
      expect(cache.has('a')).toBe(false);
    });
  });

  describe('recency refresh', () => {
    it('should refresh recency on a successful get', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      cache.get('a');
      cache.set('c', 3);
      expect(cache.get('a')).toBe(1);
      expect(cache.get('b')).toBeUndefined();
    });

    it('should refresh recency when updating an existing key via set', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      cache.set('a', 10);
      cache.set('c', 3);
      expect(cache.get('a')).toBe(10);
      expect(cache.get('b')).toBeUndefined();
    });
  });

  describe('LRU eviction', () => {
    it('should evict the least recently used entry when capacity is exceeded', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 1000, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      cache.set('c', 3);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
      expect(cache.get('c')).toBe(3);
    });

    it('should evict expired entries before evicting live entries', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(2, 100, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      clock.advanceBy(101);
      cache.set('c', 3);
      expect(cache.size).toBe(2);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBeUndefined();
      expect(cache.get('c')).toBe(3);
    });

    it('should handle capacity edge case of 1 correctly', () => {
      const clock = buildFakeClock();
      const cache = createLruCache<string, number>(1, 1000, clock);
      cache.set('a', 1);
      cache.set('b', 2);
      expect(cache.size).toBe(1);
      expect(cache.get('a')).toBeUndefined();
      expect(cache.get('b')).toBe(2);
    });
  });
});
