// FILE: impl.ts
export function createAsyncMemo<K, V>(fn: (key: K) => Promise<V>): (key: K) => Promise<V> {
  const cache = new Map<K, Promise<V>>();
  return (key: K): Promise<V> => {
    const cached = cache.get(key);
    if (cached !== undefined) {
      return cached;
    }
    const p = fn(key).catch((error: unknown): never => {
      cache.delete(key);
      throw error;
    });
    cache.set(key, p);
    return p;
  };
}

// FILE: impl.test.ts
import { describe, it, expect, vi } from 'vitest';
import { createAsyncMemo } from './impl';

describe('createAsyncMemo', () => {
  it('should deduplicate concurrent calls when called concurrently with the same key', async () => {
    const fn = vi.fn(async (key: string): Promise<number> => {
      await new Promise(r => setTimeout(r, 5));
      return parseInt(key, 10);
    });
    const memoized = createAsyncMemo(fn);

    const [r1, r2, r3] = await Promise.all([
      memoized('42'),
      memoized('42'),
      memoized('42'),
    ]);

    expect(r1).toBe(42);
    expect(r2).toBe(42);
    expect(r3).toBe(42);
    expect(fn).toHaveBeenCalledTimes(1);
  });

  it('should retry the computation when it rejects', async () => {
    let attempt = 0;
    const fn = vi.fn(async (key: string): Promise<number> => {
      attempt++;
      if (attempt === 1) throw new Error('fail');
      return parseInt(key, 10);
    });
    const memoized = createAsyncMemo(fn);

    await expect(memoized('42')).rejects.toThrow('fail');
    const result = await memoized('42');
    expect(result).toBe(42);
    expect(fn).toHaveBeenCalledTimes(2);
  });

  it('should cache the result when the computation succeeds', async () => {
    const fn = vi.fn(async (key: string): Promise<number> => {
      return parseInt(key, 10);
    });
    const memoized = createAsyncMemo(fn);

    const r1 = await memoized('42');
    const r2 = await memoized('42');
    expect(r1).toBe(42);
    expect(r2).toBe(42);
    expect(fn).toHaveBeenCalledTimes(1);
  });
});
