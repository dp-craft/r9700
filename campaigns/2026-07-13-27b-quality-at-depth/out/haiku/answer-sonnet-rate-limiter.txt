// FILE: impl.ts
import type { Clock } from "../../lib/clock";

const MILLISECONDS_PER_SECOND = 1000;
const DEFAULT_TOKENS = 1;

export interface RateLimiter {
  tryAcquire(key: string, tokens?: number): boolean;
}

interface Bucket {
  readonly tokens: number;
  readonly lastRefillMs: number;
}

interface AcquireResult {
  readonly bucket: Bucket;
  readonly acquired: boolean;
}

function createFullBucket(capacity: number, nowMs: number): Bucket {
  return { tokens: capacity, lastRefillMs: nowMs };
}

function refillBucket(
  bucket: Bucket,
  capacity: number,
  refillPerSec: number,
  nowMs: number,
): Bucket {
  const elapsedSeconds = (nowMs - bucket.lastRefillMs) / MILLISECONDS_PER_SECOND;
  const replenished = bucket.tokens + elapsedSeconds * refillPerSec;
  return {
    tokens: Math.min(capacity, replenished),
    lastRefillMs: nowMs,
  };
}

function acquireFromBucket(bucket: Bucket, tokens: number): AcquireResult {
  if (bucket.tokens < tokens) {
    return { bucket, acquired: false };
  }
  return {
    bucket: { tokens: bucket.tokens - tokens, lastRefillMs: bucket.lastRefillMs },
    acquired: true,
  };
}

export function createRateLimiter(
  capacity: number,
  refillPerSec: number,
  clock: Clock,
): RateLimiter {
  const buckets = new Map<string, Bucket>();

  function getOrCreateBucket(key: string, nowMs: number): Bucket {
    const existing = buckets.get(key);
    return existing ?? createFullBucket(capacity, nowMs);
  }

  function tryAcquire(key: string, tokens: number = DEFAULT_TOKENS): boolean {
    const nowMs = clock.now();
    const current = getOrCreateBucket(key, nowMs);
    const refilled = refillBucket(current, capacity, refillPerSec, nowMs);
    const result = acquireFromBucket(refilled, tokens);
    buckets.set(key, result.bucket);
    return result.acquired;
  }

  return { tryAcquire };
}

// FILE: impl.test.ts
import { describe, expect, it } from "vitest";
import type { Clock } from "../../lib/clock";
import { createRateLimiter } from "./impl";

interface FakeClock extends Clock {
  advance(ms: number): void;
}

function createFakeClock(initialMs: number): FakeClock {
  let currentMs = initialMs;
  return {
    now: (): number => currentMs,
    advance: (ms: number): void => {
      currentMs += ms;
    },
  };
}

describe("createRateLimiter", () => {
  it("should allow acquiring up to capacity when bucket starts full", () => {
    const clock = createFakeClock(0);
    const limiter = createRateLimiter(3, 1, clock);

    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(true);
  });

  it("should reject acquire when bucket is empty and no time has passed", () => {
    const clock = createFakeClock(0);
    const limiter = createRateLimiter(2, 1, clock);

    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(false);
  });

  it("should not deduct tokens when acquire is rejected", () => {
    const clock = createFakeClock(0);
    const limiter = createRateLimiter(1, 1, clock);

    expect(limiter.tryAcquire("user-1", 5)).toBe(false);
    expect(limiter.tryAcquire("user-1", 1)).toBe(true);
  });

  it("should lazily refill tokens based on elapsed time", () => {
    const clock = createFakeClock(0);
    const limiter = createRateLimiter(5, 1, clock);

    for (let i = 0; i < 5; i += 1) {
      expect(limiter.tryAcquire("user-1")).toBe(true);
    }
    expect(limiter.tryAcquire("user-1")).toBe(false);

    clock.advance(3000);

    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(false);
  });

  it("should cap refilled tokens at capacity when a long time has passed", () => {
    const clock = createFakeClock(0);
    const limiter = createRateLimiter(3, 10, clock);

    expect(limiter.tryAcquire("user-1", 3)).toBe(true);

    clock.advance(1_000_000);

    expect(limiter.tryAcquire("user-1", 3)).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(false);
  });

  it("should isolate buckets between different keys", () => {
    const clock = createFakeClock(0);
    const limiter = createRateLimiter(2, 1, clock);

    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(false);

    expect(limiter.tryAcquire("user-2")).toBe(true);
    expect(limiter.tryAcquire("user-2")).toBe(true);
    expect(limiter.tryAcquire("user-2")).toBe(false);
  });

  it("should default tokens to 1 when not specified", () => {
    const clock = createFakeClock(0);
    const limiter = createRateLimiter(1, 1, clock);

    expect(limiter.tryAcquire("user-1")).toBe(true);
    expect(limiter.tryAcquire("user-1")).toBe(false);
  });

  it("should allow acquiring multiple tokens at once when sufficient are available", () => {
    const clock = createFakeClock(0);
    const limiter = createRateLimiter(5, 1, clock);

    expect(limiter.tryAcquire("user-1", 4)).toBe(true);
    expect(limiter.tryAcquire("user-1", 2)).toBe(false);
    expect(limiter.tryAcquire("user-1", 1)).toBe(true);
  });
});
