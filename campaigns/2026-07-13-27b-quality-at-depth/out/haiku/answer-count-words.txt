// FILE: impl.ts
import { normalizeToken } from "../../lib/tokenize";

export function countWords(text: string): ReadonlyMap<string, number> {
  const tokens = text.split(/\s+/);
  const counts = new Map<string, number>();

  for (const token of tokens) {
    const normalized = normalizeToken(token);
    if (normalized.length > 0) {
      counts.set(normalized, (counts.get(normalized) ?? 0) + 1);
    }
  }

  return counts;
}

// FILE: impl.test.ts
import { describe, it, expect } from "vitest";
import { countWords } from "./impl";

describe("countWords", () => {
  it("should return empty map when text is empty", () => {
    // Arrange
    const text = "";

    // Act
    const result = countWords(text);

    // Assert
    expect(result.size).toBe(0);
  });

  it("should count single word occurrence", () => {
    // Arrange
    const text = "hello";

    // Act
    const result = countWords(text);

    // Assert
    expect(result.get("hello")).toBe(1);
  });

  it("should count multiple words with correct occurrences", () => {
    // Arrange
    const text = "hello world hello";

    // Act
    const result = countWords(text);

    // Assert
    expect(result.get("hello")).toBe(2);
    expect(result.get("world")).toBe(1);
  });

  it("should normalize tokens when counting", () => {
    // Arrange
    const text = "Hello HELLO hello";

    // Act
    const result = countWords(text);

    // Assert
    expect(result.get("hello")).toBe(3);
  });

  it("should drop punctuation during normalization", () => {
    // Arrange
    const text = "hello, hello! hello?";

    // Act
    const result = countWords(text);

    // Assert
    expect(result.get("hello")).toBe(3);
  });

  it("should ignore multiple whitespace separators", () => {
    // Arrange
    const text = "hello    world";

    // Act
    const result = countWords(text);

    // Assert
    expect(result.size).toBe(2);
    expect(result.get("hello")).toBe(1);
    expect(result.get("world")).toBe(1);
  });

  it("should return ReadonlyMap instance", () => {
    // Arrange
    const text = "test";

    // Act
    const result = countWords(text);

    // Assert
    expect(result instanceof Map).toBe(true);
  });
});
