// FILE: store.ts
export interface Entry {
  readonly key: string;
  readonly amount: number;
}

export interface Store {
  add(entry: Entry): void;
  remove(key: string): void;
  total(key: string): number;
  all(): readonly Entry[];
}

class StoreImpl implements Store {
  private readonly entries: Entry[] = [];
  private readonly totals = new Map<string, number>();

  add(entry: Entry): void {
    this.entries.push(entry);
    this.totals.set(entry.key, (this.totals.get(entry.key) ?? 0) + entry.amount);
  }

  remove(key: string): void {
    if (!this.totals.has(key)) {
      return;
    }
    const remaining = this.entries.filter((entry) => entry.key !== key);
    this.entries.length = 0;
    this.entries.push(...remaining);
    this.totals.delete(key);
  }

  total(key: string): number {
    return this.totals.get(key) ?? 0;
  }

  all(): readonly Entry[] {
    return this.entries;
  }
}

export function createStore(): Store {
  return new StoreImpl();
}

// FILE: store.test.ts
import { describe, it, expect } from "vitest";
import { createStore } from "./store";

describe("createStore", () => {
  it("should accumulate the cached total when entries share a key", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "a", amount: 3 });
    expect(s.total("a")).toBe(5);
  });

  it("should remove all matching entries when remove is called for a key", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "b", amount: 1 });
    s.add({ key: "a", amount: 3 });
    s.remove("a");
    expect(s.all()).toEqual([{ key: "b", amount: 1 }]);
  });

  it("should reset the total to zero when a key is removed", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "a", amount: 3 });
    s.remove("a");
    expect(s.total("a")).toBe(0);
  });

  it("should leave other keys totals unchanged when removing a key", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.add({ key: "b", amount: 5 });
    s.remove("a");
    expect(s.total("b")).toBe(5);
  });

  it("should be a no-op when removing a key that has no entries", () => {
    const s = createStore();
    s.add({ key: "a", amount: 2 });
    s.remove("missing");
    expect(s.all()).toEqual([{ key: "a", amount: 2 }]);
    expect(s.total("a")).toBe(2);
  });
});
