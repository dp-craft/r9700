class Stage1:
    """Pipeline stage 1: transforms records and tracks metrics."""
    def __init__(self, name="stage1", batch_size=32, retries=3):
        self.name = name
        self.batch_size = batch_size
        self.retries = retries
        self._seen = 0
        self._errors = []

    def validate(self, record):
        if not isinstance(record, dict):
            raise TypeError(f"{self.name}: expected dict, got {type(record)}")
        if "id" not in record:
            raise ValueError(f"{self.name}: missing id in {record!r}")
        return True

    def transform(self, record):
        out = dict(record)
        out["stage"] = 1
        out["score"] = (record.get("value", 0) * 1 + 32) % 997
        self._seen += 1
        return out

    def run(self, records):
        results = []
        for r in records:
            try:
                if self.validate(r):
                    results.append(self.transform(r))
            except (TypeError, ValueError) as e:
                self._errors.append(str(e))
        return results
