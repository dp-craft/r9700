## Corrected D2 + D3 (tools/dpf/src/screens/preflight.ts)

```ts
import path from 'node:path';

import {
  HIDDEN_LEAD,
  hiddenRef,
  NO_CONTENT_LEAD,
  NOT_A_FILE_LEAD,
  notAFileRef,
  REFUSED_LEAD,
  refusedRef,
  requestDocRefs
} from '../../../shared/request-refs.js';
import type { Screen } from '../ui/screen.js';

const KB = 1024;
const BARE_FILE_RE = /\.[\p{L}\p{N}_-]+$/u;
const EDGE_PUNCTUATION_RE = /^[.,;:!?]+|[.,;:!?]+$/gu;
const CONTINUE_PROMPT = 'c to continue, e to edit request.txt, q to quit';
const BLOCKED_PROMPT = 'e to edit request.txt, q to quit';

export interface PreflightFsDeps {
  readonly exists: (full: string) => boolean;
  readonly size: (full: string) => number;
}

export type PreflightResult = 'continue' | 'edit' | 'quit';

interface RefLine {
  readonly line: string;
  readonly blocking: boolean;
}

const UNREAD: readonly (readonly [(ref: string) => boolean, string])[] = [
  [refusedRef, REFUSED_LEAD],
  [hiddenRef, HIDDEN_LEAD],
  [notAFileRef, NOT_A_FILE_LEAD],
];

const KEYS: Readonly<Record<string, PreflightResult>> = { c: 'continue', e: 'edit', q: 'quit' };

const refLine = (ref: string, root: string, fsDeps: PreflightFsDeps): RefLine => {
  const unread = UNREAD.find(([test]) => test(ref));
  if (unread !== undefined) return { line: `${unread[1]}${ref}`, blocking: true };
  const full = path.join(root, ref);
  return fsDeps.exists(full) && fsDeps.size(full) > 0
    ? { line: `${ref} (${String(Math.round(fsDeps.size(full) / KB))} KB)`, blocking: false }
    : { line: `${NO_CONTENT_LEAD}${ref}`, blocking: true };
};

const bareWarnings = (text: string, root: string, fsDeps: PreflightFsDeps): readonly string[] =>
  [...new Set(text.split(/\s+/u).filter(token => !token.startsWith('@')).map(token => token.replace(EDGE_PUNCTUATION_RE, '')))]
    .filter(bare => bare.includes('/') && BARE_FILE_RE.test(bare) && fsDeps.exists(path.join(root, bare)))
    .map(bare => `warn: ${bare} names an existing file (no @)`);

export const preflightLines = (
  text: string,
  root: string,
  fsDeps: PreflightFsDeps
): { lines: string[]; blocking: boolean } => {
  const refs = requestDocRefs(text).map(ref => refLine(ref, root, fsDeps));
  return {
    lines: [...refs.map(entry => entry.line), ...bareWarnings(text, root, fsDeps)],
    blocking: refs.some(entry => entry.blocking),
  };
};

const promptFor = (blocking: boolean): string => (blocking ? BLOCKED_PROMPT : CONTINUE_PROMPT);

const resultFor = (line: string, blocking: boolean): PreflightResult | undefined => {
  const result = KEYS[line];
  return blocking && result === 'continue' ? undefined : result;
};

export const preflightScreen = (lines: readonly string[], blocking: boolean): Screen<boolean, PreflightResult> => ({
  start: { state: blocking, out: [...lines, promptFor(blocking)] },
  next: (state, line) => {
    const result = resultFor(line, state);
    return result === undefined ? { state, out: [...lines, promptFor(state)] } : { state, out: [], result };
  },
});
```

import { describe, expect, it } from 'vitest';

import { preflightLines, preflightScreen } from '../../src/screens/preflight.js';

const ROOT = '/repo';
const SIZES: Readonly<Record<string, number>> = { '/repo/docs/good.md': 2048, '/repo/docs/roadmap.md': 4096, '/repo/docs/empty.md': 0 };
const fsDeps = { exists: (full: string) => full in SIZES, size: (full: string) => SIZES[full] ?? 0 };
const CONTINUE = 'c to continue, e to edit request.txt, q to quit';

describe('preflightLines (C7)', () => {
  it('names each ref that will not be read, and warns on a bare existing path', () => {
    expect(preflightLines('read @docs/missing.md and @docs/.hidden.md and docs/roadmap.md and @docs/good.md', ROOT, fsDeps)).toEqual({
      lines: [
        'no content (missing, empty or unreadable): docs/missing.md',
        'refused (hidden file): docs/.hidden.md',
        'docs/good.md (2 KB)',
        'warn: docs/roadmap.md names an existing file (no @)',
      ],
      blocking: true,
    });
  });

  it('passes a request whose refs are all readable', () => {
    expect(preflightLines('read @docs/good.md', ROOT, fsDeps)).toEqual({ lines: ['docs/good.md (2 KB)'], blocking: false });
  });

  it('refuses a ref outside the root, one that names no file, and an empty file', () => {
    expect(preflightLines('@../x.md @docs/plans @docs/empty.md', ROOT, fsDeps)).toEqual({
      lines: [
        'refused (outside the project root): ../x.md',
        'not read (names no file — a document ref carries its extension): docs/plans',
        'no content (missing, empty or unreadable): docs/empty.md',
      ],
      blocking: true,
    });
  });
});

describe('preflightScreen (C7)', () => {
  it('offers continue, edit and quit when nothing blocks', () => {
    const screen = preflightScreen(['docs/good.md (2 KB)'], false);
    expect(screen.start.out).toEqual(['docs/good.md (2 KB)', CONTINUE]);
    expect(screen.next(false, 'c')).toEqual({ state: false, out: [], result: 'continue' });
    expect(screen.next(false, 'e')).toEqual({ state: false, out: [], result: 'edit' });
    expect(screen.next(false, 'q')).toEqual({ state: false, out: [], result: 'quit' });
    expect(screen.next(false, 'x')).toEqual({ state: false, out: ['docs/good.md (2 KB)', CONTINUE] });
  });

  it('does not continue past a ref that will not be read', () => {
    const screen = preflightScreen(['refused (hidden file): .env'], true);
    expect(screen.start.out).toEqual(['refused (hidden file): .env', 'e to edit request.txt, q to quit']);
    expect(screen.next(true, 'c')).toEqual({ state: true, out: ['refused (hidden file): .env', 'e to edit request.txt, q to quit'] });
    expect(screen.next(true, 'e')).toEqual({ state: true, out: [], result: 'edit' });
  });
});
